If you've built or used an i18n system, you've already built a vocabulary. `{ "greeting": "Hello", "farewell": "Goodbye" }` maps keys to values. An ML vocabulary maps tokens to indices: `{ "patient": 0, "reports": 1, "pain": 2 }`.

## Building the Dictionary

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="vocabulary-builder.ts">
{`class Vocabulary {
  private tokenToId: Map<string, number>;
  private idToToken: string[];

  constructor() {
    this.tokenToId = new Map();
    this.idToToken = [];

    // Special tokens always come first
    this.addToken('<PAD>');   // 0: padding for fixed-length sequences
    this.addToken('<UNK>');   // 1: unknown/out-of-vocabulary words
    this.addToken('<BOS>');   // 2: beginning of sequence
    this.addToken('<EOS>');   // 3: end of sequence
  }

  addToken(token: string): number {
    if (this.tokenToId.has(token)) return this.tokenToId.get(token)!;
    const id = this.idToToken.length;
    this.tokenToId.set(token, id);
    this.idToToken.push(token);
    return id;
  }

  encode(token: string): number {
    return this.tokenToId.get(token) ?? 1; // 1 = <UNK>
  }

  decode(id: number): string {
    return this.idToToken[id] ?? '<UNK>';
  }

  get size(): number { return this.idToToken.length; }
}

const vocab = new Vocabulary();
vocab.addToken('patient');  // 4
vocab.addToken('reports');  // 5
vocab.addToken('pain');     // 6

console.log(vocab.encode('patient')); // 4
console.log(vocab.decode(4));         // 'patient'
console.log(vocab.encode('unknown')); // 1 (<UNK>)
console.log(vocab.size);              // 7`}
</CodeBlock>

## Frequency-Based Vocabulary

In practice, you limit vocabulary size to the most frequent tokens. Rare words become `<UNK>`.

<CodeBlock language="typescript" title="frequency-vocab.ts">
{`function buildVocabFromCorpus(
  corpus: string[][],  // array of tokenized documents
  maxSize: number = 5000
): Vocabulary {
  // Count frequencies
  const freq = new Map<string, number>();
  for (const doc of corpus) {
    for (const token of doc) {
      freq.set(token, (freq.get(token) ?? 0) + 1);
    }
  }

  // Sort by frequency (descending) and take top N
  const sorted = [...freq.entries()]
    .sort((a, b) => b[1] - a[1])
    .slice(0, maxSize - 4); // reserve 4 for special tokens

  const vocab = new Vocabulary();
  for (const [token] of sorted) {
    vocab.addToken(token);
  }

  console.log('Vocabulary size:', vocab.size);
  console.log('Most common:', sorted.slice(0, 5).map(([t, c]) => t + ':' + c));
  return vocab;
}`}
</CodeBlock>
