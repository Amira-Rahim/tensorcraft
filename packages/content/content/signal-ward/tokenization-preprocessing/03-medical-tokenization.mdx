Three spellings of one drug is a Map lookup, one line, done. `RA` is not. In a rheumatology note it is rheumatoid arthritis; in a cardiology note it is right atrium; a Map has room for exactly one answer. Pick wrong and the record now claims a chronic autoimmune disease, or a heart chamber. Every normalization rule from here is a bet that context does not matter, and the ambiguous entries are where that bet loses.

## The Abbreviation Problem

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="medical-abbreviations.ts">
{`// Common medical abbreviations
const MEDICAL_ABBREVS = new Map<string, string>([
  ['pt', 'patient'],
  ['pts', 'patients'],
  ['hx', 'history'],
  ['dx', 'diagnosis'],
  ['rx', 'prescription'],
  ['tx', 'treatment'],
  ['sx', 'symptoms'],
  ['pcn', 'penicillin'],
  ['pen', 'penicillin'],
  ['abx', 'antibiotics'],
  ['bp', 'blood-pressure'],
  ['hr', 'heart-rate'],
  ['c/o', 'complains-of'],
  ['s/p', 'status-post'],
  ['w/o', 'without'],
  ['prn', 'as-needed'],
  ['bid', 'twice-daily'],
  ['tid', 'three-times-daily'],
]);

function normalizeToken(token: string): string {
  return MEDICAL_ABBREVS.get(token) ?? token;
}

function medicalTokenize(text: string): string[] {
  const prepared = text.toLowerCase();

  // c/o, s/p and w/o survive the match below because of the (?:\\/[a-z])? branch
  const tokens = prepared.match(/[a-z]+(?:\\/[a-z])?|[0-9]+(?:\\.[0-9]+)?/g) || [];
  return tokens.map(normalizeToken);
}

console.log(medicalTokenize('Pt c/o chest pain. Hx of PCN allergy.'));
// ['patient', 'complains-of', 'chest', 'pain', 'history', 'of', 'penicillin', 'allergy']`}
</CodeBlock>

## Spelling Normalization

Clinical notes are often typed quickly. Common misspellings need handling too.

<CodeBlock language="typescript" title="spelling-normalization.ts">
{`const COMMON_MISSPELLINGS = new Map<string, string>([
  ['patinet', 'patient'],
  ['diagnossi', 'diagnosis'],
  ['symtoms', 'symptoms'],
  ['medcation', 'medication'],
  ['prescritpion', 'prescription'],
]);

function normalizeSpelling(token: string): string {
  return COMMON_MISSPELLINGS.get(token) ?? token;
}

// Full preprocessing pipeline
function preprocessClinicalText(text: string): string[] {
  const tokens = medicalTokenize(text);
  return tokens.map(normalizeSpelling);
}

console.log(preprocessClinicalText('Patinet reports symtoms after PCN dose'));
// ['patient', 'reports', 'symptoms', 'after', 'penicillin', 'dose']`}
</CodeBlock>
