Each pipeline step receives text (or tokens), transforms it, and passes the result forward. One rule has no middleware equivalent: the exact same pipeline must run at training time and at inference time. If the two drift apart, the model quietly degrades. Training/serving skew is the canonical preprocessing bug.

## The Pipe Pattern

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="functional-pipe.ts">
{`// Functional pipe, identical to lodash/fp pipe or RxJS pipe
type Transform = (input: string) => string;

function pipe(...fns: Transform[]): Transform {
  return (input: string) => fns.reduce((acc, fn) => fn(acc), input);
}

// Define individual steps
const stripHtml: Transform = (text) =>
  text.replace(/<[^>]*>/g, ' ').replace(/&#?\\w+;/g, ' ');

const normalizeUnicode: Transform = (text) =>
  text.normalize('NFKD')
    .replace(/[\\u0300-\\u036f]/g, '')
    .replace(/[\\u2018\\u2019]/g, "'")
    .replace(/[\\u201C\\u201D]/g, '"');

const lowercase: Transform = (text) => text.toLowerCase();

const collapseWhitespace: Transform = (text) =>
  text.replace(/\\s+/g, ' ').trim();

const removeSpecialChars: Transform = (text) =>
  text.replace(/[^a-z0-9\\s.\\-']/g, ' ');

// Compose the pipeline
const cleanText = pipe(
  stripHtml,
  normalizeUnicode,
  lowercase,
  collapseWhitespace,
  removeSpecialChars,
  collapseWhitespace  // Run again after special char removal
);

console.log(cleanText('<p>Pt\\u2019s <b>BP</b>: 120/80</p>'));
// "pt's bp 120 80"`}
</CodeBlock>

## The Pipeline Class

For more complex scenarios, a class-based pipeline offers logging, error handling, and step management.

<CodeBlock language="typescript" title="pipeline-class.ts">
{`type StepFn = (input: string) => string;

interface PipelineStep {
  name: string;
  fn: StepFn;
}

interface PipelineResult {
  output: string;
  log: { step: string; input: string; output: string; durationMs: number }[];
}

class TextPipeline {
  private steps: PipelineStep[] = [];

  add(name: string, fn: StepFn): this {
    this.steps.push({ name, fn });
    return this; // Enable chaining
  }

  run(input: string): PipelineResult {
    const log: PipelineResult['log'] = [];
    let current = input;

    for (const step of this.steps) {
      const start = performance.now();
      try {
        const output = step.fn(current);
        log.push({
          step: step.name,
          input: current.slice(0, 50),
          output: output.slice(0, 50),
          durationMs: performance.now() - start,
        });
        current = output;
      } catch (err) {
        log.push({
          step: step.name,
          input: current.slice(0, 50),
          output: 'ERROR: ' + (err as Error).message,
          durationMs: performance.now() - start,
        });
        throw new Error('Pipeline failed at step "' + step.name + '": ' + (err as Error).message);
      }
    }

    return { output: current, log };
  }
}

// Usage: reads like Express middleware
const pipeline = new TextPipeline()
  .add('stripHtml', (t) => t.replace(/<[^>]*>/g, ' '))
  .add('lowercase', (t) => t.toLowerCase())
  .add('collapseWS', (t) => t.replace(/\\s+/g, ' ').trim());

const result = pipeline.run('<p>Patient BP Elevated</p>');
console.log(result.output);  // 'patient bp elevated'
console.log(result.log);     // Step-by-step trace`}
</CodeBlock>

## End-to-End Clinical Pipeline

Chain cleaning, tokenization, abbreviation expansion, and stop-word removal into a full pipeline.

<CodeBlock language="typescript" title="clinical-pipeline.ts">
{`const MEDICAL_ABBREVS = new Map([
  ['pt', 'patient'], ['hx', 'history'], ['dx', 'diagnosis'],
  ['rx', 'prescription'], ['pcn', 'penicillin'], ['bp', 'blood-pressure'],
  ['hr', 'heart-rate'], ['prn', 'as-needed'],
]);

const STOP_WORDS = new Set([
  'the', 'a', 'an', 'is', 'was', 'were', 'of', 'in', 'on', 'for',
  'to', 'with', 'and', 'or', 'but', 'at', 'by', 'from',
]);

class ClinicalPipeline {
  private textPipeline: TextPipeline;

  constructor() {
    this.textPipeline = new TextPipeline()
      .add('stripHtml', (t) => t.replace(/<[^>]*>/g, ' ').replace(/&#?\\w+;/g, ' '))
      .add('unicode', (t) => t.normalize('NFKD').replace(/[\\u0300-\\u036f]/g, ''))
      .add('lowercase', (t) => t.toLowerCase())
      .add('specialChars', (t) => t.replace(/[^a-z0-9\\s.\\-]/g, ' '))
      .add('whitespace', (t) => t.replace(/\\s+/g, ' ').trim());
  }

  process(rawNote: string): string[] {
    // Step 1: Clean the text
    const { output: cleaned } = this.textPipeline.run(rawNote);

    // Step 2: Tokenize
    const tokens = cleaned.match(/[a-z]+(?:-[a-z]+)*|[0-9]+(?:\\.[0-9]+)?/g) || [];

    // Step 3: Expand abbreviations
    const expanded = tokens.map(t => MEDICAL_ABBREVS.get(t) ?? t);

    // Step 4: Remove stop words
    const filtered = expanded.filter(t => !STOP_WORDS.has(t));

    return filtered;
  }
}

const pipeline = new ClinicalPipeline();
const note = '<p>Pt hx of <b>PCN</b> allergy. BP was 120/80 on the monitor.</p>';
console.log(pipeline.process(note));
// ['patient', 'history', 'penicillin', 'allergy', 'blood-pressure', '120', '80', 'monitor']`}
</CodeBlock>

## Step Ordering Matters

The order of pipeline steps affects the output. Consider what happens when you lowercase before vs. after abbreviation handling.

<CodeBlock language="typescript" title="ordering-matters.ts">
{`// WRONG: Expand abbreviations before lowercasing
// "PCN" won't match the map key "pcn"
const wrongOrder = (text: string) => {
  const tokens = text.match(/[a-zA-Z]+/g) || [];
  const expanded = tokens.map(t => MEDICAL_ABBREVS.get(t) ?? t); // "PCN" misses
  return expanded.map(t => t.toLowerCase());
};

// RIGHT: Lowercase first, then expand
const rightOrder = (text: string) => {
  const tokens = text.toLowerCase().match(/[a-z]+/g) || [];
  return tokens.map(t => MEDICAL_ABBREVS.get(t) ?? t); // "pcn" matches
};

console.log(wrongOrder('PCN allergy'));  // ['pcn', 'allergy']: not expanded!
console.log(rightOrder('PCN allergy'));  // ['penicillin', 'allergy']: correct`}
</CodeBlock>
