## Stripping HTML Tags

Legacy clinical systems often store notes with embedded HTML from WYSIWYG editors.

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="strip-html.ts">
{`// Clinical note from legacy system
const dirtyNote = '<p>Pt presents with <b>chest pain</b>.</p><br/>' +
  '<span style="color:red">ALLERGY: PCN</span>&nbsp;&mdash; confirmed 2024.';

function stripHtml(text: string): string {
  return text
    .replace(/<[^>]*>/g, ' ')        // Remove HTML tags
    .replace(/&nbsp;/g, ' ')          // Replace common entities
    .replace(/&mdash;/g, '—')
    .replace(/&amp;/g, '&')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&#?\\w+;/g, ' ');        // Catch remaining entities
}

console.log(stripHtml(dirtyNote));
// ' Pt presents with  chest pain .  ALLERGY: PCN  — confirmed 2024.'`}
</CodeBlock>

## Unicode Normalization

Clinical notes may contain accented characters, smart quotes, or invisible control characters from copy-paste.

<CodeBlock language="typescript" title="unicode-normalization.ts">
{`function normalizeUnicode(text: string): string {
  return text
    .normalize('NFKD')                          // Decompose Unicode (é → e + ́)
    .replace(/[\\u0300-\\u036f]/g, '')             // Strip combining diacritical marks
    .replace(/[\\u2018\\u2019]/g, "'")             // Smart single quotes → straight
    .replace(/[\\u201C\\u201D]/g, '"')             // Smart double quotes → straight
    .replace(/[\\u2013\\u2014]/g, '-')             // En/em dashes → hyphen
    .replace(/[\\u00A0]/g, ' ')                    // Non-breaking space → space
    .replace(/[\\u200B-\\u200D\\uFEFF]/g, '');     // Zero-width characters → remove
}

console.log(normalizeUnicode('Patient\u2019s dx: caf\u00e9 au lait spots'));
// "Patient's dx: cafe au lait spots"`}
</CodeBlock>

## Lowercasing and Whitespace

Inconsistent casing and extra whitespace create false distinctions between identical tokens.

<CodeBlock language="typescript" title="case-whitespace.ts">
{`function normalizeWhitespace(text: string): string {
  return text
    .toLowerCase()
    .replace(/\\s+/g, ' ')    // Collapse multiple spaces, tabs, newlines
    .trim();
}

// "CHEST PAIN" and "chest pain" and "Chest  Pain" all become "chest pain"
console.log(normalizeWhitespace('  CHEST   PAIN\\n'));  // 'chest pain'
console.log(normalizeWhitespace('Chest  Pain'));        // 'chest pain'`}
</CodeBlock>

## Basic Stemming

Stemming reduces words to their root form: "medications," "medicated," and "medicating" all become "medic." This is a crude but effective approach for reducing vocabulary size.

<CodeBlock language="typescript" title="simple-stemmer.ts">
{`// Simple suffix-stripping stemmer for clinical text
const SUFFIX_RULES: [RegExp, string][] = [
  [/ational$/, 'ate'],
  [/ization$/, 'ize'],
  [/iveness$/, 'ive'],
  [/fulness$/, 'ful'],
  [/ousli$/, 'ous'],
  [/ation$/, 'ate'],
  [/ities$/, 'ity'],
  [/ings$/, ''],
  [/ment$/, ''],
  [/ness$/, ''],
  [/ing$/, ''],
  [/ies$/, 'y'],
  [/ed$/, ''],
  [/s$/, ''],
];

function stem(word: string): string {
  if (word.length < 4) return word; // Skip short words
  for (const [pattern, replacement] of SUFFIX_RULES) {
    if (pattern.test(word)) {
      const result = word.replace(pattern, replacement);
      if (result.length >= 3) return result; // Ensure minimum length
    }
  }
  return word;
}

console.log(stem('medications'));  // 'medicate'  (via -ation → -ate, then -s)
console.log(stem('presenting'));   // 'present'
console.log(stem('allergies'));    // 'allergy'`}
</CodeBlock>

## The Full Cleaning Function

Chain all the steps into a single reusable cleaner.

<CodeBlock language="typescript" title="clean-clinical-text.ts">
{`function cleanClinicalText(text: string): string {
  let cleaned = text;

  // Step 1: Strip HTML
  cleaned = cleaned.replace(/<[^>]*>/g, ' ');
  cleaned = cleaned.replace(/&#?\\w+;/g, ' ');

  // Step 2: Unicode normalization
  cleaned = cleaned.normalize('NFKD');
  cleaned = cleaned.replace(/[\\u0300-\\u036f]/g, '');
  cleaned = cleaned.replace(/[\\u2018\\u2019]/g, "'");
  cleaned = cleaned.replace(/[\\u201C\\u201D]/g, '"');
  cleaned = cleaned.replace(/[\\u2013\\u2014]/g, '-');
  cleaned = cleaned.replace(/[\\u200B-\\u200D\\uFEFF]/g, '');

  // Step 3: Remove non-alphanumeric (keep hyphens, periods for decimals)
  cleaned = cleaned.replace(/[^a-zA-Z0-9\\s.\\-']/g, ' ');

  // Step 4: Lowercase and collapse whitespace
  cleaned = cleaned.toLowerCase().replace(/\\s+/g, ' ').trim();

  return cleaned;
}

const legacy = '<p>Pt\\u2019s <b>BP</b>: 120/80.&nbsp;Dx: HTN.</p>';
console.log(cleanClinicalText(legacy));
// "pt's bp 120 80. dx htn."`}
</CodeBlock>
