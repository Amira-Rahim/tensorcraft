Every frontend application I have built handles sensitive data: passwords, payment details, personal information. We follow GDPR rules: consent banners, data minimization, right to deletion, audit trails. HIPAA is the same regulatory mindset applied to medical data. The consequences of failure are higher.
</Dialogue>

<Dialogue speaker="Dr. Karimi">
The advantage of browser-based inference is architectural. The model weights travel to the browser; the clinical notes stay on the hospital network. Patient data never touches a cloud server, so there is nothing to exfiltrate.
</Dialogue>

One clinical note in a crash report is a reportable HIPAA breach, a federal filing: who was exposed, when, how, and what you are doing about it. The toolkit is the one GDPR already taught the web industry: ask before collecting, minimize what you store, keep audit trails. HIPAA applies it to Protected Health Information (PHI). And browser-based ML adds a structural advantage: if inference runs on-device, patient data never leaves the hospital network.

<CodeBlock language="typescript" title="phi-detector.ts">
{`// PHI Detection: like input sanitization for XSS, but for medical data
// Catch sensitive data before it reaches logs, caches, or analytics

interface PHIDetection {
  type: 'NAME' | 'MRN' | 'DOB' | 'SSN' | 'PHONE' | 'ADDRESS';
  start: number;
  end: number;
  text: string;
}

const PHI_PATTERNS: { type: PHIDetection['type']; pattern: RegExp }[] = [
  { type: 'MRN', pattern: /\b(MRN|Medical Record)\s*[#:]?\s*\d{6,10}\b/gi },
  { type: 'SSN', pattern: /\b\d{3}-\d{2}-\d{4}\b/g },
  { type: 'DOB', pattern: /\b(DOB|Date of Birth)\s*[:]?\s*\d{1,2}[\/\-]\d{1,2}[\/\-]\d{2,4}\b/gi },
  { type: 'PHONE', pattern: /\b\(?\d{3}\)?[-.\s]?\d{3}[-.\s]?\d{4}\b/g },
];

function detectPHI(text: string): PHIDetection[] {
  const detections: PHIDetection[] = [];

  for (const { type, pattern } of PHI_PATTERNS) {
    let match;
    const regex = new RegExp(pattern.source, pattern.flags);
    while ((match = regex.exec(text)) !== null) {
      detections.push({
        type,
        start: match.index,
        end: match.index + match[0].length,
        text: match[0],
      });
    }
  }

  return detections;
}

function redactPHI(text: string): string {
  const detections = detectPHI(text);
  let redacted = text;

  // Process in reverse order to maintain indices
  for (const det of detections.sort((a, b) => b.start - a.start)) {
    redacted = redacted.substring(0, det.start)
      + \`[REDACTED_\${det.type}]\`
      + redacted.substring(det.end);
  }

  return redacted;
}`}
</CodeBlock>

## Audit Logging Without Storing Content

<CodeBlock language="typescript" title="hipaa-audit.ts">
{`// HIPAA requires audit trails for all PHI access
// Log the event metadata, never the content, like GDPR consent logs

interface AuditEntry {
  timestamp: string;
  userId: string;
  action: 'inference' | 'view_prediction' | 'export' | 'model_load';
  resourceType: 'clinical_note' | 'prediction' | 'model';
  resourceId: string;
  metadata: {
    inputLength?: number;
    modelVersion?: string;
    predictionConfidence?: number;
    accessJustification?: string;
  };
  // Never log the actual PHI content
}

class HIPAAAuditLogger {
  private entries: AuditEntry[] = [];

  log(entry: AuditEntry): void {
    // Validate: no PHI leaked into metadata
    const metaString = JSON.stringify(entry.metadata);
    const phiDetections = detectPHI(metaString);
    if (phiDetections.length > 0) {
      throw new Error('PHI detected in audit metadata: redact before logging');
    }
    this.entries.push(entry);
  }

  report(startDate: string, endDate: string): {
    totalAccesses: number;
    byAction: Record<string, number>;
    uniqueUsers: number;
  } {
    const filtered = this.entries.filter(
      e => e.timestamp >= startDate && e.timestamp <= endDate
    );

    const byAction: Record<string, number> = {};
    const users = new Set<string>();

    for (const entry of filtered) {
      byAction[entry.action] = (byAction[entry.action] || 0) + 1;
      users.add(entry.userId);
    }

    return { totalAccesses: filtered.length, byAction, uniqueUsers: users.size };
  }
}

// On-device inference: PHI never leaves the browser
async function privacyAwareInference(
  clinicalNote: string,
  model: tf.LayersModel,
  tokenize: (text: string) => number[],
  logger: HIPAAAuditLogger,
  userId: string
): Promise<{ prediction: number[]; noteId: string }> {
  const noteId = crypto.randomUUID();

  // Log access event; never log content
  logger.log({
    timestamp: new Date().toISOString(),
    userId,
    action: 'inference',
    resourceType: 'clinical_note',
    resourceId: noteId,
    metadata: {
      inputLength: clinicalNote.length,
      modelVersion: '1.0.0',
    },
  });

  // All computation happens locally: no network call
  const tokens = tokenize(clinicalNote);
  const input = tf.tensor2d([tokens]);
  const output = model.predict(input) as tf.Tensor;
  const prediction = await output.array() as number[];

  input.dispose();
  output.dispose();

  return { prediction, noteId };
}`}
</CodeBlock>
