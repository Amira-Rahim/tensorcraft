{`import * as tf from '@tensorflow/tfjs';

// Typed request/response — same as any API contract
interface NLPRequest {
  text: string;
  options?: {
    confidenceThreshold?: number;
    maxEntities?: number;
    includeAttention?: boolean;
  };
}

interface NLPResponse {
  requestId: string;
  predictions: {
    classification: { label: string; confidence: number };
    entities: { text: string; label: string; confidence: number }[];
  };
  meta: {
    modelVersion: string;
    processingTimeMs: number;
    tokenCount: number;
  };
}

interface APIError {
  requestId: string;
  error: string;
  code: 'VALIDATION_ERROR' | 'TIMEOUT' | 'MODEL_ERROR' | 'SERVICE_UNAVAILABLE';
  statusCode: number;
}

// Request validation — same as Zod/Joi validation middleware
function validateRequest(body: unknown): { valid: true; data: NLPRequest } | { valid: false; error: string } {
  if (!body || typeof body !== 'object') {
    return { valid: false, error: 'Request body must be a JSON object' };
  }
  const req = body as Record<string, unknown>;
  if (typeof req.text !== 'string' || req.text.trim().length === 0) {
    return { valid: false, error: 'Field "text" is required and must be a non-empty string' };
  }
  if (req.text.length > 10000) {
    return { valid: false, error: 'Field "text" must be <= 10,000 characters' };
  }
  return { valid: true, data: req as unknown as NLPRequest };
}

// Latency budget enforcement — like a fetch() timeout wrapper
function withTimeout<T>(promise: Promise<T>, budgetMs: number): Promise<T> {
  return Promise.race([
    promise,
    new Promise<never>((_, reject) =>
      setTimeout(() => reject(new Error('TIMEOUT')), budgetMs)
    ),
  ]);
}

// Production NLP API server
class NLPAPIServer {
  private model: tf.LayersModel | null = null;
  private tokenizer: (text: string) => number[];
  private labels: string[];
  private latencyBudgetMs: number;
  private requestCount = 0;

  constructor(
    tokenizer: (text: string) => number[],
    labels: string[],
    latencyBudgetMs: number = 200
  ) {
    this.tokenizer = tokenizer;
    this.labels = labels;
    this.latencyBudgetMs = latencyBudgetMs;
  }

  async loadModel(url: string): Promise<void> {
    this.model = await tf.loadLayersModel(url);
    // Warm-up inference — like preloading critical resources
    const warmup = tf.zeros([1, 128]);
    this.model.predict(warmup);
    warmup.dispose();
  }

  // POST /api/predict — the core handler
  async predict(body: unknown): Promise<NLPResponse | APIError> {
    const requestId = \`req-\${++this.requestCount}\`;
    const startTime = performance.now();

    // 1. Validate — same as Express validation middleware
    const validation = validateRequest(body);
    if (!validation.valid) {
      return { requestId, error: validation.error, code: 'VALIDATION_ERROR', statusCode: 400 };
    }

    if (!this.model) {
      return { requestId, error: 'Model not loaded', code: 'SERVICE_UNAVAILABLE', statusCode: 503 };
    }

    try {
      // 2. Infer — with latency budget enforcement
      const result = await withTimeout(
        this.runInference(validation.data),
        this.latencyBudgetMs
      );

      return {
        requestId,
        predictions: result.predictions,
        meta: {
          modelVersion: '1.0.0',
          processingTimeMs: performance.now() - startTime,
          tokenCount: result.tokenCount,
        },
      };
    } catch (err) {
      if ((err as Error).message === 'TIMEOUT') {
        return { requestId, error: 'Inference exceeded latency budget', code: 'TIMEOUT', statusCode: 504 };
      }
      return { requestId, error: 'Internal model error', code: 'MODEL_ERROR', statusCode: 500 };
    }
  }

  private async runInference(req: NLPRequest) {
    const tokens = this.tokenizer(req.text);
    const input = tf.tensor2d([tokens]);
    const output = this.model!.predict(input) as tf.Tensor;
    const probs = (await output.array() as number[][])[0];

    const threshold = req.options?.confidenceThreshold ?? 0.5;
    const topIdx = probs.indexOf(Math.max(...probs));

    input.dispose();
    output.dispose();

    return {
      predictions: {
        classification: {
          label: this.labels[topIdx],
          confidence: probs[topIdx],
        },
        entities: [],
      },
      tokenCount: tokens.length,
    };
  }

  // GET /api/health — readiness probe
  health(): { status: string; modelLoaded: boolean; requestCount: number; tensorCount: number } {
    return {
      status: this.model ? 'healthy' : 'degraded',
      modelLoaded: this.model !== null,
      requestCount: this.requestCount,
      tensorCount: tf.memory().numTensors,
    };
  }
}`}
</CodeBlock>

## Request Batching for Throughput

<CodeBlock language="typescript" title="batch-api.ts">
{`// Batch requests for GPU efficiency — like batching DB queries to avoid N+1
class BatchPredictor {
  private queue: { tokens: number[]; resolve: (r: number[]) => void }[] = [];
  private timer: ReturnType<typeof setTimeout> | null = null;

  constructor(
    private model: tf.LayersModel,
    private batchSize = 8,
    private batchTimeoutMs = 10
  ) {}

  async predict(tokens: number[]): Promise<number[]> {
    return new Promise((resolve) => {
      this.queue.push({ tokens, resolve });

      if (this.queue.length >= this.batchSize) {
        this.flush();
      } else if (!this.timer) {
        this.timer = setTimeout(() => this.flush(), this.batchTimeoutMs);
      }
    });
  }

  private async flush(): Promise<void> {
    if (this.timer) { clearTimeout(this.timer); this.timer = null; }
    if (this.queue.length === 0) return;

    const batch = this.queue.splice(0, this.batchSize);
    const input = tf.tensor2d(batch.map(b => b.tokens));
    const output = this.model.predict(input) as tf.Tensor;
    const results = await output.array() as number[][];

    input.dispose();
    output.dispose();

    batch.forEach((item, i) => item.resolve(results[i]));
  }
}`}
</CodeBlock>
