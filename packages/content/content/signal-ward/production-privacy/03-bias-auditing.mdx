<Dialogue speaker="Dr. Karimi">
Your model performs 12% worse on records written in non-standard English. Our night shift nurses — many of whom speak English as a second language — write differently. Shorter sentences. Different abbreviations. No formal headers. Your model was trained on attending physician notes. It works well for people who write like attending physicians. That is a bias, and in this hospital, it is unacceptable.
</Dialogue>

<Dialogue speaker="Khalil">
It is the same problem as accessibility. If you only test your app with a mouse on a large monitor, you miss keyboard users, screen reader users, mobile users. Run Lighthouse and suddenly you see a score of 40 where you expected 95. If you only evaluate your model on one style of clinical writing, you miss everyone who writes differently. We need a Lighthouse for fairness.
</Dialogue>

Accessibility auditing taught the frontend world a critical lesson: you cannot build for everyone by testing with one user profile. You run Lighthouse, test with VoiceOver, check color contrast ratios, verify keyboard navigation. Bias auditing applies the same discipline to ML: test across demographic groups, measure performance per group, and fix the gaps. A model that works well on average but fails for specific populations is not a good model — just as a website that works for sighted mouse users but fails for screen reader users is not an accessible website.

{`import * as tf from '@tensorflow/tfjs';

interface AuditSlice {
  name: string;
  description: string;
  samples: { text: string; trueLabel: number }[];
}

interface SliceMetrics {
  sliceName: string;
  accuracy: number;
  falsePositiveRate: number;
  falseNegativeRate: number;
  sampleCount: number;
}

interface AuditReport {
  overall: SliceMetrics;
  slices: SliceMetrics[];
  disparities: {
    worstSlice: string;
    bestSlice: string;
    accuracyGap: number;
    meetsEquityThreshold: boolean;
  };
}

// Compute metrics for a single demographic slice
function computeSliceMetrics(
  predictions: number[],
  trueLabels: number[],
  sliceName: string
): SliceMetrics {
  let correct = 0;
  let falsePositives = 0;
  let falseNegatives = 0;
  let totalPositives = 0;
  let totalNegatives = 0;

  for (let i = 0; i < predictions.length; i++) {
    if (predictions[i] === trueLabels[i]) correct++;
    if (trueLabels[i] === 1) totalPositives++;
    if (trueLabels[i] === 0) totalNegatives++;
    if (predictions[i] === 1 && trueLabels[i] === 0) falsePositives++;
    if (predictions[i] === 0 && trueLabels[i] === 1) falseNegatives++;
  }

  return {
    sliceName,
    accuracy: correct / predictions.length,
    falsePositiveRate: totalNegatives > 0 ? falsePositives / totalNegatives : 0,
    falseNegativeRate: totalPositives > 0 ? falseNegatives / totalPositives : 0,
    sampleCount: predictions.length,
  };
}

// Run a complete fairness audit — like running Lighthouse across device profiles
async function auditFairness(
  model: tf.LayersModel,
  slices: AuditSlice[],
  tokenize: (text: string) => number[],
  equityThreshold: number = 0.05
): Promise<AuditReport> {
  const allPredictions: number[] = [];
  const allLabels: number[] = [];
  const sliceMetrics: SliceMetrics[] = [];

  for (const slice of slices) {
    const predictions: number[] = [];
    const labels = slice.samples.map(s => s.trueLabel);

    for (const sample of slice.samples) {
      const tokens = tokenize(sample.text);
      const input = tf.tensor2d([tokens]);
      const output = model.predict(input) as tf.Tensor;
      const probs = (await output.array() as number[][])[0];
      predictions.push(probs.indexOf(Math.max(...probs)));
      input.dispose();
      output.dispose();
    }

    sliceMetrics.push(computeSliceMetrics(predictions, labels, slice.name));
    allPredictions.push(...predictions);
    allLabels.push(...labels);
  }

  const overall = computeSliceMetrics(allPredictions, allLabels, 'overall');
  const worstSlice = sliceMetrics.reduce((a, b) => a.accuracy < b.accuracy ? a : b);
  const bestSlice = sliceMetrics.reduce((a, b) => a.accuracy > b.accuracy ? a : b);
  const accuracyGap = bestSlice.accuracy - worstSlice.accuracy;

  return {
    overall,
    slices: sliceMetrics,
    disparities: {
      worstSlice: worstSlice.sliceName,
      bestSlice: bestSlice.sliceName,
      accuracyGap,
      meetsEquityThreshold: accuracyGap <= equityThreshold,
    },
  };
}`}
</CodeBlock>

## The Fairness Card

<CodeBlock language="typescript" title="fairness-card.ts">
{`// A fairness card is like a Lighthouse report — per-category scores with pass/fail
function generateFairnessCard(report: AuditReport): string {
  const lines: string[] = [];
  lines.push('=== Diagnostic One — Fairness Audit ===');
  lines.push(\`Overall Accuracy: \${(report.overall.accuracy * 100).toFixed(1)}%\`);
  lines.push('');
  lines.push('Per-Group Performance:');

  for (const slice of report.slices) {
    const bar = '#'.repeat(Math.round(slice.accuracy * 20));
    const gap = slice.accuracy - report.overall.accuracy;
    const gapStr = gap >= 0 ? \`+\${(gap * 100).toFixed(1)}%\` : \`\${(gap * 100).toFixed(1)}%\`;
    const fprStr = \`FPR: \${(slice.falsePositiveRate * 100).toFixed(1)}%\`;
    const fnrStr = \`FNR: \${(slice.falseNegativeRate * 100).toFixed(1)}%\`;
    lines.push(\`  \${slice.sliceName.padEnd(25)} \${bar.padEnd(20)} \${(slice.accuracy * 100).toFixed(1)}% (\${gapStr}) | \${fprStr} | \${fnrStr}\`);
  }

  lines.push('');
  lines.push(\`Accuracy Gap: \${(report.disparities.accuracyGap * 100).toFixed(1)}%\`);
  lines.push(\`Equity Threshold Met: \${report.disparities.meetsEquityThreshold ? 'PASS' : 'FAIL'}\`);

  if (!report.disparities.meetsEquityThreshold) {
    lines.push(\`  Worst performing group: \${report.disparities.worstSlice}\`);
    lines.push('  Action required: augment training data for underperforming groups');
  }

  return lines.join('\\n');
}

// Define the audit slices for Diagnostic One
const diagnosticOneSlices: AuditSlice[] = [
  {

  },

    samples: [],
  },
  {
    name: 'spanish-language-notes',
    description: 'Spanish-language clinical notes from bilingual staff',
    samples: [],
  },
];`}
