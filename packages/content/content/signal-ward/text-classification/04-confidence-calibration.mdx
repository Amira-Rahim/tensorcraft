## The Danger of Confident Mistakes

<BridgeCallout bridge={frontmatter.concept.bridge} />

TypeScript strict mode forces you to handle every uncertain case explicitly. Without it, the compiler silently allows `any` types — and bugs hide in the ambiguity. Model calibration is the same idea: without it, a model can be 95% "confident" about a prediction and still be wrong half the time.

A well-calibrated model's confidence matches its actual accuracy. When it says 80%, it should be right about 80% of the time. Most neural networks are overconfident — they need calibration.

<CodeBlock language="typescript" title="calibration.ts">
{`import * as tf from '@tensorflow/tfjs';

// Temperature scaling: the simplest calibration method
function temperatureScale(
  logits: tf.Tensor,
  temperature: number
): tf.Tensor {
  // Higher temperature -> softer probabilities -> less overconfident
  // Lower temperature -> sharper probabilities -> more confident
  return tf.softmax(tf.div(logits, temperature));
}

// Entropy measures how uncertain the prediction distribution is
function predictionEntropy(probs: tf.Tensor): tf.Tensor {
  // H = -sum(p * log(p))
  // High entropy = uncertain; low entropy = confident
  const logProbs = tf.log(tf.add(probs, 1e-10)); // avoid log(0)
  return tf.neg(tf.sum(tf.mul(probs, logProbs), -1));
}

// Maximum entropy for reference (uniform distribution)
function maxEntropy(numClasses: number): number {
  return Math.log(numClasses);
}

// Example: uncalibrated vs calibrated predictions
const logits = tf.tensor2d([[3.5, 1.2, 0.1, -0.5, 0.3, -1.0]]);

// Uncalibrated: very confident
const uncalibrated = tf.softmax(logits);
console.log('Uncalibrated:', await uncalibrated.data());

// Calibrated with temperature = 2.0: softer distribution
const calibrated = temperatureScale(logits, 2.0);
console.log('Calibrated (T=2):', await calibrated.data());

// Check entropy
const uncalEntropy = predictionEntropy(uncalibrated);
const calEntropy = predictionEntropy(calibrated);
console.log('Uncalibrated entropy:', uncalEntropy.dataSync()[0].toFixed(3));
console.log('Calibrated entropy:', calEntropy.dataSync()[0].toFixed(3));
console.log('Max entropy:', maxEntropy(6).toFixed(3));`}
</CodeBlock>

## Reliability Diagrams

A reliability diagram bins predictions by confidence and plots actual accuracy against predicted confidence. A perfectly calibrated model falls on the diagonal — 70% confidence means 70% accuracy. Most models curve above the diagonal (overconfident).

<CodeBlock language="typescript" title="reliability-diagram.ts">
{`interface CalibrationBin {
  binCenter: number;
  avgConfidence: number;
  avgAccuracy: number;
  count: number;
}

function computeCalibrationBins(
  predictions: number[],
  actuals: boolean[],
  numBins: number = 10
): CalibrationBin[] {
  const bins: CalibrationBin[] = [];

  for (let i = 0; i < numBins; i++) {
    const lower = i / numBins;
    const upper = (i + 1) / numBins;
    const binCenter = (lower + upper) / 2;

    const inBin = predictions
      .map((p, idx) => ({ confidence: p, correct: actuals[idx] }))
      .filter(x => x.confidence >= lower && x.confidence < upper);

    if (inBin.length === 0) {
      bins.push({ binCenter, avgConfidence: 0, avgAccuracy: 0, count: 0 });
      continue;
    }

    const avgConfidence = inBin.reduce((s, x) => s + x.confidence, 0) / inBin.length;
    const avgAccuracy = inBin.filter(x => x.correct).length / inBin.length;

    bins.push({ binCenter, avgConfidence, avgAccuracy, count: inBin.length });
  }

  return bins;
}

// Expected Calibration Error (ECE)
function expectedCalibrationError(
  bins: CalibrationBin[],
  totalSamples: number
): number {
  return bins.reduce((ece, bin) => {
    return ece + (bin.count / totalSamples) *
      Math.abs(bin.avgAccuracy - bin.avgConfidence);
  }, 0);
}

// Example: check calibration
const predictions = [0.91, 0.88, 0.92, 0.87, 0.93, 0.90, 0.89, 0.91, 0.88, 0.92];
const correct = [true, true, false, true, true, false, true, false, true, true];

const bins = computeCalibrationBins(predictions, correct);
const ece = expectedCalibrationError(bins, predictions.length);
console.log('ECE (lower is better):', ece.toFixed(4));
// High ECE means the model is poorly calibrated`}
</CodeBlock>

## Uncertainty-Aware Routing

<CodeBlock language="typescript" title="uncertainty-routing.ts">
{`interface ClassificationResult {
  label: string;
  confidence: number;
  entropy: number;
  needsReview: boolean;
}

function classifyWithUncertainty(
  logits: tf.Tensor,
  labels: string[],
  temperature: number = 1.5,
  reviewThreshold: number = 0.7
): ClassificationResult {
  // Calibrate with temperature scaling
  const calibratedProbs = temperatureScale(logits, temperature);
  const probArray = calibratedProbs.dataSync();

  // Find top prediction
  const topIdx = calibratedProbs.argMax(-1).dataSync()[0];
  const confidence = probArray[topIdx];

  // Compute entropy
  const entropy = predictionEntropy(calibratedProbs).dataSync()[0];
  const maxEnt = maxEntropy(labels.length);
  const normalizedEntropy = entropy / maxEnt; // 0 = certain, 1 = maximally uncertain

  return {
    label: labels[topIdx],
    confidence,
    entropy: normalizedEntropy,
    needsReview: confidence < reviewThreshold || normalizedEntropy > 0.5
  };
}
