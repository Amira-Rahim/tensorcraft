## Dependency Resolution for Medicine

<BridgeCallout bridge={frontmatter.concept.bridge} />

When you run `npm install`, the package manager resolves dependencies. If `package-a` requires `react@17` and `package-b` requires `react@18`, you get a conflict warning. The system catches incompatibilities that a developer manually reviewing two `package.json` files might miss.

Drug interaction detection works identically. Each clinical note is a `package.json`. Each prescribed drug is a dependency. The NER pipeline extracts all "dependencies" across all "packages" and checks for conflicts against a known interaction database.

Dependency conflicts span multiple `package.json` files from different teams; drug interactions span multiple clinical notes from different prescribers. No single doctor sees the full picture. The system does.

<CodeBlock language="typescript" title="drug-interaction-detector.ts">
{`import * as tf from '@tensorflow/tfjs';

// Known drug interactions, like a database of incompatible package versions
const INTERACTION_DB: Record<string, { severity: string; effect: string }> = {
  'aspirin+warfarin': {
    severity: 'HIGH',
    effect: 'Increased bleeding risk due to combined anticoagulant/antiplatelet effect'
  },
  'ibuprofen+methotrexate': {
    severity: 'HIGH',
    effect: 'NSAIDs reduce methotrexate clearance, increasing toxicity risk'
  },
  'lisinopril+potassium': {
    severity: 'MODERATE',
    effect: 'Risk of hyperkalemia'
  },
  'aspirin+ibuprofen': {
    severity: 'MODERATE',
    effect: 'Reduced aspirin cardioprotective efficacy'
  }
};

interface DrugEntity {
  name: string;
  source: string;      // which clinical note, like which package.json
  prescriber: string;   // who added the dependency
  date: string;
}

interface InteractionAlert {
  drug1: DrugEntity;
  drug2: DrugEntity;
  severity: string;
  effect: string;
  crossDocument: boolean;  // true = different prescribers, like cross-package conflicts
}

// Check all drug pairs, like npm checking all dependency pairs for conflicts
function checkInteractions(drugs: DrugEntity[]): InteractionAlert[] {
  const alerts: InteractionAlert[] = [];

  for (let i = 0; i < drugs.length; i++) {
    for (let j = i + 1; j < drugs.length; j++) {
      // Sort alphabetically to create a canonical key, like normalizing package names
      const pair = [drugs[i].name.toLowerCase(), drugs[j].name.toLowerCase()]
        .sort()
        .join('+');

      if (pair in INTERACTION_DB) {
        const interaction = INTERACTION_DB[pair];
        alerts.push({
          drug1: drugs[i],
          drug2: drugs[j],
          severity: interaction.severity,
          effect: interaction.effect,
          crossDocument: drugs[i].source !== drugs[j].source
        });
      }
    }
  }

  // Sort by severity: HIGH first, like npm showing errors before warnings
  const severityOrder: Record<string, number> = { HIGH: 0, MODERATE: 1, LOW: 2 };
  return alerts.sort((a, b) => severityOrder[a.severity] - severityOrder[b.severity]);
}

// Full pipeline: NER → drug extraction → interaction check
// Like: npm install → resolve deps → check conflicts → report
function drugSafetyPipeline(
  clinicalNotes: Array<{ text: string; source: string; prescriber: string; date: string }>,
  nerExtract: (text: string) => string[]  // NER function that returns drug names
): InteractionAlert[] {
  const allDrugs: DrugEntity[] = [];

  for (const note of clinicalNotes) {
    const drugNames = nerExtract(note.text);
    for (const name of drugNames) {
      allDrugs.push({
        name,
        source: note.source,
        prescriber: note.prescriber,
        date: note.date
      });
    }
  }

  return checkInteractions(allDrugs);
}

// Example: Mrs. Begum's case
const drugs: DrugEntity[] = [
  { name: 'warfarin', source: 'Cardiology Note', prescriber: 'Dr. Patel', date: '2025-01-15' },
  { name: 'aspirin', source: 'Rheumatology Note', prescriber: 'Dr. Qureshi', date: '2025-02-03' }
];

const alerts = checkInteractions(drugs);
console.log(\`Found \${alerts.length} interaction(s):\`);
alerts.forEach(a => {
  console.log(\`  ALERT [\${a.severity}]: \${a.drug1.name} + \${a.drug2.name}\`);
  console.log(\`    Effect: \${a.effect}\`);
  console.log(\`    Cross-document: \${a.crossDocument}\`);
  console.log(\`    Sources: \${a.drug1.source} (\${a.drug1.prescriber}) + \${a.drug2.source} (\${a.drug2.prescriber})\`);
});`}
