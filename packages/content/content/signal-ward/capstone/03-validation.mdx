## Precision, Recall, and F1

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="validation-metrics.ts">
{`// Clinical validation metrics: the same math behind test coverage reports
// Precision: of everything the system flagged, how many were correct?
// Recall: of everything that should have been flagged, how many did the system find?

interface Annotation {
  text: string;
  label: string;
  span: [number, number];
}

interface Prediction {
  text: string;
  label: string;
  span: [number, number];
  confidence: number;
}

interface ValidationMetrics {
  precision: number;
  recall: number;
  f1: number;
  truePositives: number;
  falsePositives: number;
  falseNegatives: number;
}

function computeMetrics(
  predictions: Prediction[],
  annotations: Annotation[]
): ValidationMetrics {
  let truePositives = 0;
  let falsePositives = 0;
  let falseNegatives = 0;

  const matched = new Set<number>();

  // For each prediction, check if it matches an annotation
  for (const pred of predictions) {
    const matchIdx = annotations.findIndex((ann, idx) =>
      !matched.has(idx) &&
      ann.label === pred.label &&
      spansOverlap(ann.span, pred.span)
    );

    if (matchIdx >= 0) {
      truePositives++;
      matched.add(matchIdx);
    } else {
      falsePositives++; // System flagged something that was not in annotations
    }
  }

  // Annotations not matched by any prediction
  falseNegatives = annotations.length - matched.size;

  const precision = truePositives / (truePositives + falsePositives) || 0;
  const recall = truePositives / (truePositives + falseNegatives) || 0;
  const f1 = 2 * (precision * recall) / (precision + recall) || 0;

  return { precision, recall, f1, truePositives, falsePositives, falseNegatives };
}

function spansOverlap(a: [number, number], b: [number, number]): boolean {
  return a[0] < b[1] && b[0] < a[1];
}

// Example: validate NER against expert annotations
const expertAnnotations: Annotation[] = [
  { text: 'methotrexate', label: 'DRUG', span: [45, 57] },
  { text: 'trimethoprim', label: 'DRUG', span: [72, 84] },
  { text: 'rheumatoid arthritis', label: 'CONDITION', span: [12, 32] },
];

const systemPredictions: Prediction[] = [
  { text: 'methotrexate', label: 'DRUG', span: [45, 57], confidence: 0.97 },
  { text: 'trimethoprim', label: 'DRUG', span: [72, 84], confidence: 0.94 },
  { text: 'rheumatoid arthritis', label: 'CONDITION', span: [12, 32], confidence: 0.91 },
  { text: 'weekly', label: 'DOSAGE', span: [58, 64], confidence: 0.62 },
];

const metrics = computeMetrics(systemPredictions, expertAnnotations);
console.log('Precision:', metrics.precision.toFixed(3)); // 0.750 (3/4 predictions correct)
console.log('Recall:', metrics.recall.toFixed(3));       // 1.000 (found all 3 annotations)
console.log('F1:', metrics.f1.toFixed(3));               // 0.857`}

## Confusion Matrix

<CodeBlock language="typescript" title="confusion-matrix.ts">
{`// Confusion matrix: see exactly where the system makes mistakes
// Like a test report that shows which tests fail and why

type EntityLabel = 'DRUG' | 'CONDITION' | 'PROCEDURE' | 'DOSAGE' | 'ANATOMY';

interface ConfusionMatrix {
  labels: EntityLabel[];
  matrix: number[][];
  perClassMetrics: Map<EntityLabel, ValidationMetrics>;
}

function buildConfusionMatrix(
  predictions: Prediction[],
  annotations: Annotation[],
  labels: EntityLabel[]
): ConfusionMatrix {
  const size = labels.length;
  const matrix: number[][] = Array.from({ length: size }, () =>
    new Array(size).fill(0)
  );
  const perClassMetrics = new Map<EntityLabel, ValidationMetrics>();

  // Fill the matrix: rows = actual, columns = predicted
  for (const ann of annotations) {
    const actualIdx = labels.indexOf(ann.label as EntityLabel);
    const matchingPred = predictions.find(p =>
      spansOverlap(p.span, ann.span)
    );

    if (matchingPred) {
      const predIdx = labels.indexOf(matchingPred.label as EntityLabel);
      matrix[actualIdx][predIdx]++;
    }
  }

  // Compute per-class metrics
  for (const label of labels) {
    const classPreds = predictions.filter(p => p.label === label);
    const classAnns = annotations.filter(a => a.label === label);
    perClassMetrics.set(label, computeMetrics(classPreds, classAnns));
  }

  return { labels, matrix, perClassMetrics };
}

// Print the confusion matrix
function printConfusionMatrix(cm: ConfusionMatrix): void {
  console.log('Confusion Matrix:');
  console.log('Actual \\\\ Predicted', ...cm.labels);
  for (let i = 0; i < cm.labels.length; i++) {
    console.log(cm.labels[i], ...cm.matrix[i]);
  }

  console.log('\\nPer-class metrics:');
  for (const [label, metrics] of cm.perClassMetrics) {
    console.log(\`  \${label}: P=\${metrics.precision.toFixed(3)} R=\${metrics.recall.toFixed(3)} F1=\${metrics.f1.toFixed(3)}\`);
  }
}`}
</CodeBlock>
