<Dialogue speaker="Vikram">
API documentation. I write it for every service I build. What the endpoint accepts, what it returns, what errors it can throw, what the rate limits are, what edge cases to watch for. A model card is the same thing, except the endpoint is a trained model and the users are doctors.
</Dialogue>

<Dialogue speaker="Dr. Karimi">
And the edge cases could affect patient safety. So the documentation needs to be honest. Not marketing copy. Clinical documentation. What does Diagnostic One do well? Where does it fail? When should a clinician override it? If we cannot answer those questions clearly, it does not deploy.
</Dialogue>

A model card has five sections: intended use, validation results, known limitations, fairness analysis, override guidance. What the system accepts, what it returns, how accurate it is, where it fails, when not to trust it. If that skeleton feels familiar, it should. It is API documentation (endpoint, request, response, error codes, rate limits) written for readers who are clinicians making patient care decisions, not developers making API calls.

## The Model Card Template

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="model-card.ts">
{`// Model card generator — structured documentation for clinical ML systems
// Same pattern as API documentation generators (Swagger, OpenAPI)

interface ModelCardMetadata {
  name: string;
  version: string;
  description: string;
  intendedUse: string[];
  outOfScope: string[];
  metrics: {
    precision: number;
    recall: number;
    f1: number;
    validationSetSize: number;
    annotatorAgreement: number;
  };
  limitations: string[];
  fairnessAnalysis: {
    demographic: string;
    metric: string;
    value: number;
  }[];
  overrideGuidance: string[];
  inputFormat: string;
  outputFormat: string;
  lastUpdated: string;
}

function generateModelCard(meta: ModelCardMetadata): string {
  const lines = [
    \`# \${meta.name} v\${meta.version}\`,
    '',
    \`> \${meta.description}\`,
    '',
    '## Intended Use',
    ...meta.intendedUse.map(u => \`- \${u}\`),
    '',
    '## Out of Scope',
    ...meta.outOfScope.map(u => \`- \${u}\`),
    '',
    '## Performance',
    \`- Precision: \${(meta.metrics.precision * 100).toFixed(1)}%\`,
    \`- Recall: \${(meta.metrics.recall * 100).toFixed(1)}%\`,
    \`- F1 Score: \${(meta.metrics.f1 * 100).toFixed(1)}%\`,
    \`- Validation set: \${meta.metrics.validationSetSize} annotated notes\`,
    \`- Annotator agreement: \${(meta.metrics.annotatorAgreement * 100).toFixed(1)}%\`,
    '',
    '## Known Limitations',
    ...meta.limitations.map(l => \`- \${l}\`),
    '',
    '## Fairness Analysis',
    ...meta.fairnessAnalysis.map(f =>
      \`- \${f.demographic}: \${f.metric} = \${(f.value * 100).toFixed(1)}%\`
    ),
    '',
    '## When to Override',
    ...meta.overrideGuidance.map(g => \`- \${g}\`),
    '',
    '## Technical Details',
    \`- Input: \${meta.inputFormat}\`,
    \`- Output: \${meta.outputFormat}\`,
    \`- Last updated: \${meta.lastUpdated}\`,
  ];

  return lines.join('\\n');
}

// Generate Diagnostic One's model card
const diagnosticOneCard = generateModelCard({
  name: 'Diagnostic One',
  version: '1.0.0',
  description: 'Clinical NLP system for drug interaction detection and entity recognition in clinical notes',
  intendedUse: [
    'Flag potential drug-drug interactions in clinical notes for clinician review',
    'Extract drug names, conditions, and dosages from unstructured clinical text',
    'Support clinical decision-making as a secondary check, not a primary diagnostic tool',
  ],
  outOfScope: [
    'Independent diagnosis without clinician review',
    'Pediatric drug interactions (trained on adult data only)',
    'Non-English clinical notes',
    'Handwritten prescriptions or scanned documents',
  ],
  metrics: {
    precision: 0.89,
    recall: 0.84,
    f1: 0.86,
    validationSetSize: 500,
    annotatorAgreement: 0.92,
  },
  limitations: [
    'Higher false positive rate for rare drug combinations (precision drops to 0.71 for drugs with fewer than 10 training examples)',
    'Does not detect drug-food interactions',
    'Performance degrades on clinical abbreviations not in training vocabulary',
    'Confidence scores are not calibrated — 0.85 confidence does not mean 85% accuracy',
  ],
  fairnessAnalysis: [
    { demographic: 'Age 18-65', metric: 'F1', value: 0.87 },
    { demographic: 'Age 65+', metric: 'F1', value: 0.83 },
    { demographic: 'Male patients', metric: 'F1', value: 0.86 },
    { demographic: 'Female patients', metric: 'F1', value: 0.86 },
  ],
  overrideGuidance: [
    'Always verify flagged interactions against current drug reference databases',
    'Override the system when clinical context contradicts the flag (e.g., intentional co-prescription with monitoring)',
    'Report false positives and false negatives through the feedback interface to improve future versions',
    'Do not rely on absence of a flag as evidence of safety — the system may miss interactions',
  ],
  inputFormat: 'Unstructured English clinical text (UTF-8, max 2048 tokens)',
  outputFormat: '{ entities[], interactions[], classification, confidence, reviewTier }',
  lastUpdated: '2025-03-13',
});

console.log(diagnosticOneCard);`}
</CodeBlock>

## Version History

<CodeBlock language="typescript" title="version-history.ts">
{`// Track model versions — same pattern as API versioning (v1, v2, v2.1)
// Each version documents what changed, why, and how performance shifted

interface ModelVersion {
  version: string;
  date: string;
  changes: string[];
  metrics: { precision: number; recall: number; f1: number };
  breakingChanges: boolean;
}

const versionHistory: ModelVersion[] = [
  {
    version: '0.1.0',
    date: '2025-01-15',
    changes: ['Initial prototype with basic NER'],
    metrics: { precision: 0.72, recall: 0.65, f1: 0.68 },
    breakingChanges: false,
  },
  {
    version: '0.5.0',
    date: '2025-02-10',
    changes: [
      'Added transformer-based NER (replaced rule-based)',
      'Added drug interaction detection',
      'Added confidence-based review routing',
    ],
    metrics: { precision: 0.83, recall: 0.78, f1: 0.80 },
    breakingChanges: true,
  },
  {
    version: '1.0.0',
    date: '2025-03-13',
    changes: [
      'Validated against 500 expert-annotated notes',
      'Added bias auditing and fairness metrics',
      'Added human-in-the-loop review pipeline',
      'Approved for pilot deployment',
    ],
    metrics: { precision: 0.89, recall: 0.84, f1: 0.86 },
    breakingChanges: false,
  },
];

// Like a CHANGELOG.md — every version documented
function printVersionHistory(versions: ModelVersion[]): void {
  for (const v of versions) {
    console.log(\`## v\${v.version} (\${v.date})\${v.breakingChanges ? ' [BREAKING]' : ''}\`);
    for (const change of v.changes) {
      console.log(\`  - \${change}\`);
    }
    console.log(\`  Metrics: P=\${v.metrics.precision} R=\${v.metrics.recall} F1=\${v.metrics.f1}\`);
    console.log();
  }
}`}
