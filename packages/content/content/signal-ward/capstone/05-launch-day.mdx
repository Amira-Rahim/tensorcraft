// Infrastructure thresholds are illustrative, not clinical acceptance criteria
interface DeploymentConfig {
  modelPath: string;
  version: string;
  rolloutPercentage: number;
  healthCheckInterval: number;
  rollbackThreshold: {
    errorRate: number;
    latencyP95: number;
    memoryMB: number;
  };
}

const config: DeploymentConfig = {
  modelPath: '/models/diagnostic-one-v1/model.json',
  version: '1.0.0',
  rolloutPercentage: 10, // Start with 10% of traffic
  healthCheckInterval: 30000, // Check every 30 seconds
  rollbackThreshold: {
    errorRate: 0.05,    // Rollback if >5% errors
    latencyP95: 500,    // Rollback if p95 latency >500ms
    memoryMB: 512,      // Rollback if memory >512MB
  },
};

// Feature flag, same pattern as LaunchDarkly, Unleash, or a simple config
class FeatureFlag {
  private flags: Map<string, { enabled: boolean; percentage: number }> = new Map();

  setFlag(name: string, enabled: boolean, percentage: number = 100): void {
    this.flags.set(name, { enabled, percentage });
  }

  isEnabled(name: string, userId?: string): boolean {
    const flag = this.flags.get(name);
    if (!flag || !flag.enabled) return false;
    if (flag.percentage >= 100) return true;

    // Consistent hashing: same user always gets same result
    const hash = userId
      ? userId.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0) % 100
      : Math.random() * 100;
    return hash < flag.percentage;
  }

  // Stop new requests; this does not reverse prior side effects
  rollback(name: string): void {
    this.setFlag(name, false, 0);
    console.log(\`ROLLBACK: \${name} disabled\`);
  }
}

const flags = new FeatureFlag();
flags.setFlag('diagnostic-one', true, 10); // 10% rollout`}

## Health Checks and Monitoring

<CodeBlock language="typescript" title="monitoring.ts">
{`// Monitoring dashboard, same as Datadog, Grafana, or your custom metrics
// Track model health alongside application health

interface HealthCheck {
  name: string;
  status: 'healthy' | 'degraded' | 'unhealthy';
  latencyMs: number;
  message: string;
}

interface MonitoringMetrics {
  totalPredictions: number;
  errorCount: number;
  avgLatencyMs: number;
  p95LatencyMs: number;
  memoryUsageMB: number;
  alertsFired: number;
  falsePositives: number;
  truePositives: number;
}

class DiagnosticOneMonitor {
  private metrics: MonitoringMetrics = {
    totalPredictions: 0,
    errorCount: 0,
    avgLatencyMs: 0,
    p95LatencyMs: 0,
    memoryUsageMB: 0,
    alertsFired: 0,
    falsePositives: 0,
    truePositives: 0,
  };
  private latencies: number[] = [];

  recordPrediction(latencyMs: number, success: boolean): void {
    this.metrics.totalPredictions++;
    if (!success) this.metrics.errorCount++;
    this.latencies.push(latencyMs);
    this.updateLatencyMetrics();
  }

  recordMemory(memoryMB: number): void {
    this.metrics.memoryUsageMB = memoryMB;
  }

  recordAlert(isTrue: boolean): void {
    this.metrics.alertsFired++;
    if (isTrue) this.metrics.truePositives++;
    else this.metrics.falsePositives++;
  }

  private updateLatencyMetrics(): void {
    const sorted = [...this.latencies].sort((a, b) => a - b);
    this.metrics.avgLatencyMs = sorted.reduce((a, b) => a + b, 0) / sorted.length;
    this.metrics.p95LatencyMs = sorted[Math.floor(sorted.length * 0.95)] ?? 0;
  }

  async runHealthChecks(): Promise<HealthCheck[]> {
    const checks: HealthCheck[] = [];

    // Check 1: Model loads
    const modelStart = performance.now();
    try {
      const model = await tf.loadLayersModel(config.modelPath);
      model.dispose();
      checks.push({
        name: 'model-load',
        status: 'healthy',
        latencyMs: performance.now() - modelStart,
        message: 'Model loaded successfully',
      });
    } catch (error) {
      checks.push({
        name: 'model-load',
        status: 'unhealthy',
        latencyMs: performance.now() - modelStart,
        message: \`Model failed to load: \${error}\`,
      });
    }

    // Check 2: Error rate
    const errorRate = this.metrics.errorCount / (this.metrics.totalPredictions || 1);
    checks.push({
      name: 'error-rate',
      status: errorRate < 0.05 ? 'healthy' : errorRate < 0.1 ? 'degraded' : 'unhealthy',
      latencyMs: 0,
      message: \`Error rate: \${(errorRate * 100).toFixed(2)}%\`,
    });

    // Check 3: Latency
    checks.push({
      name: 'latency',
      status: this.metrics.p95LatencyMs < 500 ? 'healthy' : 'degraded',
      latencyMs: this.metrics.p95LatencyMs,
      message: \`p95 latency: \${this.metrics.p95LatencyMs.toFixed(1)}ms\`,
    });

    return checks;
  }

  shouldRollback(): boolean {
    const errorRate = this.metrics.errorCount / (this.metrics.totalPredictions || 1);
    return (
      errorRate > config.rollbackThreshold.errorRate ||
      this.metrics.p95LatencyMs > config.rollbackThreshold.latencyP95 ||
      this.metrics.memoryUsageMB > config.rollbackThreshold.memoryMB
    );
  }

  getMetrics(): MonitoringMetrics {
    return { ...this.metrics };
  }
}`}
</CodeBlock>

## One Alert and Its Limits

The fictional system has found a known interaction. A model score cannot establish that a drug combination is safe. The [NHS Specialist Pharmacy Service](https://www.sps.nhs.uk/articles/managing-interactions-with-methotrexate/) warns against combining trimethoprim with methotrexate because of severe bone marrow suppression risk. This exercise concerns routing evidence for qualified review, not making a prescribing decision.
