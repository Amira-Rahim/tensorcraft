<Dialogue speaker="Khalil">
It is the same problem as building a search bar for an e-commerce site. Users type "red shoes under fifty bucks" and the system needs to parse that into structured filters. Here, doctors and nurses type clinical questions and we parse them into structured queries against patient records.
</Dialogue>

Query parsing is a solved problem in one direction and open in the other. `red shoes under $50` has a closed vocabulary: a color list, a category list, a number. Clinical language has neither. "Blood thinner" and "anticoagulant" name the same drug class, "MS" is multiple sclerosis in neurology and morphine sulfate on a medication list, and the synonym table is never finished. Every JavaScript engineer who has shipped a search bar knows the first half of this job; the second half is where one missed expansion quietly returns fewer patients than it should.

## From Natural Language to Structured Queries

<CodeBlock language="typescript" title="clinical-query-parser.ts">
{`import * as tf from '@tensorflow/tfjs';

// Clinical query intents
type QueryIntent = 'patient_lookup' | 'lab_check' | 'medication_review'
  | 'allergy_check' | 'schedule_query' | 'comparison';

// Structured clinical query
interface ClinicalQuery {
  intent: QueryIntent;
  entities: {
    medications?: string[];
    conditions?: string[];
    labTests?: string[];
    timeRange?: { start: string; end: string };
    threshold?: { field: string; operator: '>' | '<' | '='; value: number };
  };
  originalText: string;
}

// Intent classifier: which type of clinical question is this?
function buildIntentClassifier(vocabSize: number, maxLen: number): tf.LayersModel {
  const input = tf.input({ shape: [maxLen] });

  const embedded = tf.layers.embedding({ inputDim: vocabSize, outputDim: 64 })
    .apply(input) as tf.SymbolicTensor;

  const encoded = tf.layers.bidirectional({
    layer: tf.layers.lstm({ units: 32 }),
  }).apply(embedded) as tf.SymbolicTensor;

  const output = tf.layers.dense({
    units: 6,  // 6 intent types
    activation: 'softmax',
  }).apply(encoded) as tf.SymbolicTensor;

  return tf.model({ inputs: input, outputs: output });
}

// Query expansion: normalize clinical synonyms
const medicalSynonyms: Record<string, string> = {
  'blood thinner': 'anticoagulant',
  'blood pressure medicine': 'antihypertensive',
  'sugar levels': 'glucose',
  'blood checked': 'complete blood count',
  'heart medicine': 'cardiovascular medication',
  'pain killer': 'analgesic',
};

function expandQuery(query: string): string {
  let expanded = query.toLowerCase();
  for (const [informal, formal] of Object.entries(medicalSynonyms)) {
    expanded = expanded.replace(informal, \`\${informal} (\${formal})\`);
  }
  return expanded;
}

// Pad with zeros or truncate so every sequence matches the model's input width
function fitTo(ids: number[], width: number): number[] {
  return ids.length >= width
    ? ids.slice(0, width)
    : [...ids, ...new Array(width - ids.length).fill(0)];
}

// Parse a natural language clinical query
async function parseClinicalQuery(
  queryText: string,
  intentModel: tf.LayersModel,
  nerModel: (text: string) => Promise<{ entity: string; label: string }[]>,
  tokenize: (text: string) => number[]
): Promise<ClinicalQuery> {
  // 1. Expand query with medical synonyms
  const expanded = expandQuery(queryText);

  // 2. Classify intent. predict() takes fixed-width input,
  //    so pad or truncate before calling it
  const width = intentModel.inputs[0].shape[1] as number;
  const tokens = fitTo(tokenize(expanded), width);
  const intentPred = intentModel.predict(
    tf.tensor2d([tokens])
  ) as tf.Tensor;
  const intentIndex = (await intentPred.argMax(1).array() as number[])[0];
  const intents: QueryIntent[] = [
    'patient_lookup', 'lab_check', 'medication_review',
    'allergy_check', 'schedule_query', 'comparison'
  ];

  // 3. Extract clinical entities
  const entities = await nerModel(expanded);
  const medications = entities.filter(e => e.label === 'MEDICATION').map(e => e.entity);
  const conditions = entities.filter(e => e.label === 'CONDITION').map(e => e.entity);
  const labTests = entities.filter(e => e.label === 'LAB_TEST').map(e => e.entity);

  intentPred.dispose();

  return {
    intent: intents[intentIndex],
    entities: { medications, conditions, labTests },
    originalText: queryText,
  };
}`}
</CodeBlock>

## Handling Role-Based Query Variability

<CodeBlock language="typescript" title="query-normalization.ts">
{`// Same clinical intent, different phrasing by role
const queryVariants = [
  {
    doctor: "List current anticoagulant patients with subtherapeutic INR",
    nurse: "Who needs their blood thinner levels checked?",
    expectedIntent: 'lab_check' as QueryIntent,
    expectedMedication: 'anticoagulant',
  },
  {
    doctor: "Patients on methotrexate requiring hepatic function monitoring",
    nurse: "Who on methotrexate needs their liver checked?",
    expectedIntent: 'lab_check' as QueryIntent,
    expectedMedication: 'methotrexate',
  },
];

// Both phrasings should produce the same structured query
for (const variant of queryVariants) {
  const doctorQuery = await parseClinicalQuery(variant.doctor, intentModel, nerModel, tokenize);
  const nurseQuery = await parseClinicalQuery(variant.nurse, intentModel, nerModel, tokenize);

  console.log('Doctor:', doctorQuery.intent, doctorQuery.entities.medications);
  console.log('Nurse: ', nurseQuery.intent, nurseQuery.entities.medications);
  console.log('Intent match:', doctorQuery.intent === nurseQuery.intent);
}`}
</CodeBlock>
