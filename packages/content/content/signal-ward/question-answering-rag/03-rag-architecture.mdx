<Dialogue speaker="Khalil">
It is the same pattern as React Server Components. Fetch the data you need on the server, then render on the client. Retrieve the relevant documents first, then pass them to the QA model. Retrieval-Augmented Generation.
</Dialogue>

React Server Components changed how we think about data fetching. Instead of shipping a loading spinner and fetching on the client, you fetch on the server and stream the result. RAG does the same thing for question answering: instead of feeding the entire corpus to the model, you retrieve the relevant documents first, then pass only those as context.

## The RAG Pattern

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="rag-pipeline.ts">
{`import * as tf from '@tensorflow/tfjs';

// RAG Pipeline: Retrieve → Augment → Generate
// Same pattern as React Server Components: fetch data, then render

interface Document {
  id: string;
  text: string;
  metadata: { language: string; docType: string; date: string };
}

interface RetrievalResult {
  document: Document;
  score: number;
}

class VectorIndex {
  private embeddings: tf.Tensor2D | null = null;
  private documents: Document[] = [];

  // Index documents by computing and storing their embeddings
  async index(docs: Document[], embedFn: (text: string) => Promise<tf.Tensor1D>): Promise<void> {
    this.documents = docs;
    const vectors: tf.Tensor1D[] = [];

    for (const doc of docs) {
      const embedding = await embedFn(doc.text);
      vectors.push(embedding);
    }

    // Stack into a matrix: [numDocs, embDim]
    this.embeddings = tf.stack(vectors) as tf.Tensor2D;
    vectors.forEach(v => v.dispose());  // Clean up individual tensors
  }

  // Retrieve top-k documents by cosine similarity
  async retrieve(query: string, embedFn: (text: string) => Promise<tf.Tensor1D>, k: number = 3): Promise<RetrievalResult[]> {
    if (!this.embeddings) throw new Error('Index is empty');

    const queryEmb = await embedFn(query);

    // Cosine similarity: dot product of normalized vectors
    const queryNorm = queryEmb.div(queryEmb.norm());
    const docsNorm = this.embeddings.div(
      this.embeddings.norm('euclidean', 1, true)
    );

    const scores = tf.matMul(docsNorm, queryNorm.expandDims(1))
      .squeeze([1]);

    const scoreArray = await scores.array() as number[];

    // Sort by score, take top k
    const ranked = scoreArray
      .map((score, i) => ({ document: this.documents[i], score }))
      .sort((a, b) => b.score - a.score)
      .slice(0, k);

    queryEmb.dispose();
    queryNorm.dispose();
    docsNorm.dispose();
    scores.dispose();

    return ranked;
  }
}

// Complete RAG pipeline
async function ragAnswer(
  question: string,
  index: VectorIndex,
  embedFn: (text: string) => Promise<tf.Tensor1D>,
  qaModel: (question: string, context: string) => Promise<string>
): Promise<{ answer: string; sources: RetrievalResult[] }> {
  // 1. Retrieve relevant documents
  const retrieved = await index.retrieve(question, embedFn, 3);

  // 2. Augment: combine retrieved docs into context
  const context = retrieved
    .map(r => r.document.text)
    .join('\\n---\\n');

  // 3. Generate answer from context
  const answer = await qaModel(question, context);

  return { answer, sources: retrieved };
}`}
