<Dialogue speaker="Vikram">
Your QA model found the word "penicillin" in the record. But read this note carefully: "Patient's mother has documented penicillin allergy. Patient has no known drug allergies." A keyword search would flag this as a positive. A comprehension model wouldn't.
</Dialogue>

<Dialogue speaker="Khalil">
It's like the difference between a REST endpoint that dumps everything and a GraphQL resolver that returns exactly what you asked for. The model needs to understand the relationship between the question and the specific context.
</Dialogue>

A GraphQL resolver does not return the entire database. It parses your query, understands what you need, and fetches precisely that. Reading comprehension works the same way. The model uses cross-attention to connect question tokens to the most relevant context tokens, building a query-specific understanding of the passage.

{`import * as tf from '@tensorflow/tfjs';

// Cross-attention: question attends to context
// Like a GraphQL resolver matching query fields to database columns
function crossAttention(
  questionEmb: tf.Tensor2D,  // [questionLen, dim]
  contextEmb: tf.Tensor2D    // [contextLen, dim]
): tf.Tensor2D {
  const dim = questionEmb.shape[1];
  const scale = Math.sqrt(dim);

  // Attention scores: which context tokens matter for each question token
  const scores = tf.matMul(questionEmb, contextEmb, false, true)
    .div(tf.scalar(scale));

  // Normalize to probabilities
  const weights = tf.softmax(scores, -1);  // [questionLen, contextLen]

  // Weighted sum of context: question-aware representation
  const attended = tf.matMul(weights, contextEmb);  // [questionLen, dim]
  return attended as tf.Tensor2D;
}

// Build a comprehension model with cross-attention
function buildComprehensionModel(
  vocabSize: number,
  embDim: number,
  maxLen: number
): tf.LayersModel {
  const questionInput = tf.input({ shape: [maxLen], name: 'question' });
  const contextInput = tf.input({ shape: [maxLen], name: 'context' });

  const embedding = tf.layers.embedding({
    inputDim: vocabSize,
    outputDim: embDim,
    name: 'shared_embedding',
  });

  const questionEmb = embedding.apply(questionInput) as tf.SymbolicTensor;
  const contextEmb = embedding.apply(contextInput) as tf.SymbolicTensor;

  // Bidirectional encoding for richer representations
  const biLSTM = tf.layers.bidirectional({
    layer: tf.layers.lstm({ units: embDim / 2, returnSequences: true }),
  });

  const questionEncoded = biLSTM.apply(questionEmb) as tf.SymbolicTensor;
  const contextEncoded = biLSTM.apply(contextEmb) as tf.SymbolicTensor;

  // Attention-based fusion. TFJS ships no attention layer, so build
  // Luong-style attention from layers that do exist: dot for scores,
  // softmax for weights, dot again to pool the question, concat to fuse.
  const scores = tf.layers.dot({ axes: -1 })
    .apply([contextEncoded, questionEncoded]) as tf.SymbolicTensor;
  const weights = tf.layers.softmax()
    .apply(scores) as tf.SymbolicTensor;
  const pooledQuestion = tf.layers.dot({ axes: [2, 1] })
    .apply([weights, questionEncoded]) as tf.SymbolicTensor;
  const attended = tf.layers.concatenate()
    .apply([contextEncoded, pooledQuestion]) as tf.SymbolicTensor;

  // Predict answer span
  const startLogits = tf.layers.dense({ units: 1, name: 'start' })
    .apply(attended) as tf.SymbolicTensor;
  const endLogits = tf.layers.dense({ units: 1, name: 'end' })
    .apply(attended) as tf.SymbolicTensor;

  return tf.model({
    inputs: [questionInput, contextInput],
    outputs: [
      tf.layers.flatten().apply(startLogits) as tf.SymbolicTensor,
      tf.layers.flatten().apply(endLogits) as tf.SymbolicTensor,
    ],
  });
}`}

## Adversarial Clinical Examples

{`// Test comprehension with tricky clinical scenarios
const adversarialTests = [
  {
    question: "Is the patient allergic to penicillin?",
    context: "Patient's mother has documented penicillin allergy. Patient has no known drug allergies.",
    expectedAnswer: "Patient has no known drug allergies",
    label: "family-vs-patient"
  },
  {
    question: "Is the patient currently taking warfarin?",
    context: "Warfarin was discontinued on 2023-01-15 due to bleeding risk. Current medications: metformin 500mg, lisinopril 10mg.",
    expectedAnswer: "Warfarin was discontinued",
    label: "current-vs-historical"
  },
  {
    question: "Does the patient have diabetes?",
    context: "Rule out diabetes. HbA1c 5.4% (normal). No evidence of insulin resistance.",
    expectedAnswer: "No evidence of insulin resistance",
    label: "rule-out-vs-diagnosis"
  },
];

// A model that truly comprehends will get all three right.
// A keyword matcher will fail on every one.
for (const test of adversarialTests) {
  const result = await comprehend(test.question, test.context);
  console.log(\`[\${test.label}] Answer: "\${result.answer}"\`);
  console.log(\`  Expected: "\${test.expectedAnswer}"\`);
  console.log(\`  Correct: \${result.answer.includes(test.expectedAnswer)}\`);
}`}
