`notes.filter(n => n.text.includes('penicillin allergy'))` returns zero rows for this patient. The discharge summary says "exhibited hypersensitivity to penicillin-class agents," and substring matching cannot see that those are the same fact. Neither can Ctrl+F. Extractive QA closes the gap: the model reads the question and the document together, matches on *meaning* rather than characters, and returns the span that answers it.

## From Ctrl+F to Semantic Search

<BridgeCallout bridge={frontmatter.concept.bridge} />

The model reads both the question and the context passage, then predicts two positions: where the answer **starts** and where it **ends**.

<CodeBlock language="typescript" title="extractive-qa.ts">
{`import * as tf from '@tensorflow/tfjs';

// Extractive QA: predict start and end positions of the answer span
// Input: question + context encoded as token IDs
// Output: start logits and end logits over all token positions

interface QAInput {
  questionTokens: number[];
  contextTokens: number[];
  // Combined: [CLS] question [SEP] context [SEP]
}

interface QAOutput {
  answer: string;
  startIndex: number;
  endIndex: number;
  confidence: number;
}

function buildQAHead(embeddingDim: number, seqLength: number): tf.LayersModel {
  const input = tf.input({ shape: [seqLength, embeddingDim] });

  // Start position predictor
  const startDense = tf.layers.dense({ units: 1, name: 'start_logits' })
    .apply(input) as tf.SymbolicTensor;
  const startLogits = tf.layers.flatten().apply(startDense) as tf.SymbolicTensor;

  // End position predictor
  const endDense = tf.layers.dense({ units: 1, name: 'end_logits' })
    .apply(input) as tf.SymbolicTensor;
  const endLogits = tf.layers.flatten().apply(endDense) as tf.SymbolicTensor;

  return tf.model({ inputs: input, outputs: [startLogits, endLogits] });
}

// Extract the answer span
function extractAnswer(
  startLogits: Float32Array,
  endLogits: Float32Array,
  tokens: string[],
  contextStart: number
): QAOutput {
  // Only consider positions within the context (not the question)
  const startProbs = tf.softmax(tf.tensor1d(startLogits)).arraySync() as number[];
  const endProbs = tf.softmax(tf.tensor1d(endLogits)).arraySync() as number[];

  let bestScore = -Infinity;
  let bestStart = contextStart;
  let bestEnd = contextStart;

  for (let s = contextStart; s < tokens.length; s++) {
    for (let e = s; e < Math.min(s + 30, tokens.length); e++) {
      const score = startProbs[s] + endProbs[e];
      if (score > bestScore) {
        bestScore = score;
        bestStart = s;
        bestEnd = e;
      }
    }
  }

  const answerTokens = tokens.slice(bestStart, bestEnd + 1);
  return {
    answer: answerTokens.join(' '),
    startIndex: bestStart,
    endIndex: bestEnd,
    confidence: Math.exp(bestScore) / 2,
  };
}`}
</CodeBlock>
