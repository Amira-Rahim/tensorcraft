## The Stale Cache Problem

<BridgeCallout bridge={frontmatter.concept.bridge} />

Every frontend engineer has fought stale caches. You update user data in the database, but the CDN keeps serving the old version. The data changed, but the system's representation didn't. MEDI has the same problem — "Type 2 diabetes" activates strong patterns from training data, and the model's internal representation fails to update when it encounters "resolved."

<Dialogue speaker="Vikram">
This is a negation detection failure. MEDI's model learned strong associations: "Type 2 diabetes" maps to "active condition" in the vast majority of training examples. The word "resolved" should invalidate that classification, but the model's attention weights didn't give it enough importance.
</Dialogue>

interface AttentionAnalysis {

  attentionWeight: number;
  contribution: 'positive' | 'negative' | 'neutral';

function analyzeAttention(

### Defense 1: Confidence Thresholds

<CodeBlock language="typescript" title="confidence-threshold.ts">
{`import * as tf from '@tensorflow/tfjs';

interface PredictionResult {
  label: string;
  confidence: number;
  requiresReview: boolean;
  reason?: string;
}

function predictWithConfidence(
  logits: tf.Tensor1D,
  labels: string[],
  threshold: number = 0.85
): PredictionResult {
  const probabilities = tf.softmax(logits);
  const values = probabilities.arraySync() as number[];
  const maxIdx = values.indexOf(Math.max(...values));
  const confidence = values[maxIdx];

  return {
    label: labels[maxIdx],
    confidence,
    requiresReview: confidence < threshold,
    reason: confidence < threshold
      ? \`Confidence \${(confidence * 100).toFixed(1)}% below threshold \${(threshold * 100).toFixed(1)}%\`
      : undefined
  };
}

const labels = ['Active', 'Resolved', 'Chronic', 'Suspected', 'Ruled-out'];

// High confidence — model is certain
const clearCase = tf.tensor1d([3.2, -1.0, 0.1, -0.5, -2.0]);
console.log('Clear case:', predictWithConfidence(clearCase, labels));
// → { label: 'Active', confidence: 0.94, requiresReview: false }

// MEDI's problematic case — model is uncertain
const ambiguousCase = tf.tensor1d([1.1, 0.9, 0.3, -0.2, -0.5]);
console.log('Ambiguous case:', predictWithConfidence(ambiguousCase, labels));
// → { label: 'Active', confidence: 0.38, requiresReview: true }`}
</CodeBlock>

### Defense 2: Negation Detection

<CodeBlock language="typescript" title="negation-detection.ts">
{`function detectNegation(text: string, entity: string): {
  isNegated: boolean;
  negationCue: string | null;
  windowText: string;
} {
  const negationPatterns = [
    /\b(no|not|without|denies?|denied|absent|negative|resolved|ruled[- ]?out|discontinued|former|history of)\b/i
  ];

  const entityIdx = text.toLowerCase().indexOf(entity.toLowerCase());
  if (entityIdx === -1) return { isNegated: false, negationCue: null, windowText: '' };

  // Check a window of 6 words before and 3 words after the entity
  const words = text.split(/\s+/);
  const entityWordIdx = words.findIndex(w =>
    w.toLowerCase().includes(entity.toLowerCase().split(' ')[0])
  );

  const windowStart = Math.max(0, entityWordIdx - 6);
  const windowEnd = Math.min(words.length, entityWordIdx + entity.split(' ').length + 3);
  const window = words.slice(windowStart, windowEnd).join(' ');

  for (const pattern of negationPatterns) {
    const match = window.match(pattern);
    if (match) {
      return { isNegated: true, negationCue: match[1], windowText: window };
    }
  }

  return { isNegated: false, negationCue: null, windowText: window };
}

// MEDI's problematic note
const note = "History of Type 2 diabetes mellitus, resolved following bariatric surgery";
const result = detectNegation(note, "diabetes");
console.log(result);
// → { isNegated: true, negationCue: 'resolved', windowText: 'History of Type 2 diabetes mellitus, resolved following' }

// A truly active condition
const activeNote = "Patient presents with uncontrolled Type 2 diabetes mellitus, A1C 9.8%";
console.log(detectNegation(activeNote, "diabetes"));
// → { isNegated: false, negationCue: null, windowText: '...' }`}
</CodeBlock>

### Defense 3: Human-in-the-Loop

<CodeBlock language="typescript" title="human-in-the-loop.ts">
{`interface ClinicalPrediction {
  condition: string;
  status: string;
  confidence: number;
  negationDetected: boolean;
  action: 'auto-accept' | 'flag-for-review' | 'auto-reject';
}

function triagePrediction(
  condition: string,
  predictedStatus: string,
  confidence: number,
  negationDetected: boolean,
  confidenceThreshold: number = 0.85
): ClinicalPrediction {
  let action: ClinicalPrediction['action'];

  if (negationDetected && predictedStatus === 'Active') {
    // Model says active, but negation detected — conflict!
    action = 'flag-for-review';
  } else if (confidence < confidenceThreshold) {
    // Low confidence — human should verify
    action = 'flag-for-review';
  } else if (negationDetected && predictedStatus !== 'Resolved') {
    // Negation detected but model didn't predict resolved
    action = 'flag-for-review';
  } else {
    action = 'auto-accept';
  }

  return { condition, status: predictedStatus, confidence, negationDetected, action };
}

// MEDI's mistake — caught by the safety net
const mediResult = triagePrediction('Type 2 diabetes', 'Active', 0.72, true);
console.log(mediResult);
// → { action: 'flag-for-review' } — negation + active = conflict

// Clear active condition — no negation, high confidence
const clearResult = triagePrediction('Hypertension', 'Active', 0.95, false);
console.log(clearResult);
// → { action: 'auto-accept' }`}
</CodeBlock>
