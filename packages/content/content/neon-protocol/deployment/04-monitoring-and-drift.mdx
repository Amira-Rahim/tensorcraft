Deploying a model is not the end of the job. It is the beginning of a new one. Models decay. The real world changes. Data drift is the ML equivalent of an API breaking because a third-party service changed their response format. You need monitoring to catch it.

## Error Tracking for Models

<CodeBlock language="typescript" title="model-monitor.ts">
{`interface MonitorConfig {
  confidenceThreshold: number;
  windowSize: number;
  driftAlertThreshold: number;
}

interface PredictionLog {
  timestamp: number;
  confidence: number;
  classIndex: number;
  inferenceMs: number;
}

class ModelMonitor {
  private config: MonitorConfig;
  private logs: PredictionLog[] = [];
  private baselineConfidences: number[] = [];
  private alertCallbacks: ((message: string) => void)[] = [];

  constructor(config: MonitorConfig) {
    this.config = config;
  }

  // Record baseline during initial healthy deployment
  setBaseline(confidences: number[]): void {
    this.baselineConfidences = confidences;
  }

  // Log every prediction
  log(prediction: PredictionLog): void {
    this.logs.push(prediction);

    // Check for low confidence
    if (prediction.confidence < this.config.confidenceThreshold) {
      this.alert(
        \`Low confidence prediction: \${(prediction.confidence * 100).toFixed(1)}% at \${new Date(prediction.timestamp).toISOString()}\`
      );
    }

    // Check for drift on rolling window
    if (this.logs.length >= this.config.windowSize) {
      this.checkDrift();
    }
  }

  private checkDrift(): void {
    const recent = this.logs.slice(-this.config.windowSize);
    const recentMean = recent.reduce((s, l) => s + l.confidence, 0) / recent.length;
    const baselineMean = this.baselineConfidences.reduce((a, b) => a + b, 0)
      / this.baselineConfidences.length;

    const drift = baselineMean - recentMean;

    if (drift > this.config.driftAlertThreshold) {
      this.alert(
        \`Data drift detected: confidence dropped \${(drift * 100).toFixed(1)}% from baseline (\${(baselineMean * 100).toFixed(1)}% → \${(recentMean * 100).toFixed(1)}%)\`
      );
    }
  }

  onAlert(callback: (message: string) => void): void {
    this.alertCallbacks.push(callback);
  }

  private alert(message: string): void {
    this.alertCallbacks.forEach(cb => cb(message));
  }

  getMetrics() {
    const recent = this.logs.slice(-this.config.windowSize);
    return {
      totalPredictions: this.logs.length,
      avgConfidence: recent.reduce((s, l) => s + l.confidence, 0) / recent.length,
      avgLatencyMs: recent.reduce((s, l) => s + l.inferenceMs, 0) / recent.length,
      lowConfidenceRate: recent.filter(
        l => l.confidence < this.config.confidenceThreshold
      ).length / recent.length
    };
  }
}`}
</CodeBlock>

## Detecting Distribution Shift

Data drift means the incoming data looks different from what the model was trained on. You can detect it by comparing simple statistics.

<CodeBlock language="typescript" title="drift-detection.ts">
{`interface DistributionStats {
  mean: number;
  stdDev: number;
  min: number;
  max: number;
}

function computeStats(values: number[]): DistributionStats {
  const mean = values.reduce((a, b) => a + b, 0) / values.length;
  const variance = values.reduce((sum, v) => sum + (v - mean) ** 2, 0) / values.length;
  return {
    mean,
    stdDev: Math.sqrt(variance),
    min: Math.min(...values),
    max: Math.max(...values)
  };
}

function detectDrift(
  baseline: DistributionStats,
  current: DistributionStats,
  threshold: number = 2.0
): { isDrifted: boolean; severity: number; details: string } {
  // Normalized mean shift: how many baseline std devs has the mean moved?
  const meanShift = Math.abs(current.mean - baseline.mean) / baseline.stdDev;

  // Variance ratio: has the spread of predictions changed?
  const varianceRatio = current.stdDev / baseline.stdDev;

  const isDrifted = meanShift > threshold || varianceRatio > 2.0 || varianceRatio < 0.5;

  return {
    isDrifted,
    severity: meanShift,
    details: isDrifted
      ? \`Mean shifted \${meanShift.toFixed(2)} std devs. Variance ratio: \${varianceRatio.toFixed(2)}\`
      : 'Distribution stable'
  };
}

// Usage with the monitor
const baselineStats = computeStats(baselineConfidences);

setInterval(() => {
  const recentConfidences = monitor.getRecentConfidences(100);
  const currentStats = computeStats(recentConfidences);
  const driftResult = detectDrift(baselineStats, currentStats);

  if (driftResult.isDrifted) {
    console.warn('DRIFT ALERT:', driftResult.details);
    // Trigger retraining pipeline or alert the team
  }
}, 60_000); // check every minute`}
</CodeBlock>

// Alert handler
monitor.onAlert((msg) => {
  console.warn('[SENTINEL MONITOR]', msg);
  // Send to your alerting system
  fetch('/api/alerts', {
    method: 'POST',
    body: JSON.stringify({ source: 'sentinel', message: msg })
  });
});

// Monitored prediction
async function monitoredPredict(imageData: ImageData) {
  const start = performance.now();
  const result = await sentinel.predict(imageData);
  const inferenceMs = performance.now() - start;

  monitor.log({
    timestamp: Date.now(),
    confidence: result.confidence,
    classIndex: result.classIndex,
    inferenceMs
  });

  return result;
}`}
