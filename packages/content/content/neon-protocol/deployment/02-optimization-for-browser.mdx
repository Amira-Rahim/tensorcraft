## Tree-Shaking for Neural Networks

<BridgeCallout bridge={frontmatter.concept.bridge} />

In JavaScript builds, tree-shaking analyzes your dependency graph and strips code that is never called. Model optimization applies the same logic to neural networks: quantization reduces the precision of every weight (like minification), and pruning removes weights that contribute almost nothing to the output (like dead code elimination).

<CodeBlock language="typescript" title="quantization.ts">
{`import * as tf from '@tensorflow/tfjs';

// Load the full-precision model (float32, 4 bytes per weight)
const model = await tf.loadLayersModel('/models/sentinel/model.json');

// Check original size
const originalWeights = model.getWeights();
const originalBytes = originalWeights.reduce(
  (sum, w) => sum + w.size * 4, 0  // float32 = 4 bytes
);
console.log('Original size:', (originalBytes / 1024 / 1024).toFixed(1), 'MB');

// Save with float16 quantization (2 bytes per weight, 50% reduction)
await model.save('downloads://sentinel-q16', {
  weightDataQuantizationBits: 16
});

// Save with uint8 quantization (1 byte per weight, 75% reduction)
await model.save('downloads://sentinel-q8', {
  weightDataQuantizationBits: 8
});

// Compare: 45MB → 22.5MB (float16) → 11.25MB (uint8)`}
</CodeBlock>

## Weight Pruning

Pruning zeros out weights that are close to zero; they contribute almost nothing to the model's decisions. After pruning, you retrain briefly to recover any lost accuracy.

<CodeBlock language="typescript" title="pruning.ts">
{`import * as tf from '@tensorflow/tfjs';

function pruneWeights(
  model: tf.LayersModel,
  sparsityTarget: number
): tf.LayersModel {
  const weights = model.getWeights();

  const pruned = weights.map(weightTensor => {
    return tf.tidy(() => {
      const absWeights = weightTensor.abs();
      const flat = absWeights.reshape([-1]);
      const sorted = tf.topk(flat, flat.size).values;

      // Find the threshold: bottom X% of weights get zeroed
      const cutoffIndex = Math.floor(flat.size * sparsityTarget);
      const thresholdValue = sorted.slice(
        [sorted.size - cutoffIndex - 1], [1]
      );
      const threshold = thresholdValue.dataSync()[0];

      // Create binary mask: 1 where weight > threshold, 0 otherwise
      const mask = absWeights.greater(threshold).cast('float32');
      return weightTensor.mul(mask);
    });
  });

  model.setWeights(pruned);
  return model;
}

// Prune 50% of smallest weights
const prunedModel = pruneWeights(model, 0.5);

// Fine-tune to recover accuracy (a few epochs with low learning rate)
prunedModel.compile({
  optimizer: tf.train.adam(0.0001),
  loss: 'categoricalCrossentropy',
  metrics: ['accuracy']
});`}
</CodeBlock>

## Measuring the Trade-Off

Every optimization has a cost. You need to measure it.

<CodeBlock language="typescript" title="benchmark.ts">
{`interface OptimizationResult {
  label: string;
  sizeBytes: number;
  sizeMB: number;
  accuracy: number;
  loadTimeMs: number;
  inferenceTimeMs: number;
}

async function benchmarkModel(
  modelPath: string,
  testData: tf.Tensor,
  testLabels: tf.Tensor,
  label: string
): Promise<OptimizationResult> {
  // Measure load time
  const loadStart = performance.now();
  const model = await tf.loadLayersModel(modelPath);
  const loadTimeMs = performance.now() - loadStart;

  // Measure size
  const weights = model.getWeights();
  const sizeBytes = weights.reduce((sum, w) => sum + w.size * 4, 0);

  // Measure accuracy
  const predictions = model.predict(testData) as tf.Tensor;
  const predLabels = predictions.argMax(-1);
  const trueLabels = testLabels.argMax(-1);
  const correct = predLabels.equal(trueLabels).sum();
  const accuracy = (await correct.data())[0] / testLabels.shape[0];

  // Measure inference time (average of 100 runs)
  const sample = testData.slice([0], [1]);
  const times: number[] = [];
  for (let i = 0; i < 100; i++) {
    const start = performance.now();
    const pred = model.predict(sample) as tf.Tensor;
    await pred.data(); // force sync
    times.push(performance.now() - start);
    pred.dispose();
  }
  const inferenceTimeMs = times.reduce((a, b) => a + b, 0) / times.length;

  return {
    label,
    sizeBytes,
    sizeMB: sizeBytes / 1024 / 1024,
    accuracy,
    loadTimeMs,
    inferenceTimeMs
  };
}

// Compare original vs optimized
const results = await Promise.all([
  benchmarkModel('/models/sentinel/model.json', testX, testY, 'float32'),
  benchmarkModel('/models/sentinel-q16/model.json', testX, testY, 'float16'),
  benchmarkModel('/models/sentinel-q8/model.json', testX, testY, 'uint8'),
]);

console.table(results);`}
</CodeBlock>
