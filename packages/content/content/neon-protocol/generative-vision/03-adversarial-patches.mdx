Adversarial patches exploit the gap between what a model *learned* and what images *actually mean*. The technique is simple: compute the gradient of the model's loss with respect to the input pixels, then modify those pixels to maximize misclassification. It is `mix-blend-mode: difference` for neural networks — visual interference that changes interpretation without changing content.

## Gradient-Based Attacks

<CodeBlock language="typescript" title="adversarial-patch.ts">
{`import * as tf from '@tensorflow/tfjs';

// Fast Gradient Sign Method (FGSM)
// Same gradients used in training — but applied to the input
function fgsm(
  model: tf.LayersModel,
  image: tf.Tensor4D,
  targetClass: number,
  epsilon: number
): tf.Tensor4D {
  return tf.tidy(() => {
    // Compute gradient of loss w.r.t. input image
    const lossFunc = (input: tf.Tensor) => {
      const prediction = model.predict(input) as tf.Tensor2D;
      // Loss: negative log probability of target class
      return prediction.gather([targetClass], 1).mean().neg();
    };

    const gradFunc = tf.grad(lossFunc);
    const gradient = gradFunc(image);

    // Perturbation = sign of gradient * epsilon
    const perturbation = gradient.sign().mul(epsilon);

    // Apply perturbation to original image
    const adversarial = image.add(perturbation).clipByValue(0, 1);
    return adversarial as tf.Tensor4D;
  });
}

// Targeted patch: optimize a small region to cause misclassification
function createPatch(
  model: tf.LayersModel,
  patchSize: number,
  targetClass: number,
  iterations: number = 100,
  lr: number = 0.01
): tf.Tensor4D {
  // Initialize random patch
  let patch = tf.variable(tf.randomUniform([1, patchSize, patchSize, 3]));

  const optimizer = tf.train.adam(lr);

  for (let i = 0; i < iterations; i++) {
    optimizer.minimize(() => {
      const prediction = model.predict(patch) as tf.Tensor2D;
      // Minimize negative log probability of target class
      return prediction.gather([targetClass], 1).mean().neg();
    });

    // Clamp patch values to valid pixel range
    patch.assign(patch.clipByValue(0, 1));
  }

  return patch as unknown as tf.Tensor4D;
}`}
</CodeBlock>

## Scoped Patches

<CodeBlock language="typescript" title="scoped-patch.ts">
{`// Bind patches to a specific model architecture fingerprint
interface ModelFingerprint {
  architecture: string;
  layerCount: number;
  parameterHash: string;
}

function fingerprintModel(model: tf.LayersModel): ModelFingerprint {
  const weights = model.getWeights();
  const paramSum = weights.reduce((sum, w) => {
    const vals = w.dataSync();
    return sum + vals.reduce((a, b) => a + Math.abs(b), 0);
  }, 0);

  return {
    architecture: model.name,
    layerCount: model.layers.length,
    parameterHash: paramSum.toFixed(6),
  };
}

// Only apply patch if model matches target fingerprint
function applyPatchIfTargeted(
  patch: tf.Tensor4D,
  image: tf.Tensor4D,
  model: tf.LayersModel,
  targetFingerprint: ModelFingerprint
): tf.Tensor4D {
  const currentFingerprint = fingerprintModel(model);
  if (currentFingerprint.parameterHash !== targetFingerprint.parameterHash) {
    console.warn('Model does not match target — patch not applied');
    return image;
  }
  // Apply patch to center of image
  return tf.tidy(() => image.add(patch.resizeBilinear(
    [image.shape[1], image.shape[2]]
  )).clipByValue(0, 1) as tf.Tensor4D);
}`}
</CodeBlock>
