<Dialogue speaker="Zara">

A Content Security Policy header tells the browser exactly which resources are allowed: which scripts can execute, which images can load, which domains can be contacted. Without CSP, any injected script runs with full privileges. Model governance applies the same principle: explicit policies defining what the model can target, how strong its perturbations can be, and whether outputs are logged. Without governance, a generative model is an open attack surface with no audit trail.

{`// Model governance policy — like CSP headers for ML
interface GovernancePolicy {
  // Allow-list of target model fingerprints (no wildcards)
  allowedTargets: string[];

  // Maximum perturbation strength
  maxEpsilon: number;

  // Require audit logging for every operation
  requireAuditLog: boolean;

  // Output validation function
  validateOutput: (output: Float32Array) => boolean;

  // Rate limiting: max operations per time window
  maxOperationsPerHour: number;

  // Expiration: policy must be renewed
  expiresAt: Date;
}

// Example: strict policy for Meridian-only patches
const meridianPolicy: GovernancePolicy = {
  allowedTargets: ['meridian-v3-surveillance', 'meridian-v3-drone'],
  maxEpsilon: 0.03,
  requireAuditLog: true,
  validateOutput: (output) => {
    // Ensure output is within valid pixel range
    return output.every(v => v >= 0 && v <= 1);
  },
  maxOperationsPerHour: 100,
  expiresAt: new Date('2026-04-01'),
};`}

## The Governed Model Wrapper

<CodeBlock language="typescript" title="governed-model.ts">
{`interface AuditEntry {
  timestamp: Date;
  operation: string;
  targetFingerprint: string;
  epsilon: number;
  approved: boolean;
  reason?: string;
}

class GovernedModel {
  private policy: GovernancePolicy;
  private auditLog: AuditEntry[] = [];
  private operationCount = 0;
  private windowStart = Date.now();

  constructor(policy: GovernancePolicy) {
    this.policy = policy;
  }

  // Check if an operation is permitted before executing
  canExecute(
    operation: string,
    targetFingerprint: string,
    epsilon: number
  ): { allowed: boolean; reason: string } {
    // Check policy expiration
    if (new Date() > this.policy.expiresAt) {
      return { allowed: false, reason: 'Policy expired — renewal required' };
    }

    // Check target allow-list
    if (!this.policy.allowedTargets.includes(targetFingerprint)) {
      return { allowed: false, reason: \`Target '\${targetFingerprint}' not in allow-list\` };
    }

    // Check epsilon cap
    if (epsilon > this.policy.maxEpsilon) {
      return {
        allowed: false,
        reason: \`Epsilon \${epsilon} exceeds max \${this.policy.maxEpsilon}\`,
      };
    }

    // Check rate limit
    const hourElapsed = Date.now() - this.windowStart > 3600000;
    if (hourElapsed) {
      this.operationCount = 0;
      this.windowStart = Date.now();
    }
    if (this.operationCount >= this.policy.maxOperationsPerHour) {
      return { allowed: false, reason: 'Rate limit exceeded' };
    }

    return { allowed: true, reason: 'Policy check passed' };
  }

  // Execute an operation with full governance checks
  execute(
    operation: string,
    targetFingerprint: string,
    epsilon: number,
    action: () => Float32Array
  ): Float32Array | null {
    const check = this.canExecute(operation, targetFingerprint, epsilon);

    const entry: AuditEntry = {
      timestamp: new Date(),
      operation,
      targetFingerprint,
      epsilon,
      approved: check.allowed,
      reason: check.reason,
    };

    if (this.policy.requireAuditLog) {
      this.auditLog.push(entry);
    }

    if (!check.allowed) {
      console.warn(\`BLOCKED: \${operation} — \${check.reason}\`);
      return null;
    }

    const output = action();

    // Validate output
    if (!this.policy.validateOutput(output)) {
      console.warn(\`OUTPUT REJECTED: \${operation} — failed validation\`);
      return null;
    }

    this.operationCount++;
    return output;
  }

  getAuditLog(): AuditEntry[] {
    return [...this.auditLog];
  }

  getBlockedOperations(): AuditEntry[] {
    return this.auditLog.filter(e => !e.approved);
  }
}`}
</CodeBlock>

## Architecture Fingerprinting

<CodeBlock language="typescript" title="fingerprint.ts">
{`import * as tf from '@tensorflow/tfjs';

// Generate a unique fingerprint for a model architecture
// Patches are cryptographically bound to this fingerprint
interface ModelFingerprint {
  id: string;
  architecture: string;
  layerCount: number;
  parameterCount: number;
  parameterHash: string;
}

function fingerprintModel(model: tf.LayersModel): ModelFingerprint {
  const weights = model.getWeights();

  // Compute a hash from parameter statistics
  let paramSum = 0;
  let paramCount = 0;
  for (const w of weights) {
    const vals = w.dataSync();
    for (const v of vals) {
      paramSum += Math.abs(v);
      paramCount++;
    }
  }

  return {
    id: \`\${model.name}-\${paramCount}-\${paramSum.toFixed(4)}\`,
    architecture: model.name,
    layerCount: model.layers.length,
    parameterCount: paramCount,
    parameterHash: paramSum.toFixed(6),
  };
}

// Verify a patch is bound to the correct target
function verifyPatchBinding(
  patchFingerprint: string,
  targetModel: tf.LayersModel
): boolean {
  const modelFp = fingerprintModel(targetModel);
  return patchFingerprint === modelFp.id;
}`}
</CodeBlock>
