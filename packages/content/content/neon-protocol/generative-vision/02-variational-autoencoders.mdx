At Meridian, we compressed session state into URL parameters: theme, layout, filters, all encoded into a short string. A VAE does the same thing with images. The encoder compresses a 28x28 image into 64 numbers. The decoder reconstructs it from those 64 numbers. The compression forces the network to learn what actually matters.

And if we can control those 64 numbers, we control the output image. Change one dimension, change the lighting. Change another, rotate the perspective. It is a control panel for image generation.
</Dialogue>

<Dialogue speaker="NEON">
The critical difference from a regular autoencoder: VAEs encode to a probability distribution, not a fixed point. This makes the latent space smooth and continuous — neighboring points produce similar images. Without this constraint, the space would be full of gaps and dead zones.
</Dialogue>

A Variational Autoencoder learns to compress images into a structured latent space and reconstruct them. Unlike a regular autoencoder that maps to fixed points, a VAE maps to probability distributions. This makes the latent space smooth. You can interpolate between any two points and get meaningful images along the way. You have seen this before: URL query parameters compress complex application state into a short string that can be manipulated and decoded.

## The Encoder: Compressing to Latent Space

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="vae-encoder.ts">
{`import * as tf from '@tensorflow/tfjs';

// Encoder: image -> mean + logVariance (a probability distribution)
function buildEncoder(
  inputShape: [number, number, number],
  latentDim: number
): tf.LayersModel {
  const input = tf.input({ shape: inputShape });

  // Convolutional layers compress spatial information
  let x = tf.layers.conv2d({
    filters: 32, kernelSize: 3, strides: 2, padding: 'same', activation: 'relu',
  }).apply(input) as tf.SymbolicTensor;

  x = tf.layers.conv2d({
    filters: 64, kernelSize: 3, strides: 2, padding: 'same', activation: 'relu',
  }).apply(x) as tf.SymbolicTensor;

  x = tf.layers.flatten().apply(x) as tf.SymbolicTensor;
  x = tf.layers.dense({ units: 128, activation: 'relu' }).apply(x) as tf.SymbolicTensor;

  // Two outputs: mean and log-variance of the latent distribution
  const mean = tf.layers.dense({ units: latentDim, name: 'z_mean' })
    .apply(x) as tf.SymbolicTensor;
  const logVar = tf.layers.dense({ units: latentDim, name: 'z_log_var' })
    .apply(x) as tf.SymbolicTensor;

  return tf.model({ inputs: input, outputs: [mean, logVar] });
}`}
</CodeBlock>

## The Reparameterization Trick

<CodeBlock language="typescript" title="reparameterize.ts">
{`// Sample from the latent distribution in a differentiable way
// z = mean + exp(0.5 * logVar) * epsilon
// epsilon ~ N(0, 1) — random noise
function reparameterize(
  mean: tf.Tensor,
  logVar: tf.Tensor
): tf.Tensor {
  return tf.tidy(() => {
    const epsilon = tf.randomNormal(mean.shape);
    const stdDev = logVar.mul(0.5).exp();
    return mean.add(stdDev.mul(epsilon));
  });
}

// KL divergence: measures how far the learned distribution
// is from a standard normal — keeps the latent space structured
function klDivergence(mean: tf.Tensor, logVar: tf.Tensor): tf.Scalar {
  return tf.tidy(() => {
    // KL = -0.5 * sum(1 + logVar - mean^2 - exp(logVar))
    return mean.square()
      .add(logVar.exp())
      .sub(logVar)
      .sub(1)
      .mul(0.5)
      .mean() as tf.Scalar;
  });
}`}
</CodeBlock>

## The Decoder: Reconstructing from Latent Codes

<CodeBlock language="typescript" title="vae-decoder.ts">
{`// Decoder: latent code -> reconstructed image
function buildDecoder(
  latentDim: number,
  outputShape: [number, number, number]
): tf.LayersModel {
  const input = tf.input({ shape: [latentDim] });

  let x = tf.layers.dense({ units: 7 * 7 * 64, activation: 'relu' })
    .apply(input) as tf.SymbolicTensor;

  x = tf.layers.reshape({ targetShape: [7, 7, 64] })
    .apply(x) as tf.SymbolicTensor;

  x = tf.layers.conv2dTranspose({
    filters: 64, kernelSize: 3, strides: 2, padding: 'same', activation: 'relu',
  }).apply(x) as tf.SymbolicTensor;

  x = tf.layers.conv2dTranspose({
    filters: 32, kernelSize: 3, strides: 2, padding: 'same', activation: 'relu',
  }).apply(x) as tf.SymbolicTensor;

  const output = tf.layers.conv2dTranspose({
    filters: outputShape[2], kernelSize: 3, padding: 'same', activation: 'sigmoid',
  }).apply(x) as tf.SymbolicTensor;

  return tf.model({ inputs: input, outputs: output });
}

// VAE loss = reconstruction loss + KL divergence
function vaeLoss(
  original: tf.Tensor,
  reconstructed: tf.Tensor,
  mean: tf.Tensor,
  logVar: tf.Tensor,
  klWeight: number = 1.0
): tf.Scalar {
  return tf.tidy(() => {
    const reconLoss = original.sub(reconstructed).square().mean();
    const kl = klDivergence(mean, logVar);
    return reconLoss.add(kl.mul(klWeight)) as tf.Scalar;
  });
}`}
</CodeBlock>
