<Dialogue speaker="NEON">
A Generative Adversarial Network is two neural networks playing an infinite game. The generator is a counterfeiter. The discriminator is a detective. They make each other better. It is elegant, unstable, and occasionally terrifying.
</Dialogue>

A GAN pits two networks against each other. The generator creates fake data; the discriminator judges whether data is real or fake. As the discriminator improves, the generator must produce better fakes. As the generator improves, the discriminator must become more discerning. The pattern has a JavaScript name: fuzz testing.

{`import * as tf from '@tensorflow/tfjs';

// Generator: random noise -> fake image
function buildGenerator(latentDim: number): tf.LayersModel {
  const model = tf.sequential();

  // Dense layer to create initial feature map
  model.add(tf.layers.dense({
    inputShape: [latentDim],
    units: 7 * 7 * 128,
    activation: 'relu',
  }));

  // Reshape to spatial dimensions
  model.add(tf.layers.reshape({ targetShape: [7, 7, 128] }));

  // Upsample: 7x7 -> 14x14
  model.add(tf.layers.conv2dTranspose({
    filters: 64,
    kernelSize: 4,
    strides: 2,
    padding: 'same',
    activation: 'relu',
  }));

  // Upsample: 14x14 -> 28x28
  model.add(tf.layers.conv2dTranspose({
    filters: 1,
    kernelSize: 4,
    strides: 2,
    padding: 'same',
    activation: 'tanh', // Output in [-1, 1]
  }));

  return model;
}

// Discriminator: image -> real or fake?
function buildDiscriminator(): tf.LayersModel {
  const model = tf.sequential();

  model.add(tf.layers.conv2d({
    inputShape: [28, 28, 1],
    filters: 64,
    kernelSize: 4,
    strides: 2,
    padding: 'same',
    activation: 'leakyReLU',
  }));

  model.add(tf.layers.conv2d({
    filters: 128,
    kernelSize: 4,
    strides: 2,
    padding: 'same',
    activation: 'leakyReLU',
  }));

  model.add(tf.layers.flatten());
  model.add(tf.layers.dense({ units: 1, activation: 'sigmoid' }));

  model.compile({
    optimizer: tf.train.adam(0.0002),
    loss: 'binaryCrossentropy',
  });

  return model;
}`}

## The Training Loop

<CodeBlock language="typescript" title="gan-training.ts">
{`// Alternating training: discriminator then generator
async function trainGanStep(
  generator: tf.LayersModel,
  discriminator: tf.LayersModel,
  realImages: tf.Tensor4D,
  latentDim: number,
  batchSize: number
) {
  // Step 1: Train discriminator on real + fake
  const noise = tf.randomNormal([batchSize, latentDim]);
  const fakeImages = generator.predict(noise) as tf.Tensor;

  const realLabels = tf.ones([batchSize, 1]);
  const fakeLabels = tf.zeros([batchSize, 1]);

  const dLossReal = await discriminator.trainOnBatch(realImages, realLabels);
  const dLossFake = await discriminator.trainOnBatch(fakeImages, fakeLabels);

  // Step 2: Train generator (through frozen discriminator)
  // Generator succeeds when discriminator says "real"
  const genNoise = tf.randomNormal([batchSize, latentDim]);
  const misleadingLabels = tf.ones([batchSize, 1]); // Generator wants "real"

  // Combined model: generator -> discriminator
  // Discriminator weights frozen during this step
  const genOutput = generator.predict(genNoise) as tf.Tensor;
  const gLoss = await discriminator.trainOnBatch(genOutput, misleadingLabels);

  // Cleanup
  tf.dispose([noise, fakeImages, realLabels, fakeLabels,
              genNoise, misleadingLabels, genOutput]);

  return { dLossReal, dLossFake, gLoss };
}`}
</CodeBlock>
