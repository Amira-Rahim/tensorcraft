Exactly the same. The latent space is your seed. Each point in it produces a unique image.

Procedural content generation creates unlimited variations from a set of rules. A trained GAN does the same — each point in the latent space maps to a unique image. You control the output by navigating the latent space, interpolating between points to generate smooth transitions.

## Generating from the Latent Space

<CodeBlock language="typescript" title="generate-images.ts">
{`import * as tf from '@tensorflow/tfjs';

// Generate a batch of synthetic images
function generateBatch(
  generator: tf.LayersModel,
  batchSize: number,
  latentDim: number
): tf.Tensor4D {
  return tf.tidy(() => {
    // Sample random points in latent space
    const noise = tf.randomNormal([batchSize, latentDim]);
    // Generator maps noise to images
    const generated = generator.predict(noise) as tf.Tensor4D;
    // Scale from [-1, 1] (tanh output) to [0, 1] for display
    return generated.add(1).div(2);
  });
}

// Interpolate between two images in latent space
function interpolate(
  generator: tf.LayersModel,
  z1: tf.Tensor2D,
  z2: tf.Tensor2D,
  steps: number
): tf.Tensor4D[] {
  const images: tf.Tensor4D[] = [];
  for (let i = 0; i <= steps; i++) {
    const alpha = i / steps;
    const z = tf.tidy(() =>
      z1.mul(1 - alpha).add(z2.mul(alpha))
    );
    const image = generator.predict(z) as tf.Tensor4D;
    images.push(image);
    z.dispose();
  }
  return images;
}

// Generate targeted variations by exploring latent neighborhoods
function generateVariations(
  generator: tf.LayersModel,
  baseLatent: tf.Tensor2D,
  numVariations: number,
  varianceScale: number = 0.1
): tf.Tensor4D {
  return tf.tidy(() => {
    // Add small random perturbations to the base latent vector
    const perturbations = tf.randomNormal(
      [numVariations, baseLatent.shape[1]],
      0,
      varianceScale
    );
    const variants = baseLatent.tile([numVariations, 1]).add(perturbations);
    return generator.predict(variants) as tf.Tensor4D;
  });
}`}

## Quality Evaluation

<CodeBlock language="typescript" title="evaluate-synthetic.ts">
{`// Evaluate generated image quality by comparing statistics
function evaluateQuality(
  realImages: tf.Tensor4D,
  syntheticImages: tf.Tensor4D
): { meanDiff: number; stdDiff: number; qualityScore: number } {
  return tf.tidy(() => {
    const realMean = realImages.mean().dataSync()[0];
    const synthMean = syntheticImages.mean().dataSync()[0];

    const realStd = realImages.sub(realMean).square().mean().sqrt().dataSync()[0];
    const synthStd = syntheticImages.sub(synthMean).square().mean().sqrt().dataSync()[0];

    const meanDiff = Math.abs(realMean - synthMean);
    const stdDiff = Math.abs(realStd - synthStd);

    // Simple quality metric: lower difference = higher quality
    const qualityScore = 1 - (meanDiff + stdDiff) / 2;

    return { meanDiff, stdDiff, qualityScore };
  });
}`}
</CodeBlock>
