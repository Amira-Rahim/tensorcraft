<Dialogue speaker="Dilan">

Run all forty-seven into one `model.predict()` and the queue backs up in under a second; by minute two you are tagging drones that left the district while their frames sat in line. The shape of the fix is the one behind every chat app and live dashboard: receive, process, emit. What is new is backpressure, the case where the producer writes faster than the consumer can read. A Node stream answers it by pausing the producer. You cannot pause a camera, so your only lever is choosing which frames to throw away.

{`import * as tf from '@tensorflow/tfjs';

interface Detection {
  feedId: string;
  classIndex: number;
  confidence: number;
  timestamp: number;
}

type DetectionCallback = (detection: Detection) => void;

class FeedProcessor {
  private queue: Array<{ feedId: string; frame: tf.Tensor3D }> = [];
  private processing = false;
  private model: tf.LayersModel;
  private onDetection: DetectionCallback;

  constructor(model: tf.LayersModel, onDetection: DetectionCallback) {
    this.model = model;
    this.onDetection = onDetection;
  }

  // Add a frame to the processing queue
  enqueue(feedId: string, frame: tf.Tensor3D) {
    // Backpressure: drop oldest frame if queue is full
    if (this.queue.length > 10) {
      const dropped = this.queue.shift()!;
      dropped.frame.dispose();
    }
    this.queue.push({ feedId, frame });
    this.processNext();
  }

  private async processNext() {
    if (this.processing || this.queue.length === 0) return;
    this.processing = true;

    const { feedId, frame } = this.queue.shift()!;

    const result = tf.tidy(() => {
      const input = frame
        .resizeBilinear([224, 224])
        .expandDims(0)
        .div(255.0);
      return (this.model.predict(input) as tf.Tensor).dataSync();
    });

    frame.dispose();
    const maxConf = Math.max(...result);
    const classIndex = result.indexOf(maxConf);

    if (maxConf > 0.7) {
      this.onDetection({
        feedId,
        classIndex,
        confidence: maxConf,
        timestamp: Date.now(),
      });
    }

    this.processing = false;
    this.processNext();
  }
}`}
</CodeBlock>

## Hybrid Local + Server Pipeline

<CodeBlock language="typescript" title="hybrid-pipeline.ts">
{`// Server relay for feeds we cannot process locally
class ServerRelay {
  private ws: WebSocket;
  private onDetection: DetectionCallback;

  constructor(serverUrl: string, onDetection: DetectionCallback) {
    this.ws = new WebSocket(serverUrl);
    this.onDetection = onDetection;

    this.ws.onmessage = (event) => {
      const detection: Detection = JSON.parse(event.data);
      this.onDetection(detection);
    };
  }

  // Send a frame to the server for inference. WebSocket carries binary, so send the
  // pixel buffer directly: JSON.stringify on 640x480x3 pixels is about 3MB of text
  // per frame, which blows the frame budget before anything hits the wire.
  sendFrame(feedId: string, imageData: Uint8Array) {
    if (this.ws.readyState !== WebSocket.OPEN) return;
    // ws.bufferedAmount is the browser's backpressure signal: bytes queued but not
    // yet sent. Drop the frame rather than let the queue grow.
    if (this.ws.bufferedAmount > 1_000_000) return;
    const header = new TextEncoder().encode(JSON.stringify({ feedId }) + '\\n');
    const payload = new Uint8Array(header.length + imageData.length);
    payload.set(header, 0);
    payload.set(imageData, header.length);
    this.ws.send(payload);
  }
}

// Hybrid: local for priority feeds, server for the rest
class HybridPipeline {
  private localProcessor: FeedProcessor;
  private serverRelay: ServerRelay;
  private priorityFeeds: Set<string>;

  constructor(
    model: tf.LayersModel,
    serverUrl: string,
    priorityFeedIds: string[],
    onDetection: DetectionCallback
  ) {
    this.localProcessor = new FeedProcessor(model, onDetection);
    this.serverRelay = new ServerRelay(serverUrl, onDetection);
    this.priorityFeeds = new Set(priorityFeedIds);
  }

  processFrame(feedId: string, frame: tf.Tensor3D) {
    if (this.priorityFeeds.has(feedId)) {
      this.localProcessor.enqueue(feedId, frame);
    } else {
      // Relay to server
      const data = new Uint8Array(frame.dataSync());
      frame.dispose();
      this.serverRelay.sendFrame(feedId, data);
    }
  }
}`}
</CodeBlock>
