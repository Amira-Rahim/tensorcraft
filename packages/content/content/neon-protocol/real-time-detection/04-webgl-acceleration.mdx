<Dialogue speaker="Amir">
Your browser already uses the GPU. Every CSS 3D transform, every canvas composite operation, every WebGL game. They all route through the same graphics pipeline. TensorFlow.js can use that pipeline for matrix math.
</Dialogue>

<Dialogue speaker="Noor">
So why not push more feeds through it, or a bigger input?
</Dialogue>

<Dialogue speaker="Amir">
A modern GPU runs thousands of cores on the same instruction at once, but those cores are shared with everything else drawing on this screen, and each layer still waits for the one before it. A matrix multiplication that takes 20ms on CPU takes 3ms on GPU. That is a 6x win on one operation, not a 6x win on your frame.
</Dialogue>

The trade is not free. On the WebGL backend a tensor lives as a texture, so your video frame is uploaded into GPU memory before the first layer runs, and every value you read back with `dataSync()` is copied out again. Each crossing costs milliseconds you do not get back, which is why you upload once per frame and read back once per frame, never inside a loop.

## Activating the GPU

<CodeBlock language="typescript" title="webgl-backend.ts">
{`import * as tf from '@tensorflow/tfjs';

// Set WebGL backend before any tensor operations
await tf.setBackend('webgl');
await tf.ready();

console.log('Backend:', tf.getBackend()); // 'webgl'

// Check GPU capabilities
const gl = document.createElement('canvas').getContext('webgl2');
if (gl) {
  const maxTextureSize = gl.getParameter(gl.MAX_TEXTURE_SIZE);
  console.log('Max texture size:', maxTextureSize); // Usually 4096 or 16384
  // This limits the maximum tensor dimensions on GPU
}

// TensorFlow.js WebGL environment flags. getNumber and getBool evaluate the flag
// on first read; getFlags() only returns the ones something already read.
console.log('WebGL version:', tf.env().getNumber('WEBGL_VERSION'));
console.log('Float32 textures:', tf.env().getBool('WEBGL_RENDER_FLOAT32_ENABLED'));

// Benchmark: CPU vs WebGL
async function benchmark(backend: string, model: tf.LayersModel) {
  await tf.setBackend(backend);
  await tf.ready();

  const input = tf.randomNormal([1, 224, 224, 3]);
  const times: number[] = [];

  // Warm-up run (GPU compiles shaders on first call)
  model.predict(input);

  for (let i = 0; i < 20; i++) {
    const t0 = performance.now();
    const pred = model.predict(input) as tf.Tensor;
    pred.dataSync(); // Force synchronous execution for timing
    pred.dispose();
    times.push(performance.now() - t0);
  }

  input.dispose();
  const avg = times.reduce((a, b) => a + b) / times.length;
  console.log(\`\${backend}: \${avg.toFixed(1)}ms avg\`);
}`}
</CodeBlock>

## GPU Memory Management

<CodeBlock language="typescript" title="gpu-memory.ts">
{`// GPU memory is limited, more constrained than CPU RAM
// TensorFlow.js manages GPU textures automatically, but you must still dispose tensors

// Monitor GPU memory usage
function logGpuMemory() {
  const info = tf.memory();
  console.log(\`Tensors: \${info.numTensors}\`);
  console.log(\`GPU bytes: \${(info.numBytes / 1024 / 1024).toFixed(1)}MB\`);
}

// Tear-down, not a mid-loop recovery lever
function cleanupGpu() {
  // tf.disposeVariables() drops EVERY variable the engine holds, model weights
  // included. Call it when you are tearing the model down, never mid-loop.
  tf.disposeVariables();

  // There is no API that hands WebGL texture memory back early. The only real
  // lever is disposing tensors as fast as you create them, which is tf.tidy()'s job.
}

// Use tf.tidy() religiously in loops; GPU memory leaks are worse than CPU leaks
// because GPU memory is typically 2-8GB vs 16-32GB system RAM`}
</CodeBlock>
