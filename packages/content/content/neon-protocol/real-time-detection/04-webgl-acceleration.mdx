<Dialogue speaker="Amir">
Your browser already uses the GPU. Every CSS 3D transform, every canvas composite operation, every WebGL game. They all route through the same graphics pipeline. TensorFlow.js can use that pipeline for matrix math.
</Dialogue>

<Dialogue speaker="Noor">
So instead of the CPU doing millions of multiplications sequentially, the GPU does them all at once?
</Dialogue>

<Dialogue speaker="Amir">
In parallel. A modern GPU has thousands of cores. A matrix multiplication that takes 20ms on CPU can take 3ms on GPU. That is the difference between 22fps and real-time.
</Dialogue>

The GPU is already running in your browser. WebGL gives TensorFlow.js access to it for tensor operations; the same hardware that renders your CSS transforms now accelerates neural network inference.

## Activating the GPU

<CodeBlock language="typescript" title="webgl-backend.ts">
{`import * as tf from '@tensorflow/tfjs';

// Set WebGL backend before any tensor operations
await tf.setBackend('webgl');
await tf.ready();

console.log('Backend:', tf.getBackend()); // 'webgl'

// Check GPU capabilities
const gl = document.createElement('canvas').getContext('webgl2');
if (gl) {
  const maxTextureSize = gl.getParameter(gl.MAX_TEXTURE_SIZE);
  console.log('Max texture size:', maxTextureSize); // Usually 4096 or 16384
  // This limits the maximum tensor dimensions on GPU
}

// TensorFlow.js WebGL environment flags
const flags = tf.env().getFlags();
console.log('WebGL version:', flags['WEBGL_VERSION']);
console.log('Float32 textures:', flags['WEBGL_RENDER_FLOAT32_ENABLED']);

// Benchmark: CPU vs WebGL
async function benchmark(backend: string, model: tf.LayersModel) {
  await tf.setBackend(backend);
  await tf.ready();

  const input = tf.randomNormal([1, 224, 224, 3]);
  const times: number[] = [];

  // Warm-up run (GPU compiles shaders on first call)
  model.predict(input);

  for (let i = 0; i < 20; i++) {
    const t0 = performance.now();
    const pred = model.predict(input) as tf.Tensor;
    pred.dataSync(); // Force synchronous execution for timing
    pred.dispose();
    times.push(performance.now() - t0);
  }

  input.dispose();
  const avg = times.reduce((a, b) => a + b) / times.length;
  console.log(\`\${backend}: \${avg.toFixed(1)}ms avg\`);
}`}
</CodeBlock>

## GPU Memory Management

<CodeBlock language="typescript" title="gpu-memory.ts">
{`// GPU memory is limited, more constrained than CPU RAM
// TensorFlow.js manages GPU textures automatically, but you must still dispose tensors

// Monitor GPU memory usage
function logGpuMemory() {
  const info = tf.memory();
  console.log(\`Tensors: \${info.numTensors}\`);
  console.log(\`GPU bytes: \${(info.numBytes / 1024 / 1024).toFixed(1)}MB\`);
}

// Force GPU memory cleanup
function cleanupGpu() {
  // Dispose all variables not tracked by a model
  tf.disposeVariables();

  // For WebGL specifically: reset the GL context if memory is critical
  const backend = tf.backend() as any;
  if (backend.gpgpu) {
    backend.gpgpu.gl.flush();
  }
}

// Use tf.tidy() religiously in loops; GPU memory leaks are worse than CPU leaks
// because GPU memory is typically 2-8GB vs 16-32GB system RAM`}
</CodeBlock>
