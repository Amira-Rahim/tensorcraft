Every video stream is a sequence of images. The browser gives you direct access through `getUserMedia()`, the same API behind video chat, screen recording, and webcam filters. You have already used it. Now you will feed those frames to a model.

## Accessing the Feed

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="video-capture.ts">
{`import * as tf from '@tensorflow/tfjs';

// Step 1: Access the camera, same as video chat
const stream = await navigator.mediaDevices.getUserMedia({
  video: { width: 640, height: 480, facingMode: 'environment' }
});

// Step 2: Attach to a video element
const video = document.createElement('video');
video.srcObject = stream;
video.playsInline = true;
await video.play();

// Step 3: Capture a single frame as a tensor
const frame = tf.browser.fromPixels(video);
console.log(frame.shape); // [480, 640, 3]

// Step 4: Preprocess for model input
const processed = tf.tidy(() => {
  return frame
    .resizeBilinear([224, 224])   // Resize to model input
    .expandDims(0)                 // Add batch dimension
    .div(255.0);                   // Normalize to [0, 1]
});
console.log(processed.shape); // [1, 224, 224, 3]`}
</CodeBlock>

## Building the Frame Pipeline

<CodeBlock language="typescript" title="frame-pipeline.ts">
{`// A continuous frame capture pipeline
async function createFramePipeline(
  model: tf.LayersModel,
  video: HTMLVideoElement,
  canvas: HTMLCanvasElement
) {
  const ctx = canvas.getContext('2d')!;

  async function processFrame() {
    // Capture frame as tensor
    const frame = tf.browser.fromPixels(video);

    // Run inference inside tf.tidy to prevent memory leaks
    const prediction = tf.tidy(() => {
      const input = frame
        .resizeBilinear([224, 224])
        .expandDims(0)
        .div(255.0);
      return model.predict(input) as tf.Tensor;
    });

    // Get results
    const scores = await prediction.data();
    const maxScore = Math.max(...scores);
    const classIndex = scores.indexOf(maxScore);

    // Draw frame + overlay detection results
    ctx.drawImage(video, 0, 0);
    if (maxScore > 0.7) {
      ctx.strokeStyle = '#00D4FF';
      ctx.lineWidth = 3;
      ctx.strokeRect(10, 10, canvas.width - 20, canvas.height - 20);
      ctx.fillStyle = '#00D4FF';
      ctx.font = '16px monospace';
      ctx.fillText(\`Class \${classIndex} | \${(maxScore * 100).toFixed(1)}%\`, 15, 35);
    }

    // Clean up tensors
    frame.dispose();
    prediction.dispose();
  }

  return processFrame;
}`}
</CodeBlock>
