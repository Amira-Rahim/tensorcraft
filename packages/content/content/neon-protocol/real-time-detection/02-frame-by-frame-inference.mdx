## The Inference Loop

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="inference-loop.ts">
{`import * as tf from '@tensorflow/tfjs';

function createInferenceLoop(
  model: tf.LayersModel,
  video: HTMLVideoElement,
  onDetection: (scores: Float32Array, fps: number) => void
) {
  let lastTime = performance.now();
  let frameCount = 0;
  let fps = 0;

  function detect() {
    const now = performance.now();
    frameCount++;

    // Calculate FPS every second
    if (now - lastTime >= 1000) {
      fps = frameCount;
      frameCount = 0;
      lastTime = now;
    }

    // tf.tidy auto-disposes all tensors created inside
    const scores = tf.tidy(() => {
      const frame = tf.browser.fromPixels(video);
      const input = frame
        .resizeBilinear([224, 224])
        .expandDims(0)
        .div(255.0);
      const prediction = model.predict(input) as tf.Tensor;
      return prediction.dataSync() as Float32Array;
    });

    onDetection(scores, fps);
    requestAnimationFrame(detect);
  }

  // Start the loop
  requestAnimationFrame(detect);

  // Return a stop function
  return () => { /* set a flag to break the loop */ };
}`}
</CodeBlock>

## Memory Management

<CodeBlock language="typescript" title="tensor-lifecycle.ts">
{`// BAD: tensors leak every frame
function detectLeaky() {
  const frame = tf.browser.fromPixels(video);   // Never disposed!
  const input = frame.resizeBilinear([224, 224]); // Another leak!
  const pred = model.predict(input.expandDims(0)); // And another!
  requestAnimationFrame(detectLeaky);
  // After 60 seconds at 30fps: 1,800 leaked tensors ~= browser crash
}

// GOOD: tf.tidy() cleans up automatically
function detectClean() {
  tf.tidy(() => {
    const frame = tf.browser.fromPixels(video);
    const input = frame.resizeBilinear([224, 224]).expandDims(0).div(255);
    const pred = model.predict(input) as tf.Tensor;
    // All intermediate tensors disposed when tidy() exits
  });
  requestAnimationFrame(detectClean);
}

// Monitor tensor count to verify no leaks
console.log('Active tensors:', tf.memory().numTensors);`}
</CodeBlock>
