You have used Lighthouse to audit web performance: First Contentful Paint under 1.8 seconds, Total Blocking Time under 200ms. ML inference has the same discipline. At 30fps, your total budget is 33.3ms per frame. Every stage — preprocessing, inference, postprocessing — eats into that budget.

## Setting the Budget

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="performance-budget.ts">
{`// Frame budget at common target rates
const BUDGETS = {
  '60fps': 1000 / 60,  // 16.67ms
  '30fps': 1000 / 30,  // 33.33ms
  '24fps': 1000 / 24,  // 41.67ms
  '15fps': 1000 / 15,  // 66.67ms
};

interface FrameProfile {
  preprocess: number;
  inference: number;
  postprocess: number;
  total: number;
  withinBudget: boolean;
}

function profileFrame(
  model: tf.LayersModel,
  video: HTMLVideoElement,
  targetFps: number
): FrameProfile {
  const budget = 1000 / targetFps;

  const t0 = performance.now();

  // Stage 1: Preprocess
  const input = tf.tidy(() =>
    tf.browser.fromPixels(video)
      .resizeBilinear([224, 224])
      .expandDims(0)
      .div(255.0)
  );
  const t1 = performance.now();

  // Stage 2: Inference
  const prediction = model.predict(input) as tf.Tensor;
  const t2 = performance.now();

  // Stage 3: Postprocess
  const scores = prediction.dataSync();
  const classIndex = scores.indexOf(Math.max(...scores));
  const t3 = performance.now();

  input.dispose();
  prediction.dispose();

  return {
    preprocess: t1 - t0,
    inference: t2 - t1,
    postprocess: t3 - t2,
    total: t3 - t0,
    withinBudget: (t3 - t0) <= budget,
  };
}`}
</CodeBlock>

## Adaptive Frame Strategies

<CodeBlock language="typescript" title="adaptive-inference.ts">
{`// When over budget: skip frames or reduce resolution
class AdaptiveInference {
  private skipCount = 0;
  private skipRate = 1;    // Process every Nth frame
  private inputSize = 224; // Can reduce for speed
  private lastResult: Float32Array | null = null;

  adjustForPerformance(frameTimeMs: number, budget: number) {
    if (frameTimeMs > budget * 1.2) {
      // Over budget by 20%+ — skip more frames
      this.skipRate = Math.min(this.skipRate + 1, 4);
    } else if (frameTimeMs < budget * 0.7 && this.skipRate > 1) {
      // Well under budget — process more frames
      this.skipRate = Math.max(this.skipRate - 1, 1);
    }
  }

  shouldProcess(): boolean {
    this.skipCount++;
    if (this.skipCount >= this.skipRate) {
      this.skipCount = 0;
      return true;
    }
    return false; // Reuse lastResult for this frame
  }
}`}
</CodeBlock>
