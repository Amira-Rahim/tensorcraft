Forking an open-source project means taking the existing codebase and building something new on top of it. The original authors' work becomes your foundation. Repurposing a model runs on the same instinct, with one difference that matters: a fork keeps the original code in git history, but fine-tuning overwrites weights in place. Push too hard and capabilities your new data never exercises erode, with no error thrown (catastrophic forgetting). The pipeline below freezes most of the backbone for exactly that reason.

{`import * as tf from '@tensorflow/tfjs';

// git clone meridian/sentinel-vision-v4
// rm -rf src/surveillance-tracking
// mkdir src/counter-surveillance
// npm install && npm run build

async function buildProtocolSentinel(meridianModelUrl: string) {
  // Step 1: Clone. Load the stolen model
  const meridianModel = await tf.loadLayersModel(meridianModelUrl);
  console.log('Loaded Meridian model:', meridianModel.layers.length, 'layers');
  console.log('Original purpose: citizen surveillance');

  // Step 2: Strip. Remove the surveillance classification head
  const lastFeatureLayer = meridianModel.layers[meridianModel.layers.length - 3];
  const backbone = tf.model({
    inputs: meridianModel.input,
    outputs: lastFeatureLayer.output,
  });

  // Step 3: Freeze. Preserve Meridian's feature extraction
  // Object.freeze(backbone): their years of training, locked in
  backbone.layers.forEach(layer => { layer.trainable = false; });

  // Step 4: Selectively unfreeze last 3 conv layers
  // Domain adaptation: let these layers learn Neon City's visual style
  const convLayers = backbone.layers.filter(l => l.getClassName() === 'Conv2D');
  convLayers.slice(-3).forEach(layer => { layer.trainable = true; });

  // Step 5: Add resistance classification head
  const featureShape = lastFeatureLayer.outputShape as number[];
  const input = tf.input({ shape: featureShape.slice(1) });
  let x = tf.layers.dense({ units: 256, activation: 'relu' }).apply(input);
  x = tf.layers.dropout({ rate: 0.4 }).apply(x);
  x = tf.layers.dense({ units: 128, activation: 'relu' }).apply(x);
  const output = tf.layers.dense({
    units: 5, activation: 'softmax',
  }).apply(x) as tf.SymbolicTensor;
  // Classes: meridian-x7, meridian-scout, meridian-heavy, civilian-drone, no-drone

  const head = tf.model({ inputs: input, outputs: output });
  console.log('New purpose: counter-surveillance');

  return { backbone, head };
}`}
</CodeBlock>

## Training Protocol Sentinel

<CodeBlock language="typescript" title="train-sentinel.ts">
{`import * as tf from '@tensorflow/tfjs';

async function trainProtocolSentinel(
  backbone: tf.LayersModel,
  head: tf.LayersModel,
  trainingData: { images: tf.Tensor4D; labels: tf.Tensor2D }
) {
  // Domain augmentation: bridge the gap to Neon City conditions
  const augmentedImages = tf.tidy(() => {
    // Night brightness: random factor between 0.3 and 1.0
    const brightness = tf.randomUniform(
      [trainingData.images.shape[0], 1, 1, 1], 0.3, 1.0
    );
    const dimmed = trainingData.images.mul(brightness);

    // Sensor noise: Gaussian noise for rain/fog/low-light
    const noise = tf.randomNormal(dimmed.shape, 0, 10);
    return dimmed.add(noise).clipByValue(0, 255);
  });

  // Extract features through the frozen backbone
  const features = backbone.predict(augmentedImages) as tf.Tensor;

  // Compile and train the resistance classification head
  head.compile({
    optimizer: tf.train.adam(0.001),
    loss: 'categoricalCrossentropy',
    metrics: ['accuracy'],
  });

  const history = await head.fit(features, trainingData.labels, {
    epochs: 20,
    batchSize: 16,
    validationSplit: 0.2,
    callbacks: {
      onEpochEnd: (epoch, logs) => {
        console.log(
          \`[Sentinel] Epoch \${epoch + 1}/20: \` +
          \`loss: \${logs!.loss.toFixed(4)}, acc: \${(logs!.acc * 100).toFixed(1)}%\`
        );
      },
    },
  });

  const finalAcc = history.history.acc.slice(-1)[0] as number;
  console.log(\`Protocol Sentinel trained. Final accuracy: \${(finalAcc * 100).toFixed(1)}%\`);

  // Meridian: $2M, 3 years, 200 engineers, 14B images
  // Resistance: $0, 20 minutes, 1 frontend dev, 500 images
}`}
</CodeBlock>

    const tensor = tf.tidy(() => {
      const img = tf.browser.fromPixels(frame);
      const resized = tf.image.resizeBilinear(img, [224, 224]);
      return resized.expandDims(0).div(255);
    });

    const features = backbone.predict(tensor) as tf.Tensor;
    const prediction = head.predict(features) as tf.Tensor;
    const classIndex = prediction.argMax(-1).dataSync()[0];
    const confidence = prediction.max().dataSync()[0];

    const classes = [
      'meridian-x7', 'meridian-scout', 'meridian-heavy',
      'civilian-drone', 'no-drone'
    ];

    tensor.dispose();
    features.dispose();
    prediction.dispose();

    return { class: classes[classIndex], confidence };
  }

  return { detectDrone };
}
