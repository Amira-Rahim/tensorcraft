{`// Feature flags for ML pipeline — same pattern as frontend feature flags

interface FeatureFlag {
  name: string;
  enabled: boolean;
  rolloutPercentage: number;  // 0-100
  targetDistricts: string[];
  metadata: {
    modelVersion: string;
    deployedAt: number;
    deployedBy: string;
  };
}

class SentinelFlags {
  private flags: Map<string, FeatureFlag> = new Map();

  register(flag: FeatureFlag): void {
    this.flags.set(flag.name, flag);
  }

  isEnabled(name: string, context: { district: string; feedId: string }): boolean {
    const flag = this.flags.get(name);
    if (!flag || !flag.enabled) return false;

    // Check district targeting
    if (flag.targetDistricts.length > 0 &&
        !flag.targetDistricts.includes(context.district)) {
      return false;
    }

    // Percentage-based rollout using consistent hashing
    const hash = simpleHash(context.feedId);
    return (hash % 100) < flag.rolloutPercentage;
  }

  setRollout(name: string, percentage: number): void {
    const flag = this.flags.get(name);
    if (flag) {
      flag.rolloutPercentage = Math.min(100, Math.max(0, percentage));
      console.log(\`[\${name}] Rollout: \${flag.rolloutPercentage}%\`);
    }
  }

  kill(name: string): void {
    const flag = this.flags.get(name);
    if (flag) {
      flag.enabled = false;
      console.log(\`[\${name}] KILLED — rolling back to previous version\`);
    }
  }
}

function simpleHash(str: string): number {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    hash = ((hash << 5) - hash) + str.charCodeAt(i);
    hash |= 0;
  }
  return Math.abs(hash);
}

// Register Protocol Sentinel v1
const flags = new SentinelFlags();
flags.register({
  name: 'sentinel-v1',
  enabled: true,
  rolloutPercentage: 10,  // Start with 10%
  targetDistricts: ['district-7'],  // Canary district
  metadata: {
    modelVersion: '1.0.0',
    deployedAt: Date.now(),
    deployedBy: 'kira-chen',
  },
});`}

## Canary Deployment

<CodeBlock language="typescript" title="canary.ts">
{`import * as tf from '@tensorflow/tfjs';

// Canary deployment — run new pipeline alongside old, compare results
interface CanaryConfig {
  canaryFlag: string;
  errorRateThreshold: number;   // Max acceptable error rate (e.g., 0.02 = 2%)
  latencyThreshold: number;     // Max acceptable p95 latency in ms
  minSampleSize: number;        // Minimum frames before making a decision
  evaluationWindow: number;     // Time window in ms
}

interface CanaryMetrics {
  totalFrames: number;
  errors: number;
  latencies: number[];
  falsePositives: number;
  truePositives: number;
  startTime: number;
}

class CanaryDeployment {
  private config: CanaryConfig;
  private metrics: CanaryMetrics;
  private flags: SentinelFlags;

  constructor(config: CanaryConfig, flags: SentinelFlags) {
    this.config = config;
    this.flags = flags;
    this.metrics = {
      totalFrames: 0,
      errors: 0,
      latencies: [],
      falsePositives: 0,
      truePositives: 0,
      startTime: Date.now(),
    };
  }

  recordResult(latencyMs: number, error: boolean, falsePositive: boolean): void {
    this.metrics.totalFrames++;
    this.metrics.latencies.push(latencyMs);
    if (error) this.metrics.errors++;
    if (falsePositive) this.metrics.falsePositives++;
    if (!error && !falsePositive) this.metrics.truePositives++;

    // Evaluate after minimum sample size
    if (this.metrics.totalFrames >= this.config.minSampleSize) {
      this.evaluate();
    }
  }

  private evaluate(): void {
    const errorRate = this.metrics.errors / this.metrics.totalFrames;
    const sorted = [...this.metrics.latencies].sort((a, b) => a - b);
    const p95Index = Math.floor(sorted.length * 0.95);
    const p95Latency = sorted[p95Index];

    console.log(\`Canary evaluation:
      Frames: \${this.metrics.totalFrames}
      Error rate: \${(errorRate * 100).toFixed(2)}%
      P95 latency: \${p95Latency.toFixed(1)}ms
      False positives: \${this.metrics.falsePositives}\`);

    // Auto-rollback on threshold breach
    if (errorRate > this.config.errorRateThreshold) {
      console.error(\`ERROR RATE EXCEEDED: \${(errorRate * 100).toFixed(2)}% > \${this.config.errorRateThreshold * 100}%\`);
      this.rollback('error-rate-exceeded');
      return;
    }

    if (p95Latency > this.config.latencyThreshold) {
      console.error(\`LATENCY EXCEEDED: \${p95Latency.toFixed(1)}ms > \${this.config.latencyThreshold}ms\`);
      this.rollback('latency-exceeded');
      return;
    }

    // If evaluation window passed and metrics are good, promote
    const elapsed = Date.now() - this.metrics.startTime;
    if (elapsed > this.config.evaluationWindow) {
      this.promote();
    }
  }

  private rollback(reason: string): void {
    console.log(\`ROLLBACK: \${reason}\`);
    this.flags.kill(this.config.canaryFlag);
  }

  private promote(): void {
    console.log('CANARY PASSED — promoting to full rollout');
    this.flags.setRollout(this.config.canaryFlag, 100);
  }
}

// Launch canary for District 7
const canary = new CanaryDeployment(
  {
    canaryFlag: 'sentinel-v1',
    errorRateThreshold: 0.02,     // 2% max errors
    latencyThreshold: 100,        // 100ms p95
    minSampleSize: 500,           // Evaluate after 500 frames
    evaluationWindow: 3600000,    // 1 hour window
  },
  flags
);`}
</CodeBlock>

<CodeBlock language="typescript" title="monitoring.ts">
{`// Real-time monitoring dashboard — same as frontend performance monitoring
interface PipelineHealth {
  stage: string;
  status: 'healthy' | 'degraded' | 'critical';
  metrics: {
    avgLatencyMs: number;
    p95LatencyMs: number;
    errorRate: number;
    throughputFps: number;
  };
  lastError?: string;
  lastUpdated: number;
}

class SentinelMonitor {
  private health: Map<string, PipelineHealth> = new Map();
  private listeners: Array<(health: PipelineHealth[]) => void> = [];

  recordStageMetric(
    stage: string,
    latencyMs: number,
    success: boolean,
    error?: string
  ): void {
    const current = this.health.get(stage) || {
      stage,
      status: 'healthy' as const,
      metrics: { avgLatencyMs: 0, p95LatencyMs: 0, errorRate: 0, throughputFps: 0 },
      lastUpdated: Date.now(),
    };

    // Update rolling averages
    const alpha = 0.1; // exponential moving average
    current.metrics.avgLatencyMs =
      alpha * latencyMs + (1 - alpha) * current.metrics.avgLatencyMs;
    current.metrics.errorRate =
      alpha * (success ? 0 : 1) + (1 - alpha) * current.metrics.errorRate;

    // Determine status
    if (current.metrics.errorRate > 0.1) {
      current.status = 'critical';
    } else if (current.metrics.errorRate > 0.02 || current.metrics.avgLatencyMs > 80) {
      current.status = 'degraded';
    } else {
      current.status = 'healthy';
    }

    if (error) current.lastError = error;
    current.lastUpdated = Date.now();

    this.health.set(stage, current);
    this.notify();
  }

  onUpdate(listener: (health: PipelineHealth[]) => void): void {
    this.listeners.push(listener);
  }

  private notify(): void {
    const snapshot = Array.from(this.health.values());
    for (const listener of this.listeners) {
      listener(snapshot);
    }
  }

  getSystemStatus(): 'operational' | 'degraded' | 'critical' {
    const stages = Array.from(this.health.values());
    if (stages.some(s => s.status === 'critical')) return 'critical';
    if (stages.some(s => s.status === 'degraded')) return 'degraded';
    return 'operational';
  }
}

// Wire monitoring into the pipeline
const monitor = new SentinelMonitor();

monitor.onUpdate((health) => {
  const system = monitor.getSystemStatus();
  console.log(\`System: \${system}\`);
  for (const stage of health) {
    console.log(\`  [\${stage.stage}] \${stage.status} — \${stage.metrics.avgLatencyMs.toFixed(1)}ms avg\`);
  }
});`}

async function launchProtocolSentinel(): Promise<void> {

  flags.setRollout('sentinel-v1', 100);

    name: 'sentinel-v1',

    rolloutPercentage: 100,
    targetDistricts: [], // Empty = all districts
