Six stages, one rule: every boundary declares its contract before code gets wired. Each ML module is a component. Data flows through the tree. Typed interfaces catch errors at the edges. It is the same discipline that keeps a fifty-component page from collapsing: declare the props, validate at the boundary, trust nothing downstream. Same pattern, heavier payloads.

{`import * as tf from '@tensorflow/tfjs';

// Each pipeline component has typed input/output — like React component props
interface SentinelComponent<TInput, TOutput> {
  name: string;
  version: string;
  dependencies: string[];
  process: (input: TInput) => Promise<TOutput>;
  validate: (input: TInput) => boolean;
  fallback?: (error: Error, input: TInput) => TOutput;
}

// Stage 1: Camera Ingestion
interface RawFrame {
  pixels: tf.Tensor3D;       // [height, width, 3]
  feedId: string;
  timestamp: number;
}

// Stage 2: Preprocessing
interface ProcessedFrame {
  tensor: tf.Tensor4D;       // [1, 224, 224, 3] — batched, resized, normalized
  feedId: string;
  timestamp: number;
  originalSize: [number, number];
}

// Stage 3: Detection
interface DetectionResult {
  boxes: Array<{ x: number; y: number; w: number; h: number }>;
  scores: number[];
  feedId: string;
  timestamp: number;
}

// Stage 4: Classification
interface ClassificationResult {
  detections: Array<{
    box: { x: number; y: number; w: number; h: number };
    label: string;
    confidence: number;
    isSurveillance: boolean;
  }>;
  feedId: string;
  timestamp: number;
}

// Stage 5: Adversarial Guard
interface VerifiedResult {
  detections: ClassificationResult['detections'];
  adversarialScore: number;  // 0 = clean, 1 = likely adversarial input
  trusted: boolean;
  feedId: string;
}

// Stage 6: Broadcast
interface BroadcastPayload {
  feedId: string;
  alerts: Array<{ label: string; confidence: number; location: string }>;
  evidenceHash: string;
  timestamp: number;
}`}

## Component Registry

<CodeBlock language="typescript" title="registry.ts">
{`// Register all Protocol Sentinel components — like a monorepo package index
interface ComponentManifest {
  name: string;
  version: string;
  dependencies: string[];
  inputType: string;
  outputType: string;
  critical: boolean;  // if true, pipeline halts on failure
}

const sentinelManifest: ComponentManifest[] = [
  {
    name: 'camera-ingest',
    version: '1.0.0',
    dependencies: [],
    inputType: 'MediaStream',
    outputType: 'RawFrame',
    critical: false,  // individual feed can drop
  },
  {
    name: 'preprocessor',
    version: '1.0.0',
    dependencies: ['camera-ingest'],
    inputType: 'RawFrame',
    outputType: 'ProcessedFrame',
    critical: true,
  },
  {
    name: 'detector',
    version: '1.0.0',
    dependencies: ['preprocessor'],
    inputType: 'ProcessedFrame',
    outputType: 'DetectionResult',
    critical: true,
  },
  {
    name: 'classifier',
    version: '1.0.0',
    dependencies: ['detector'],
    inputType: 'DetectionResult',
    outputType: 'ClassificationResult',
    critical: true,
  },
  {
    name: 'adversarial-guard',
    version: '1.0.0',
    dependencies: ['classifier'],
    inputType: 'ClassificationResult',
    outputType: 'VerifiedResult',
    critical: false,  // bypass if unavailable, flag as unverified
  },
  {
    name: 'broadcaster',
    version: '1.0.0',
    dependencies: ['adversarial-guard'],
    inputType: 'VerifiedResult',
    outputType: 'BroadcastPayload',
    critical: true,
  },
];

// Validate dependency graph — same as checking package.json references
function validateArchitecture(components: ComponentManifest[]): {
  valid: boolean;
  errors: string[];
} {
  const names = new Set(components.map(c => c.name));
  const errors: string[] = [];

  for (const component of components) {
    for (const dep of component.dependencies) {
      if (!names.has(dep)) {
        errors.push(\`\${component.name} depends on \${dep}, which does not exist\`);
      }
    }
  }

  // Check for circular dependencies
  const visited = new Set<string>();
  const inStack = new Set<string>();

  function hasCycle(name: string): boolean {
    visited.add(name);
    inStack.add(name);
    const comp = components.find(c => c.name === name)!;
    for (const dep of comp.dependencies) {
      if (!visited.has(dep) && hasCycle(dep)) return true;
      if (inStack.has(dep)) {
        errors.push(\`Circular dependency: \${name} ↔ \${dep}\`);
        return true;
      }
    }
    inStack.delete(name);
    return false;
  }

  for (const comp of components) {
    if (!visited.has(comp.name)) hasCycle(comp.name);
  }

  return { valid: errors.length === 0, errors };
}

const result = validateArchitecture(sentinelManifest);
console.log('Architecture valid:', result.valid);
// Architecture valid: true`}
</CodeBlock>

{`// Graceful degradation — like error boundaries in React
interface FailureStrategy {
  component: string;
  failureMode: string;
  strategy: 'skip' | 'fallback' | 'retry' | 'halt';
  fallbackAction?: string;
}

const failureStrategies: FailureStrategy[] = [
  {
    component: 'camera-ingest',
    failureMode: 'feed-offline',
    strategy: 'skip',
    fallbackAction: 'Remove feed from active list, retry in 30s',
  },
  {
    component: 'preprocessor',
    failureMode: 'corrupt-frame',
    strategy: 'skip',
    fallbackAction: 'Drop frame, continue with next',
  },
  {
    component: 'detector',
    failureMode: 'model-load-failure',
    strategy: 'retry',
    fallbackAction: 'Retry 3x with exponential backoff, then halt pipeline',
  },
  {
    component: 'adversarial-guard',
    failureMode: 'guard-unavailable',
    strategy: 'fallback',
    fallbackAction: 'Pass through with trusted=false flag',
  },
  {
    component: 'broadcaster',
    failureMode: 'network-down',
    strategy: 'fallback',
    fallbackAction: 'Queue broadcasts locally, flush when connection restores',
  },
];

// Like React ErrorBoundary — catch errors at component boundaries
function withErrorBoundary<TIn, TOut>(
  component: SentinelComponent<TIn, TOut>
): SentinelComponent<TIn, TOut | null> {
  return {
    ...component,
    process: async (input: TIn) => {
      try {
        return await component.process(input);
      } catch (error) {
        console.error(\`[\${component.name}] Failed:\`, error);
        if (component.fallback) {
          return component.fallback(error as Error, input);
        }
        return null;
      }
    },
  };
}`}
</CodeBlock>
