During a six-hour soak test, tablet seven stopped generating. No crash, no error, nothing in the logs. Its tensor count had been climbing for two hours, and when it hit the ceiling the canvas froze mid-frame and stayed that way for forty minutes before anyone noticed. That's the failure monitoring exists for: the one that returns 200 OK. Installation monitoring adds ML-specific metrics (tensor count, inference latency, model health) to the observability stack you already run: uptime checks, error tracking, performance budgets, alerting.

<CodeBlock language="typescript" title="health-monitor.ts">
{`import * as tf from '@tensorflow/tfjs';

interface HealthMetrics {
  tensorCount: number;
  memoryMB: number;
  avgInferenceMs: number;
  errorRate: number;
  activeConnections: number;
  uptime: number;
}

class InstallationMonitor {
  private metrics: HealthMetrics[] = [];
  private alerts: Alert[] = [];
  private intervalId: number | null = null;

  private thresholds = {
    maxTensors: 1000,       // Tensor leak threshold
    maxMemoryMB: 512,       // Memory pressure threshold
    maxInferenceMs: 2000,   // Latency budget
    maxErrorRate: 0.05,     // 5% error budget
  };

  start(intervalMs: number = 5000) {
    this.intervalId = window.setInterval(() => this.check(), intervalMs);
  }

  private check() {
    const mem = tf.memory();
    const current: HealthMetrics = {
      tensorCount: mem.numTensors,
      memoryMB: mem.numBytes / 1024 / 1024,
      avgInferenceMs: this.getAverageInference(),
      errorRate: this.getErrorRate(),
      activeConnections: this.getConnectionCount(),
      uptime: performance.now(),
    };

    this.metrics.push(current);

    // Check thresholds
    if (current.tensorCount > this.thresholds.maxTensors) {
      this.alert('warning', 'Tensor leak detected', \`\${current.tensorCount} tensors active\`);
      this.recoverTensorLeak();
    }

    if (current.memoryMB > this.thresholds.maxMemoryMB) {
      this.alert('critical', 'Memory pressure', \`\${current.memoryMB.toFixed(0)}MB used\`);
    }

    if (current.avgInferenceMs > this.thresholds.maxInferenceMs) {
      this.alert('warning', 'High inference latency', \`\${current.avgInferenceMs.toFixed(0)}ms avg\`);
    }

    if (current.errorRate > this.thresholds.maxErrorRate) {
      this.alert('critical', 'Error rate exceeded', \`\${(current.errorRate * 100).toFixed(1)}%\`);
    }
  }

  private alert(severity: 'info' | 'warning' | 'critical', title: string, detail: string) {
    this.alerts.push({ severity, title, detail, timestamp: Date.now() });
    if (severity === 'critical') this.notifyTeam(title, detail);
  }

  private recoverTensorLeak() {
    // Force garbage collection of orphaned tensors
    tf.engine().startScope();
    tf.engine().endScope();
    console.log('Tensor cleanup executed, count:', tf.memory().numTensors);
  }

  private notifyTeam(title: string, detail: string) {
    // Send alert via webhook; Erdem gets a push notification
    fetch('/api/alerts', {
      method: 'POST',
      body: JSON.stringify({ title, detail, timestamp: Date.now() }),
    }).catch(() => { /* alert system itself failed; log locally */ });
  }

  private getAverageInference(): number { return 0; /* from metrics collector */ }
  private getErrorRate(): number { return 0; /* from error tracker */ }
  private getConnectionCount(): number { return 0; /* from connection manager */ }
}`}
</CodeBlock>

## Circuit Breaker

<CodeBlock language="typescript" title="circuit-breaker.ts">
{`// Circuit breaker: isolate failing modules, keep the rest running
class CircuitBreaker {
  private failureCount = 0;
  private lastFailure = 0;
  private state: 'closed' | 'open' | 'half-open' = 'closed';

  constructor(
    private threshold: number = 5,
    private resetTimeMs: number = 30000,
    private moduleName: string = 'unknown'
  ) {}

  async execute<T>(fn: () => Promise<T>, fallback: () => T): Promise<T> {
    if (this.state === 'open') {
      if (Date.now() - this.lastFailure > this.resetTimeMs) {
        this.state = 'half-open'; // Try one request
      } else {
        console.log(\`[\${this.moduleName}] Circuit OPEN; serving fallback\`);
        return fallback();
      }
    }

    try {
      const result = await fn();
      if (this.state === 'half-open') {
        this.state = 'closed'; // Recovery successful
        this.failureCount = 0;
      }
      return result;
    } catch (err) {
      this.failureCount++;
      this.lastFailure = Date.now();

      if (this.failureCount >= this.threshold) {
        this.state = 'open';
        console.log(\`[\${this.moduleName}] Circuit OPENED after \${this.threshold} failures\`);
      }

      return fallback();
    }
  }
}

// Each modality gets its own circuit breaker
const imageBreaker = new CircuitBreaker(3, 15000, 'image-generator');
const audioBreaker = new CircuitBreaker(3, 15000, 'audio-synthesizer');

// If image generation fails 3 times, serve cached images for 15 seconds
const image = await imageBreaker.execute(
  () => imageModel.generate(latent),
  () => getCachedFallback('image')
);`}
</CodeBlock>
