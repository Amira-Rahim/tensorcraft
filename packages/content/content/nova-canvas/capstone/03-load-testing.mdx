## Visitor Simulator

<CodeBlock language="typescript" title="visitor-simulator.ts">
{`// Simulate a single gallery visitor session
class VirtualVisitor {
  private ws: WebSocket;
  private metrics: VisitorMetrics;

  constructor(serverUrl: string) {
    this.ws = new WebSocket(serverUrl);
    this.metrics = { connectTime: 0, generations: [], errors: [] };
  }

  async simulateVisit(durationMs: number = 60000) {
    const connectStart = performance.now();
    await this.waitForOpen();
    this.metrics.connectTime = performance.now() - connectStart;

    const endTime = Date.now() + durationMs;

    while (Date.now() < endTime) {
      // Simulate visitor behavior: observe, then interact
      await this.wait(2000 + Math.random() * 5000); // Think time

      const genStart = performance.now();
      try {
        await this.requestGeneration({
          prompt: this.randomPrompt(),
          latentVector: Array.from({ length: 128 }, () => Math.random() * 2 - 1),
        });
        this.metrics.generations.push(performance.now() - genStart);
      } catch (err) {
        this.metrics.errors.push((err as Error).message);
      }
    }

    this.ws.close();
    return this.metrics;
  }

  private randomPrompt(): string {
    const prompts = [
      'a dragon made of stars', 'sunset over crystal mountains',
      'abstract emotion in blue', 'forest of light', 'ocean of sound',
      'geometric dreams', 'neon wilderness', 'ancient future city',
    ];
    return prompts[Math.floor(Math.random() * prompts.length)];
  }

  private requestGeneration(params: any): Promise<any> {
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => reject(new Error('Timeout')), 10000);
      this.ws.send(JSON.stringify({ type: 'generate', ...params }));
      this.ws.addEventListener('message', (e) => {
        const data = JSON.parse(e.data);
        if (data.status === 'complete') { clearTimeout(timeout); resolve(data); }
        if (data.status === 'error') { clearTimeout(timeout); reject(new Error(data.message)); }
      }, { once: true });
    });
  }

  private waitForOpen(): Promise<void> {
    return new Promise((resolve) => {
      if (this.ws.readyState === WebSocket.OPEN) return resolve();
      this.ws.addEventListener('open', () => resolve(), { once: true });
    });
  }

  private wait(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

interface VisitorMetrics {
  connectTime: number;
  generations: number[];
  errors: string[];
}`}
</CodeBlock>

## Results Analysis

<CodeBlock language="typescript" title="analyze-results.ts">
{`// Analyze load test results — same metrics as API load testing
function analyzeResults(allMetrics: VisitorMetrics[]): LoadTestReport {
  const allLatencies = allMetrics.flatMap(m => m.generations).sort((a, b) => a - b);
  const allErrors = allMetrics.flatMap(m => m.errors);

  return {
    totalVisitors: allMetrics.length,
    totalGenerations: allLatencies.length,
    totalErrors: allErrors.length,
    errorRate: allErrors.length / (allLatencies.length + allErrors.length),
    latency: {
      p50: percentile(allLatencies, 50),
      p95: percentile(allLatencies, 95),
      p99: percentile(allLatencies, 99),
      max: allLatencies[allLatencies.length - 1],
    },
    connectTime: {
      avg: avg(allMetrics.map(m => m.connectTime)),
      max: Math.max(...allMetrics.map(m => m.connectTime)),
    },
    verdict: allLatencies.length > 0 && percentile(allLatencies, 95) < 2000
      ? 'PASS' : 'FAIL',
  };
}

function percentile(sorted: number[], p: number): number {
  const idx = Math.ceil((p / 100) * sorted.length) - 1;
  return sorted[Math.max(0, idx)];
}

function avg(nums: number[]): number {
  return nums.reduce((a, b) => a + b, 0) / nums.length;
}`}
</CodeBlock>
