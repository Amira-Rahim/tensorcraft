If you've written CSS transitions, this shape is already familiar: `transition: transform 0.3s ease-in-out` tells the browser to smoothly interpolate between the start and end states. Latent space interpolation works on the same principle, but instead of interpolating CSS properties, it interpolates entire images, sounds, or artworks.

## Linear Interpolation (Lerp)

The simplest interpolation is linear: blend two vectors with a parameter `t` that goes from 0 to 1.

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="latent-interpolation.ts">
{`import * as tf from '@tensorflow/tfjs';

// Browser: CSS-like interpolation between two states
function lerpColor(colorA: number[], colorB: number[], t: number): number[] {
  return colorA.map((a, i) => a * (1 - t) + colorB[i] * t);
}
// lerpColor([255, 0, 0], [0, 0, 255], 0.5) → [127.5, 0, 127.5]

// ML: interpolation between two latent vectors
function lerpLatent(z1: tf.Tensor, z2: tf.Tensor, t: number): tf.Tensor {
  return z1.mul(1 - t).add(z2.mul(t));
}

// Generate the walkway morph: 10 steps between two artworks
async function generateMorph(
  z1: tf.Tensor,
  z2: tf.Tensor,
  decoder: tf.LayersModel,
  steps: number = 10
): Promise<tf.Tensor[]> {
  const frames: tf.Tensor[] = [];
  for (let i = 0; i <= steps; i++) {
    const t = i / steps;
    const zInterp = lerpLatent(z1, z2, t);
    const frame = decoder.predict(zInterp) as tf.Tensor;
    frames.push(frame);
  }
  return frames;
}`}
</CodeBlock>

## Why This Works

In a well-structured latent space, nearby points represent similar data. So a point halfway between "geometric pattern" and "organic swirl" produces something that is a coherent blend of both, not random noise.

<CodeBlock language="typescript" title="interpolation-quality.ts">
{`import * as tf from '@tensorflow/tfjs';

// GOOD latent space: interpolation produces meaningful results
// z1 = [0.2, 0.8, -0.3]  → sharp geometric pattern
// z2 = [-0.5, 0.1, 0.9]  → soft organic curves
// midpoint → blend of geometric and organic (coherent!)

// BAD latent space: interpolation produces garbage
// Midpoint between two valid points might be in a "dead zone"
// where the decoder produces meaningless output

// Spherical interpolation (slerp) keeps every step the same distance from the origin,
// where the training samples sit; a straight line dips toward the middle, where they don't
function slerp(z1: tf.Tensor, z2: tf.Tensor, t: number): tf.Tensor {
  const omega = tf.acos(
    z1.div(z1.norm()).dot(z2.div(z2.norm())).clipByValue(-1, 1)
  ) as tf.Scalar;
  const sinOmega = omega.sin();
  const coeff1 = omega.mul(1 - t).sin().div(sinOmega);
  const coeff2 = omega.mul(t).sin().div(sinOmega);
  return z1.mul(coeff1).add(z2.mul(coeff2));
}`}
</CodeBlock>
