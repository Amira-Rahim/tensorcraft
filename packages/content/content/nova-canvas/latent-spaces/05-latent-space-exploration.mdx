## Architecture

The explorer follows a familiar frontend pattern: user input drives state, state drives rendering.

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="explorer-architecture.ts">
{`import * as tf from '@tensorflow/tfjs';

// The pattern is identical to any slider-driven UI:
// Input → State → Render

// 1. Capture slider values
const sliders = Array.from({ length: 16 }, (_, i) => {
  const input = document.createElement('input');
  input.type = 'range';
  input.min = '-3';
  input.max = '3';
  input.step = '0.1';
  input.value = '0';
  input.addEventListener('input', () => updateLatentVector());
  return input;
});

// 2. Build latent vector from slider values
function getLatentVector(): tf.Tensor2D {
  const values = sliders.map(s => parseFloat(s.value));
  return tf.tensor2d([values]);
}

// 3. Decode and render
async function updateLatentVector() {
  const z = getLatentVector();
  const output = decoder.predict(z) as tf.Tensor;
  await renderToCanvas(output.reshape([28, 28]));
  z.dispose();
  output.dispose();  // Prevent memory leaks!
}`}
</CodeBlock>

## Performance: Memory and Speed

TensorFlow.js tensors live on the GPU. If you do not dispose them, you get memory leaks — the GPU equivalent of DOM node leaks.

<CodeBlock language="typescript" title="performance.ts">
{`import * as tf from '@tensorflow/tfjs';

// BAD: leaking tensors on every slider update
function leakyUpdate() {
  const z = getLatentVector();      // allocates GPU memory
  const output = decoder.predict(z); // allocates more GPU memory
  renderToCanvas(output);
  // z and output are never disposed — memory leak!
}

// GOOD: tidy() automatically disposes intermediate tensors
function cleanUpdate() {
  const output = tf.tidy(() => {
    const z = getLatentVector();
    return decoder.predict(z) as tf.Tensor;
  });
  // Only 'output' survives — everything else is disposed

  renderToCanvas(output).then(() => output.dispose());
}

// ALSO GOOD: debounce rapid slider updates
function debounce(fn: Function, ms: number) {
  let timer: ReturnType<typeof setTimeout>;
  return (...args: unknown[]) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), ms);
  };
}
const debouncedUpdate = debounce(cleanUpdate, 16); // ~60fps cap`}
</CodeBlock>

## The 2D Map

For an exhibition, sliders are precise but not intuitive. A 2D map lets visitors click or drag to explore two dimensions at once — like a touchpad for the latent space.

<CodeBlock language="typescript" title="latent-map.ts">
{`function create2DMap(
  canvas: HTMLCanvasElement,
  decoder: tf.LayersModel,
  dim1: number,
  dim2: number,
  baseVector: number[]
) {
  canvas.addEventListener('mousemove', async (e) => {
    if (e.buttons !== 1) return; // Only while clicking

    const rect = canvas.getBoundingClientRect();
    const x = ((e.clientX - rect.left) / rect.width) * 6 - 3;  // -3 to 3
    const y = ((e.clientY - rect.top) / rect.height) * 6 - 3;

    const z = [...baseVector];
    z[dim1] = x;
    z[dim2] = y;

    const output = tf.tidy(() => {
      const zTensor = tf.tensor2d([z]);
      return decoder.predict(zTensor) as tf.Tensor;
    });

    await renderToCanvas(output.reshape([28, 28]));
    output.dispose();
  });
}`}
</CodeBlock>
