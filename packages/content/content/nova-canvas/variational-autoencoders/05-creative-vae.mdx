## The Complete VAE

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="creative-vae.ts">
{`import * as tf from '@tensorflow/tfjs';

function buildCreativeVAE(inputDim: number, latentDim: number) {
  // --- Encoder ---
  const encoderInput = tf.input({ shape: [inputDim] });
  const encHidden1 = tf.layers.dense({ units: 512, activation: 'relu' })
    .apply(encoderInput);
  const encHidden2 = tf.layers.dense({ units: 256, activation: 'relu' })
    .apply(encHidden1);

  const mu = tf.layers.dense({ units: latentDim, name: 'mu' })
    .apply(encHidden2) as tf.SymbolicTensor;
  const logVar = tf.layers.dense({ units: latentDim, name: 'logVar' })
    .apply(encHidden2) as tf.SymbolicTensor;

  const encoder = tf.model({
    inputs: encoderInput,
    outputs: [mu, logVar],
  });

  // --- Decoder ---
  const decoderInput = tf.input({ shape: [latentDim] });
  const decHidden1 = tf.layers.dense({ units: 256, activation: 'relu' })
    .apply(decoderInput);
  const decHidden2 = tf.layers.dense({ units: 512, activation: 'relu' })
    .apply(decHidden1);
  const decoderOutput = tf.layers.dense({ units: inputDim, activation: 'sigmoid' })
    .apply(decHidden2) as tf.SymbolicTensor;

  const decoder = tf.model({
    inputs: decoderInput,
    outputs: decoderOutput,
  });

  return { encoder, decoder };
}`}
</CodeBlock>

## Custom Training Loop

Because the VAE loss has two components, we use a custom training loop with `tf.train.adam` and `tf.variableGrads`.

<CodeBlock language="typescript" title="vae-training.ts">
{`import * as tf from '@tensorflow/tfjs';

async function trainCreativeVAE(
  encoder: tf.LayersModel,
  decoder: tf.LayersModel,
  data: tf.Tensor2D,
  latentDim: number,
  epochs: number = 100,
  beta: number = 1.0
) {
  const optimizer = tf.train.adam(0.001);
  const batchSize = 32;
  const numBatches = Math.floor(data.shape[0] / batchSize);

  for (let epoch = 0; epoch < epochs; epoch++) {
    let totalLoss = 0;

    for (let b = 0; b < numBatches; b++) {
      const batch = data.slice([b * batchSize, 0], [batchSize, -1]);

      const loss = optimizer.minimize(() => {
        const [mu, logVar] = encoder.predict(batch) as tf.Tensor[];

        // Reparameterization
        const epsilon = tf.randomNormal([batchSize, latentDim]);
        const z = mu.add(logVar.mul(0.5).exp().mul(epsilon));

        // Decode
        const reconstructed = decoder.predict(z) as tf.Tensor;

        // Reconstruction loss
        const reconLoss = batch.sub(reconstructed).square().sum(1).mean();

        // KL divergence
        const klLoss = mu.square().add(logVar.exp()).sub(logVar).sub(1)
          .mul(0.5).sum(1).mean();

        return reconLoss.add(klLoss.mul(beta)) as tf.Scalar;
      }, true) as tf.Scalar;

      totalLoss += (await loss.array());
      loss.dispose();
    }

    if (epoch % 10 === 0) {
      console.log(\`Epoch \${epoch}: loss=\${(totalLoss / numBatches).toFixed(4)}\`);
    }
  }
}`}
</CodeBlock>

## Interactive Generation

<CodeBlock language="typescript" title="interactive-generation.ts">
{`// Design token style interface for the exhibition kiosk
interface ArtTokens {
  warmth: number;      // z[0]: cool blues ← → warm reds
  complexity: number;  // z[1]: minimal ← → intricate
  symmetry: number;    // z[2]: asymmetric ← → symmetric
  texture: number;     // z[3]: smooth ← → textured
}

function tokensToLatent(tokens: ArtTokens, latentDim: number): number[] {
  const z = new Array(latentDim).fill(0);
  z[0] = tokens.warmth;
  z[1] = tokens.complexity;
  z[2] = tokens.symmetry;
  z[3] = tokens.texture;
  return z;
}

async function generateFromTokens(
  tokens: ArtTokens,
  decoder: tf.LayersModel,
  latentDim: number
): Promise<number[][]> {
  const output = tf.tidy(() => {
    const z = tf.tensor2d([tokensToLatent(tokens, latentDim)]);
    return (decoder.predict(z) as tf.Tensor).reshape([28, 28]);
  });
  const result = await output.array() as number[][];
  output.dispose();
  return result;
}`}
</CodeBlock>

<DerivationDrawer summary="Latent traversal mathematics">
After training, you can sample from the prior `z ~ N(0, I)` and decode to generate new samples. Or you can traverse latent dimensions one at a time:

```
for d in 1..latent_dim:
  for v in linspace(-3, 3, 10):
    z = zero_vector_of_size(latent_dim)
    z[d] = v
    decoded = decoder(z)
    show(decoded)
```

**Interpretation:** if dim d encodes a meaningful axis (e.g., "digit width" for MNIST), varying it should produce a smooth interpolation along that axis.

**Disentanglement** is the property that distinct latent dimensions correspond to distinct semantic factors. Vanilla VAE often produces entangled latents (one factor smeared across many dims). β-VAE, FactorVAE, TCVAE all add penalties that push toward disentanglement, at the cost of reconstruction quality. There is no universal disentanglement metric; the term is empirical and contested.
</DerivationDrawer>
