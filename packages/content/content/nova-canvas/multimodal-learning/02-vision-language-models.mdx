Which is why anything accessibility-critical still routes every generated caption through a human. The model proposes. Someone who actually looked decides.
</Dialogue>

Vision-language models learn the image-text connection at scale: millions of pairs, encoded into a shared space where the image embedding and the text embedding of its description are neighbors. The scale is what changes the rules. No author saw these pairs. The alignment is statistical, and so are its mistakes.

{`import * as tf from '@tensorflow/tfjs';

// Cosine similarity: how aligned are two embeddings?
function cosineSimilarity(a: tf.Tensor1D, b: tf.Tensor1D): tf.Scalar {
  const dotProduct = tf.sum(tf.mul(a, b));
  const normA = tf.norm(a);
  const normB = tf.norm(b);
  return tf.div(dotProduct, tf.mul(normA, normB)) as tf.Scalar;
}

// Training data: image-caption pairs (like img + alt)
interface ImageTextPair {
  image: tf.Tensor4D;     // the img src
  caption: tf.Tensor2D;   // the alt text (tokenized)
}

// Contrastive loss: matched pairs should be similar,
// mismatched pairs should be dissimilar
function contrastiveLoss(
  imageEmbeddings: tf.Tensor2D, // [batchSize, embeddingDim]
  textEmbeddings: tf.Tensor2D   // [batchSize, embeddingDim]
): tf.Scalar {
  // Normalize embeddings
  const normImages = tf.div(imageEmbeddings,
    tf.norm(imageEmbeddings, 2, 1, true));
  const normTexts = tf.div(textEmbeddings,
    tf.norm(textEmbeddings, 2, 1, true));

  // Similarity matrix: every image against every caption
  const similarities = tf.matMul(normImages, normTexts, false, true);

  // The diagonal contains matched pairs (img[i] with caption[i])
  // Off-diagonal contains mismatched pairs
  const batchSize = imageEmbeddings.shape[0];
  const labels = tf.range(0, batchSize, 1, 'int32');

  // Cross-entropy: matched pairs should have highest similarity
  const temperature = 0.07;
  const logits = tf.div(similarities, tf.scalar(temperature));
  return tf.losses.softmaxCrossEntropy(
    tf.oneHot(labels, batchSize),
    logits
  );
}

// After training:
// encode(sunsetImage) ≈ encode("a golden sunset over the ocean")
// encode(sunsetImage) ≠ encode("a cat sleeping on a couch")`}
