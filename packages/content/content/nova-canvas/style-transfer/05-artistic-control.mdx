In CSS, `mix-blend-mode` layers visual effects — multiply, screen, overlay. Style blending does the same thing. We interpolate between two Gram matrices and use the blend as our target. You get a style that is half Monet, half Basquiat — or any ratio you choose.

CSS `mix-blend-mode` layers visual effects to create composites. Style blending interpolates between artistic representations to create hybrid styles. Both give you creative control over how visual identities combine.

## Blending Two Styles

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="style-blending.ts">
{`import * as tf from '@tensorflow/tfjs';

// Linear interpolation between two style representations
function blendStyles(
  gramA: tf.Tensor2D[],
  gramB: tf.Tensor2D[],
  alpha: number // 0 = pure A, 1 = pure B
): tf.Tensor2D[] {
  return gramA.map((gA, i) => {
    const gB = gramB[i];
    // blended = (1 - alpha) * A + alpha * B
    return tf.add(
      tf.mul(gA, tf.scalar(1 - alpha)),
      tf.mul(gB, tf.scalar(alpha))
    ) as tf.Tensor2D;
  });
}

// Create a gradient of styles
function styleGradient(
  gramMonet: tf.Tensor2D[],
  gramBasquiat: tf.Tensor2D[],
  steps: number
): tf.Tensor2D[][] {
  const gradient: tf.Tensor2D[][] = [];
  for (let i = 0; i <= steps; i++) {
    const alpha = i / steps;
    gradient.push(blendStyles(gramMonet, gramBasquiat, alpha));
  }
  return gradient;
  // [0] = pure Monet
  // [steps/2] = 50/50 blend
  // [steps] = pure Basquiat
}

// Spatial style blending: different styles in different regions
function spatialBlend(
  gramA: tf.Tensor2D[],
  gramB: tf.Tensor2D[],
  mask: tf.Tensor2D // 0-1 mask, same spatial dims as features
): { foreground: tf.Tensor2D[]; background: tf.Tensor2D[] } {
  // Where mask = 1, use style A; where mask = 0, use style B
  // This allows Monet sky + Basquiat buildings
  return {
    foreground: gramA,
    background: gramB
  };
}

// Yara's dial: visitors control the blend in real time
// alpha = 0.0 -> "Impressionist Dawn"
// alpha = 0.3 -> "Structured Light"
// alpha = 0.5 -> "Urban Impressionism"
// alpha = 0.7 -> "Energetic Calm"
// alpha = 1.0 -> "Raw Expression"
console.log('Each blend point is a new aesthetic that exists nowhere else');`}
