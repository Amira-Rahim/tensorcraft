function buildStyleExtractor() {
  const input = tf.input({ shape: [128, 128, 3] });

  const conv1 = tf.layers.conv2d({
    filters: 32, kernelSize: 3, padding: 'same', activation: 'relu', name: 'block1'
  }).apply(input) as tf.SymbolicTensor;
  const pool1 = tf.layers.maxPooling2d({ poolSize: 2 }).apply(conv1) as tf.SymbolicTensor;

  const conv2 = tf.layers.conv2d({
    filters: 64, kernelSize: 3, padding: 'same', activation: 'relu', name: 'block2'
  }).apply(pool1) as tf.SymbolicTensor;
  const pool2 = tf.layers.maxPooling2d({ poolSize: 2 }).apply(conv2) as tf.SymbolicTensor;

  const conv3 = tf.layers.conv2d({
    filters: 128, kernelSize: 3, padding: 'same', activation: 'relu', name: 'block3'
  }).apply(pool2) as tf.SymbolicTensor;

  // Return activations from ALL style-relevant layers
  return tf.model({
    inputs: input,
    outputs: [conv1, conv2, conv3] // multi-scale outputs
  });
}

function multiScaleStyleLoss(
  generatedGrams: tf.Tensor2D[],
  targetGrams: tf.Tensor2D[],
  weights: number[] = [1, 1, 1]
): tf.Scalar {
  let total = tf.scalar(0);

<DerivationDrawer summary="Why Gram matrix captures style">
**Gram matrix is a feature-correlation matrix:**

`G[i, j] = ⟨F[:, i], F[:, j]⟩`     (inner product of channels i and j across all spatial positions)

**What this captures:**
- Which feature channels fire together (e.g., "diagonal edges" + "red color" co-occur in this style)
- Independent of WHERE in the image they fire (sums over spatial positions destroys location)

**What it discards:**
- Spatial arrangement (which corner has the brushstroke)
- Object identity (whether the texture covers a face or a tree)

**This is exactly the right tradeoff for style:** style is about texture statistics that should be uniform across an image regardless of content layout. Content loss (single feature map L2) preserves spatial arrangement.

**Computational note:** G is `[C, C]` where C is the number of channels (typically 64-512). For VGG layer with 256 channels, G is 256×256 = 65k entries. Computing G_style for one image once + caching is a major speedup.
</DerivationDrawer>
