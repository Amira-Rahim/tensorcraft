## Generation Queue

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="generation-queue.ts">
{`// Bounded concurrency queue — same pattern as database connection pools
class GenerationQueue {
  private queue: QueueItem[] = [];
  private active = 0;
  private maxConcurrent: number;

  constructor(maxConcurrent: number = 4) {
    this.maxConcurrent = maxConcurrent;
  }

  async enqueue(request: GenerationRequest): Promise<GenerationResult> {
    return new Promise((resolve, reject) => {
      const item: QueueItem = {
        request,
        resolve,
        reject,
        enqueuedAt: Date.now(),
      };

      this.queue.push(item);
      this.processNext();
    });
  }

  private async processNext() {
    if (this.active >= this.maxConcurrent || this.queue.length === 0) return;

    this.active++;
    const item = this.queue.shift()!;

    try {
      const waitTime = Date.now() - item.enqueuedAt;
      console.log(\`Processing after \${waitTime}ms wait, \${this.queue.length} remaining\`);

      const result = await this.generate(item.request);
      item.resolve(result);
    } catch (err) {
      item.reject(err);
    } finally {
      this.active--;
      this.processNext(); // Process next in queue
    }
  }

  private async generate(req: GenerationRequest): Promise<GenerationResult> {
    const latent = tf.tensor2d([req.latentVector]);
    const output = model.predict(latent) as tf.Tensor;
    const data = await output.data() as Float32Array;
    latent.dispose();
    output.dispose();
    return { data, prompt: req.prompt };
  }

  get stats() {
    return {
      queueLength: this.queue.length,
      activeGenerations: this.active,
      estimatedWaitMs: (this.queue.length / this.maxConcurrent) * 340,
    };
  }
}

interface QueueItem {
  request: GenerationRequest;
  resolve: (result: GenerationResult) => void;
  reject: (err: any) => void;
  enqueuedAt: number;
}

interface GenerationRequest { prompt: string; latentVector: number[]; }
interface GenerationResult { data: Float32Array; prompt: string; }`}
</CodeBlock>

## WebSocket Connection Manager

<CodeBlock language="typescript" title="ws-manager.ts">
{`// Manage gallery visitor connections — each visitor gets real-time updates
class GalleryConnectionManager {
  private connections = new Map<string, WebSocket>();
  private queue: GenerationQueue;

  constructor(queue: GenerationQueue) {
    this.queue = queue;
  }

  handleConnection(ws: WebSocket, visitorId: string) {
    this.connections.set(visitorId, ws);

    ws.addEventListener('message', async (event) => {
      const request = JSON.parse(event.data);

      // Send queue position immediately
      this.sendStatus(visitorId, {
        status: 'queued',
        position: this.queue.stats.queueLength,
        estimatedWaitMs: this.queue.stats.estimatedWaitMs,
      });

      try {
        const result = await this.queue.enqueue(request);
        this.sendStatus(visitorId, { status: 'complete', data: Array.from(result.data) });
      } catch (err) {
        this.sendStatus(visitorId, { status: 'error', message: 'Generation failed' });
      }
    });

    ws.addEventListener('close', () => {
      this.connections.delete(visitorId);
    });

    // Send current stats on connect
    this.sendStatus(visitorId, {
      status: 'connected',
      activeVisitors: this.connections.size,
    });
  }

  private sendStatus(visitorId: string, status: object) {
    const ws = this.connections.get(visitorId);
    if (ws?.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify(status));
    }
  }
}`}
</CodeBlock>

## Load Shedding

<CodeBlock language="typescript" title="load-shedding.ts">
{`// Graceful degradation when queue is overwhelmed
function handleOverload(queue: GenerationQueue, request: GenerationRequest): Response {
  const stats = queue.stats;

  if (stats.queueLength > 50) {
    // Serve cached/lower-quality version instead of queueing
    return { type: 'cached', data: getCachedFallback(request.prompt) };
  }

  if (stats.estimatedWaitMs > 5000) {
    // Reduce quality for faster generation
    return { type: 'fast', quality: 'reduced', data: generateFast(request) };
  }

  // Normal path
  return { type: 'queued', estimatedWaitMs: stats.estimatedWaitMs };
}`}
