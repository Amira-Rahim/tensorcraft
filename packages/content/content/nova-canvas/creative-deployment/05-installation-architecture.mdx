self.addEventListener('install', (event: ExtendableEvent) => {
  event.waitUntil(caches.open('v1').then(c => c.addAll([
    '/app.js',                          //   1.2 MB
    '/styles.css',                      //   0.1 MB
    '/models/nova-canvas/weights.bin',  //  47 MB <- here it dies
  ])));
});
// cache.addAll() is all-or-nothing: one flaky packet at megabyte 45
// and the entire install rejects, app shell included. Now picture
// twelve tablets each retrying a 47 MB atomic download on shared
// gallery Wi-Fi. The service-worker pattern transfers; the payload
// breaks it. The architecture below splits the caches and gates
// generation on the weights, not the shell.`}

<CodeBlock language="typescript" title="sw-installation.ts">
{`// Service Worker: cache everything needed for offline operation
const MODEL_CACHE = 'nova-canvas-model-v1';
const APP_CACHE = 'nova-canvas-app-v1';

const MODEL_FILES = [
  '/models/nova-canvas-optimized/model.json',
  '/models/nova-canvas-optimized/weights.bin',
  '/models/audio-synth/model.json',
  '/models/audio-synth/weights.bin',
];

const APP_FILES = [
  '/',
  '/gallery',
  '/index.html',
  '/app.js',
  '/styles.css',
  '/tfjs/tf.min.js',
];

// Pre-cache models and app shell on install
self.addEventListener('install', (event: ExtendableEvent) => {
  event.waitUntil(
    Promise.all([
      caches.open(MODEL_CACHE).then(cache => cache.addAll(MODEL_FILES)),
      caches.open(APP_CACHE).then(cache => cache.addAll(APP_FILES)),
    ])
  );
});

// Serve from cache, fall back to network
self.addEventListener('fetch', (event: FetchEvent) => {
  event.respondWith(
    caches.match(event.request).then(cached => {
      return cached || fetch(event.request).then(response => {
        // Cache new requests for future offline use
        const cache = event.request.url.includes('/models/')
          ? MODEL_CACHE : APP_CACHE;
        caches.open(cache).then(c => c.put(event.request, response.clone()));
        return response;
      });
    })
  );
});`}
</CodeBlock>

## Local-First Generation

<CodeBlock language="typescript" title="local-first.ts">
{`// All generation runs locally; no server round-trip needed
class LocalFirstGenerator {
  private model: tf.LayersModel | null = null;
  private db: IDBDatabase | null = null;

  async initialize() {
    // Load model from Service Worker cache; no network needed
    this.model = await tf.loadLayersModel('/models/nova-canvas-optimized/model.json');

    // IndexedDB for generation history
    this.db = await this.openDB();
    console.log('Generator initialized, fully offline capable');
  }

  async generate(params: GenerationParams): Promise<GenerationResult> {
    if (!this.model) throw new Error('Model not loaded');

    const result = await this.runInference(params);

    // Store result locally
    await this.storeResult(result);

    // Background sync when online
    if (navigator.onLine) {
      this.syncResults();
    }

    return result;
  }

  private async runInference(params: GenerationParams): Promise<GenerationResult> {
    return tf.tidy(() => {
      const latent = tf.tensor2d([params.latentVector]);
      const output = this.model!.predict(latent) as tf.Tensor;
      return {
        id: crypto.randomUUID(),
        data: output.dataSync() as Float32Array,
        params,
        timestamp: Date.now(),
        syncState: 'pending',
      };
    });
  }

  private async storeResult(result: GenerationResult) {
    const tx = this.db!.transaction('generations', 'readwrite');
    tx.objectStore('generations').put(result);
  }

  private async syncResults() {
    const unsynced = await this.getUnsynced();

    for (const result of unsynced) {
      try {
        await fetch('/api/sync', {
          method: 'POST',
          body: JSON.stringify(result),
        });
        result.syncState = 'synced';
        await this.put(result);
      } catch { /* retry later */ }
    }
  }

  // IndexedDB keys cannot be booleans, so sync state is a string
  private getUnsynced(): Promise<GenerationResult[]> {
    return new Promise((resolve, reject) => {
      const tx = this.db!.transaction('generations', 'readonly');
      const req = tx.objectStore('generations').index('syncState').getAll('pending');
      req.onsuccess = () => resolve(req.result);
      req.onerror = () => reject(req.error);
    });
  }

  private put(result: GenerationResult): Promise<void> {
    return new Promise((resolve, reject) => {
      const tx = this.db!.transaction('generations', 'readwrite');
      tx.objectStore('generations').put(result);
      tx.oncomplete = () => resolve();
      tx.onerror = () => reject(tx.error);
    });
  }

  private openDB(): Promise<IDBDatabase> {
    return new Promise((resolve, reject) => {
      const req = indexedDB.open('nova-canvas', 1);
      req.onupgradeneeded = () => {
        const db = req.result;
        const store = db.createObjectStore('generations', { keyPath: 'id' });
        store.createIndex('syncState', 'syncState');
      };
      req.onsuccess = () => resolve(req.result);
      req.onerror = () => reject(req.error);
    });
  }
}`}
</CodeBlock>
