## Instrumenting Generation

<CodeBlock language="typescript" title="profiling-harness.ts">
{`// Profile each stage of the generation pipeline
// Same API you use for profiling React renders

async function profileGeneration(model: tf.LayersModel, input: tf.Tensor) {
  const metrics: Record<string, number> = {};

  // Stage 1: Tensor preparation
  performance.mark('prep-start');
  const preprocessed = input.expandDims(0).div(255);
  performance.mark('prep-end');
  performance.measure('preprocessing', 'prep-start', 'prep-end');

  // Stage 2: Inference
  performance.mark('inference-start');
  const output = model.predict(preprocessed) as tf.Tensor;
  // Force synchronous execution to get accurate timing
  await output.data();
  performance.mark('inference-end');
  performance.measure('inference', 'inference-start', 'inference-end');

  // Stage 3: Post-processing
  performance.mark('post-start');
  const result = output.squeeze().mul(255).clipByValue(0, 255);
  const imageData = await result.data();
  performance.mark('post-end');
  performance.measure('postprocessing', 'post-start', 'post-end');

  // Collect measurements
  const entries = performance.getEntriesByType('measure');
  entries.forEach(entry => {
    metrics[entry.name] = entry.duration;
  });

  // Clean up
  preprocessed.dispose();
  output.dispose();
  result.dispose();
  performance.clearMarks();
  performance.clearMeasures();

  return metrics;
}

const metrics = await profileGeneration(model, inputTensor);
console.table(metrics);
// preprocessing: 2.1ms | inference: 2847ms | postprocessing: 45ms`}
</CodeBlock>

## Memory Profiling

<CodeBlock language="typescript" title="memory-profiling.ts">
{`// Track tensor memory — the ML equivalent of tracking DOM node count
function logMemory(label: string) {
  const mem = tf.memory();
  console.log(\`[\${label}] Tensors: \${mem.numTensors}, Bytes: \${(mem.numBytes / 1024 / 1024).toFixed(2)}MB\`);
}

// Profile a generation session
logMemory('before generation');

for (let i = 0; i < 10; i++) {
  const z = tf.randomNormal([1, 128]);
  const generated = model.predict(z) as tf.Tensor;
  await generated.data();

  logMemory(\`after generation \${i}\`);

  // Without dispose(), tensor count grows every iteration — memory leak
  z.dispose();
  generated.dispose();
}

logMemory('after cleanup');

// Use tf.tidy() to auto-dispose intermediate tensors
const result = tf.tidy(() => {
  const z = tf.randomNormal([1, 128]);
  const raw = model.predict(z) as tf.Tensor;
  return raw.squeeze().mul(255);
});
// Only 'result' survives — all intermediates are freed`}
</CodeBlock>

## Continuous Monitoring

<CodeBlock language="typescript" title="perf-monitor.ts">
{`// Real-time performance overlay — like an FPS counter for generation
class PerfMonitor {
  private history: number[] = [];
  private maxHistory = 60;

  record(generationTimeMs: number) {
    this.history.push(generationTimeMs);
    if (this.history.length > this.maxHistory) this.history.shift();
  }

  get average(): number {
    return this.history.reduce((a, b) => a + b, 0) / this.history.length;
  }

  get fps(): number {
    return this.average > 0 ? 1000 / this.average : 0;
  }

  render(ctx: CanvasRenderingContext2D, x: number, y: number) {
    ctx.fillStyle = this.average < 100 ? '#00ff88' : this.average < 500 ? '#ffaa00' : '#ff4444';
    ctx.font = '14px monospace';
    ctx.fillText(\`Gen: \${this.average.toFixed(0)}ms (\${this.fps.toFixed(1)} gen/s)\`, x, y);
    ctx.fillText(\`Tensors: \${tf.memory().numTensors}\`, x, y + 18);
    ctx.fillText(\`VRAM: \${(tf.memory().numBytes / 1024 / 1024).toFixed(1)}MB\`, x, y + 36);
  }
}`}
</CodeBlock>
