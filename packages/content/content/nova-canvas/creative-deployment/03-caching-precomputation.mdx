Cache the predictable. Generate the novel. Simple. How much storage do we need?

Wrap two identical requests in `console.time('generate')` and the log reads 340ms, twice. The model did the same expensive work and billed you both times. This is the exact waste `React.memo` exists to prevent. Cache headers, Service Worker caches, `useMemo`, and CDN edges all make the same trade: storage for computation. Generation caching is no different. Store the popular outputs; skip the inference on repeats.

## Cache Architecture

<CodeBlock language="typescript" title="generation-cache.ts">
{`import * as tf from '@tensorflow/tfjs';

// LRU cache for generated outputs, same pattern as any memoization
interface CacheKeyParams {
  modelVersion: string;
  seed: number;
  steps: number;
  prompt: string;
}

class GenerationCache {
  private cache = new Map<string, Float32Array>();
  private accessOrder: string[] = [];
  private maxSize: number;
  private hits = 0;
  private misses = 0;

  constructor(maxSize: number = 200) {
    this.maxSize = maxSize;
  }

  // Everything that changes the output belongs in the key
  private hashInput(params: CacheKeyParams): string {
    return \`\${params.modelVersion}|\${params.seed}|\${params.steps}|\${params.prompt}\`;
  }

  get(params: CacheKeyParams): Float32Array | null {
    const key = this.hashInput(params);
    const result = this.cache.get(key);
    if (result) {
      this.hits++;
      // Move to end of access order (LRU update)
      this.accessOrder = this.accessOrder.filter(k => k !== key);
      this.accessOrder.push(key);
      return result;
    }
    this.misses++;
    return null;
  }

  set(params: CacheKeyParams, output: Float32Array) {
    const key = this.hashInput(params);
    // Re-setting a key must not leave a stale position behind
    this.accessOrder = this.accessOrder.filter(k => k !== key);
    // Evict least recently used if at capacity
    while (this.cache.size >= this.maxSize) {
      const oldest = this.accessOrder.shift()!;
      this.cache.delete(oldest);
    }
    this.cache.set(key, output);
    this.accessOrder.push(key);
  }

  get hitRate(): number {
    return this.hits / (this.hits + this.misses) || 0;
  }
}`}
</CodeBlock>

## Cache-First Generation

<CodeBlock language="typescript" title="cache-first.ts">
{`// Cache-first strategy, identical to Service Worker cache-first pattern
async function generateWithCache(
  model: tf.LayersModel,
  params: CacheKeyParams,
  cache: GenerationCache
): Promise<Float32Array> {
  // Check cache first: instant response for cached outputs
  const cached = cache.get(params);
  if (cached) {
    console.log('Cache HIT: 0ms generation');
    return cached;
  }

  // Cache miss: generate and store
  console.log('Cache MISS: generating...');
  const latent = paramsToLatent(params);
  const output = model.predict(latent) as tf.Tensor;
  const result = await output.data() as Float32Array;

  cache.set(params, new Float32Array(result));

  latent.dispose();
  output.dispose();
  return result;
}

function paramsToLatent(params: CacheKeyParams): tf.Tensor {
  // The seed comes in with the request, so the same key always
  // reproduces the same latent and the entry stays valid
  const values = hashToNumbers(String(params.seed));
  return tf.tensor2d([values.slice(0, 128)]);
}`}
</CodeBlock>

## Service Worker Integration

<CodeBlock language="typescript" title="sw-cache.ts">
{`// Service Worker cache for generated outputs; works offline
// sw.ts (Service Worker)

const CACHE_NAME = 'nova-canvas-generations-v1';

self.addEventListener('fetch', (event: FetchEvent) => {
  if (event.request.url.includes('/api/generate')) {
    event.respondWith(cacheFirstGeneration(event.request));
  }
});

async function cacheFirstGeneration(request: Request): Promise<Response> {
  const cache = await caches.open(CACHE_NAME);

  // Check cache
  const cached = await cache.match(request);
  if (cached) return cached;

  // Generate and cache
  const response = await fetch(request);
  const clone = response.clone();
  await cache.put(request, clone);

  return response;
}

// Pre-populate cache with popular outputs during installation
self.addEventListener('install', (event: ExtendableEvent) => {
  event.waitUntil(precomputePopular());
});

async function precomputePopular() {
  const cache = await caches.open(CACHE_NAME);
  const popularPrompts = ['sunset', 'dragon', 'abstract', 'ocean', 'forest'];

  for (const prompt of popularPrompts) {
    const response = await fetch(\`/api/generate?prompt=\${prompt}\`);
    await cache.put(new Request(\`/api/generate?prompt=\${prompt}\`), response);
  }
}`}
</CodeBlock>
