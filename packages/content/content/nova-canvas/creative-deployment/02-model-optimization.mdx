The first dry run on gallery Wi-Fi was grim: 47 megabytes of weights crawling through a shared access point while a test visitor stared at a spinner. Three levers move that number. Quantization cuts the precision of each weight, pruning deletes the weights that contribute least, and distillation retrains the knowledge into a smaller network. All three change what the model outputs, which is why every one of them ships with a benchmark attached.

{`import * as tf from '@tensorflow/tfjs';

// Float32 → Uint8 quantization
// Like converting PNG → JPEG: smaller on the wire, slightly lossy
async function quantizeWeights(model: tf.LayersModel): Promise<tf.LayersModel> {
  const quantizedWeights = model.getWeights().map(weight => {
    const data = weight.dataSync();
    // Never spread a weight tensor into Math.min: a layer this size
    // blows the call stack long before it returns a number.
    let min = Infinity, max = -Infinity;
    for (let i = 0; i < data.length; i++) {
      if (data[i] < min) min = data[i];
      if (data[i] > max) max = data[i];
    }
    const scale = (max - min) / 255;

    // Quantize: float32 → uint8 with scale and offset
    const quantized = new Uint8Array(data.length);
    for (let i = 0; i < data.length; i++) {
      quantized[i] = Math.round((data[i] - min) / scale);
    }

    // Dequantize back to float32 for inference
    const dequantized = new Float32Array(data.length);
    for (let i = 0; i < data.length; i++) {
      dequantized[i] = quantized[i] * scale + min;
    }

    return tf.tensor(dequantized, weight.shape);
  });

  model.setWeights(quantizedWeights);
  return model;
}

// Measure size reduction
const originalSize = model.getWeights()
  .reduce((sum, w) => sum + w.size * 4, 0); // float32 = 4 bytes
console.log('Original:', (originalSize / 1024 / 1024).toFixed(1), 'MB');
// With quantization: effectively 1/4 the precision, ~4x smaller transfer`}
</CodeBlock>

## Knowledge Distillation

<CodeBlock language="typescript" title="distillation.ts">
{`// Train a small "student" to mimic a large "teacher"
// Like code-splitting: the student handles common cases, teacher handles edge cases

const teacher = await tf.loadLayersModel('/models/nova-canvas-full/model.json');

// Student: 1/10th the parameters, same input/output shape
const student = tf.sequential({
  layers: [
    tf.layers.dense({ inputShape: [128], units: 64, activation: 'relu' }),
    tf.layers.dense({ units: 128, activation: 'relu' }),
    tf.layers.dense({ units: 784, activation: 'tanh' }), // same output as teacher
  ],
});

student.compile({ optimizer: tf.train.adam(0.001), loss: 'meanSquaredError' });

// Distillation: student learns from teacher's outputs, not raw data
async function distill(numBatches: number) {
  for (let i = 0; i < numBatches; i++) {
    const input = tf.randomNormal([32, 128]); // random latent vectors
    const teacherOutput = teacher.predict(input) as tf.Tensor;

    // Student learns to match teacher's output distribution
    await student.fit(input, teacherOutput, { epochs: 1, verbose: 0 });

    input.dispose();
    teacherOutput.dispose();
  }
}

await distill(500);

// Compare inference speed
async function benchmark(model: tf.LayersModel, label: string) {
  const input = tf.randomNormal([1, 128]);
  const start = performance.now();
  for (let i = 0; i < 100; i++) {
    const out = model.predict(input) as tf.Tensor;
    await out.data();
    out.dispose();
  }
  console.log(\`\${label}: \${((performance.now() - start) / 100).toFixed(1)}ms per inference\`);
  input.dispose();
}

await benchmark(teacher, 'Teacher');  // ~320ms
await benchmark(student, 'Student');  // ~34ms`}

## Quality Comparison

<CodeBlock language="typescript" title="quality-check.ts">
{`// Measure quality loss from optimization, like Lighthouse scores for models
async function compareQuality(
  teacher: tf.LayersModel,
  student: tf.LayersModel,
  numSamples: number = 50
): Promise<{ mse: number; similarity: number }> {
  let totalMSE = 0;

  for (let i = 0; i < numSamples; i++) {
    const z = tf.randomNormal([1, 128]);
    const teacherOut = (teacher.predict(z) as tf.Tensor).dataSync();
    const studentOut = (student.predict(z) as tf.Tensor).dataSync();

    const mse = Array.from(teacherOut).reduce((sum, v, j) =>
      sum + Math.pow(v - studentOut[j], 2), 0) / teacherOut.length;
    totalMSE += mse;
    z.dispose();
  }

  return {
    mse: totalMSE / numSamples,
    similarity: Math.exp(-totalMSE / numSamples), // 1.0 = identical, decays as error grows
  };
}

const quality = await compareQuality(teacher, student);
console.log('MSE:', quality.mse.toFixed(6));                             // lower is better
console.log('Similarity:', (quality.similarity * 100).toFixed(1) + '%'); // higher is better`}
</CodeBlock>
