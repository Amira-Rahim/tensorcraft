<Dialogue speaker="Samir">
CLIP is two encoders trained together. The image encoder sees pixels. The text encoder reads words. Contrastive loss pulls matching pairs together. After training on 400 million pairs, the shared space understands concepts — not just pixels and tokens.
</Dialogue>

<Dialogue speaker="Rohan">
It is like Algolia multi-index search. You type a query and get results from images, articles, and products — all ranked in the same relevance space. CLIP does the same thing: text and images, ranked by conceptual similarity.
</Dialogue>

Algolia lets you search across different content types with a single query because they share a ranking space. CLIP creates the same unified search across modalities — given text, find matching images; given an image, find matching text.

{`import * as tf from '@tensorflow/tfjs';

class CLIPModel {
  imageEncoder: tf.LayersModel;
  textEncoder: tf.LayersModel;
  imageProjection: tf.LayersModel;
  textProjection: tf.LayersModel;
  temperature: tf.Variable;

  constructor(embeddingDim: number = 512) {
    // Image encoder: CNN-based feature extractor
    this.imageEncoder = this.buildImageEncoder(embeddingDim);
    // Text encoder: embedding + pooling
    this.textEncoder = this.buildTextEncoder(embeddingDim);
    // Projection heads: map to shared space
    this.imageProjection = this.buildProjection(embeddingDim, embeddingDim);
    this.textProjection = this.buildProjection(embeddingDim, embeddingDim);
    // Learnable temperature
    this.temperature = tf.variable(tf.scalar(0.07));
  }

  private buildImageEncoder(dim: number): tf.LayersModel {
    const input = tf.input({ shape: [224, 224, 3] });
    let x = tf.layers.conv2d({
      filters: 64, kernelSize: 7, strides: 2, padding: 'same', activation: 'relu'
    }).apply(input) as tf.SymbolicTensor;
    x = tf.layers.maxPooling2d({ poolSize: 2 }).apply(x) as tf.SymbolicTensor;
    x = tf.layers.conv2d({
      filters: 128, kernelSize: 3, strides: 2, padding: 'same', activation: 'relu'
    }).apply(x) as tf.SymbolicTensor;
    x = tf.layers.globalAveragePooling2d({}).apply(x) as tf.SymbolicTensor;
    const out = tf.layers.dense({ units: dim }).apply(x) as tf.SymbolicTensor;
    return tf.model({ inputs: input, outputs: out });
  }

  private buildTextEncoder(dim: number): tf.LayersModel {
    const input = tf.input({ shape: [77] }); // CLIP uses 77 token context
    let x = tf.layers.embedding({
      inputDim: 49408, outputDim: dim
    }).apply(input) as tf.SymbolicTensor;
    x = tf.layers.globalAveragePooling1d({}).apply(x) as tf.SymbolicTensor;
    const out = tf.layers.dense({ units: dim }).apply(x) as tf.SymbolicTensor;
    return tf.model({ inputs: input, outputs: out });
  }

  private buildProjection(inputDim: number, outputDim: number): tf.LayersModel {
    const input = tf.input({ shape: [inputDim] });
    const projected = tf.layers.dense({
      units: outputDim
    }).apply(input) as tf.SymbolicTensor;
    return tf.model({ inputs: input, outputs: projected });
  }

  // Encode image to shared space
  encodeImage(image: tf.Tensor4D): tf.Tensor2D {
    const features = this.imageEncoder.predict(image) as tf.Tensor2D;
    const projected = this.imageProjection.predict(features) as tf.Tensor2D;
    return tf.div(projected, tf.norm(projected, 2, 1, true)) as tf.Tensor2D;
  }

  // Encode text to shared space
  encodeText(tokens: tf.Tensor2D): tf.Tensor2D {
    const features = this.textEncoder.predict(tokens) as tf.Tensor2D;
    const projected = this.textProjection.predict(features) as tf.Tensor2D;
    return tf.div(projected, tf.norm(projected, 2, 1, true)) as tf.Tensor2D;
  }

  // Search: find best matching image for a text query
  search(query: tf.Tensor2D, images: tf.Tensor2D): tf.Tensor1D {
    const queryEmb = this.encodeText(query);
    const similarities = tf.matMul(queryEmb, images, false, true).squeeze();
    return similarities as tf.Tensor1D;
  }
}`}
