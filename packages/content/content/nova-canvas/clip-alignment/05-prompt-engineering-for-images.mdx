It's a session brief. Tell a musician "make it moody" and you get nothing usable. Tell them "minor key, 72 BPM, tape saturation, room reverb at 1.2 seconds" and you get the take you imagined. Vague prompts produce generic output. Specific prompts produce targeted results.
</Dialogue>

Specificity is the whole skill. A vague prompt averages over everything the model has ever seen of that subject; every modifier you add (mood, lighting, palette, composition) cuts the space of possible images down toward the one you meant.

{`import * as tf from '@tensorflow/tfjs';

// CSS SPECIFICITY PARALLEL:
// div { }                          -> very broad match
// .gallery .card { }               -> moderate specificity
// .gallery > .card.featured:first  -> precise target
//
// PROMPT SPECIFICITY:
// "city"                                   -> very broad
// "cityscape at twilight"                  -> moderate
// "melancholy digital cityscape, twilight, -> precise
//  deep blues, cinematic lighting"

// Structured prompt builder (like BEM for images)
interface PromptComponents {
  subject: string;        // what: "cityscape", "portrait", "landscape"
  style?: string;         // how: "watercolor", "digital", "oil painting"
  mood?: string;          // feeling: "melancholy", "joyful", "serene"
  lighting?: string;      // light: "cinematic", "natural", "dramatic"
  colorPalette?: string;  // colors: "deep blues and amber", "monochrome"
  composition?: string;   // framing: "wide angle", "close-up", "bird's eye"
  quality?: string;       // fidelity: "highly detailed", "8k", "photorealistic"
}

function buildPrompt(components: PromptComponents): string {
  const parts: string[] = [components.subject];
  if (components.style) parts.push(components.style);
  if (components.mood) parts.push(components.mood);
  if (components.lighting) parts.push(\`\${components.lighting} lighting\`);
  if (components.colorPalette) parts.push(components.colorPalette);
  if (components.composition) parts.push(components.composition);
  if (components.quality) parts.push(components.quality);
  return parts.join(', ');
}

// Measure prompt effectiveness with CLIP
async function scorePrompt(
  clip: CLIPModel,
  prompt: string,
  generatedImage: tf.Tensor4D,
  tokenizer: (text: string) => tf.Tensor2D
): Promise<number> {
  const tokens = tokenizer(prompt);
  const textEmb = clip.encodeText(tokens);
  const imageEmb = clip.encodeImage(generatedImage);

  const similarity = tf.sum(tf.mul(
    textEmb.squeeze() as tf.Tensor1D,
    imageEmb.squeeze() as tf.Tensor1D
  ));

  return similarity.dataSync()[0];
}

// Iterate: compare prompt variations
const prompts = [
  buildPrompt({ subject: 'city at night' }),
  buildPrompt({ subject: 'cityscape', mood: 'melancholy', lighting: 'twilight' }),
  buildPrompt({
    subject: 'digital cityscape',
    mood: 'melancholy',
    lighting: 'cinematic',
    colorPalette: 'deep blues and amber',
    composition: 'wide angle',
    style: 'architectural solitude'
  }),
];
// CLIP scores: 0.18, 0.27, 0.34
// More specific = higher alignment = closer to artistic intent`}
