<Dialogue speaker="Priya Sharma">
The ensemble model is 18MB. Each edge device has 512MB total, but after the OS, monitoring agents, and sensor drivers, we have 64MB for ML. The model does not fit. We split it across 3 devices and coordinate inference.
</Dialogue>

<Dialogue speaker="Tariq Hussain">
Same idea as code splitting in a big React app. You don't ship the entire bundle to every page. You split at logical boundaries and load chunks on demand.
</Dialogue>

You use `import()` to split your JavaScript bundle into chunks loaded on demand. Model sharding applies the same principle: split a large model at layer boundaries, distribute shards across devices, and coordinate inference by piping intermediate tensors between them. Each device only holds a fraction of the model in memory.

{`// Model sharding, like dynamic import() code splitting
import * as tf from '@tensorflow/tfjs';

interface ModelShard {
  id: string;
  deviceId: string;
  layers: string[];
  memoryMB: number;
  predict: (input: tf.Tensor) => Promise<tf.Tensor>;
}

interface ShardPlan {
  totalModelSizeMB: number;
  maxShardSizeMB: number;
  shards: ModelShard[];
}

class ShardCoordinator {
  private shards: ModelShard[] = [];

  constructor(plan: ShardPlan) {
    this.shards = plan.shards;
  }

  // Split a model into shards based on memory budget
  static createPlan(
    model: tf.LayersModel,
    maxMemoryMB: number
  ): ShardPlan {
    const layers = model.layers;
    const shards: ModelShard[] = [];
    let currentShard: string[] = [];
    let currentSize = 0;

    for (const layer of layers) {
      const layerSize = layer.countParams() * 4 / (1024 * 1024); // 4 bytes per float32
      if (currentSize + layerSize > maxMemoryMB && currentShard.length > 0) {
        shards.push({
          id: \`shard-\${shards.length}\`,
          deviceId: \`device-\${shards.length}\`,
          layers: [...currentShard],
          memoryMB: currentSize,
          predict: async (input) => input, // placeholder
        });
        currentShard = [];
        currentSize = 0;
      }
      currentShard.push(layer.name);
      currentSize += layerSize;
    }

    if (currentShard.length > 0) {
      shards.push({
        id: \`shard-\${shards.length}\`,
        deviceId: \`device-\${shards.length}\`,
        layers: [...currentShard],
        memoryMB: currentSize,
        predict: async (input) => input,
      });
    }

    return {
      totalModelSizeMB: shards.reduce((s, sh) => s + sh.memoryMB, 0),
      maxShardSizeMB: maxMemoryMB,
      shards,
    };
  }

  // Run inference across shards in sequence
  async predict(input: tf.Tensor): Promise<tf.Tensor> {
    let intermediate = input;
    for (const shard of this.shards) {
      console.log(\`Running shard \${shard.id} on \${shard.deviceId}\`);
      intermediate = await shard.predict(intermediate);
    }
    return intermediate;
  }

  getMemoryReport(): { shardId: string; deviceId: string; memoryMB: number }[] {
    return this.shards.map(s => ({
      shardId: s.id,
      deviceId: s.deviceId,
      memoryMB: s.memoryMB,
    }));
  }
}

// 18MB model split into shards under 6MB each

  console.log(\`  \${s.id}: \${s.memoryMB.toFixed(1)}MB on \${s.deviceId}\`)
);`}
