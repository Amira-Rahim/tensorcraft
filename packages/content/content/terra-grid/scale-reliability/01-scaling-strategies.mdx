A 400% request spike, and latency is holding at 14ms. That number is the whole game. Nobody watching the stress test cares how many workers are running or what the CPUs are doing; they care whether predictions still arrive inside the budget. So that is what the auto-scaler targets. The machinery is the auto-scaling you have configured for web servers (min and max instances, an orchestrator doing the arithmetic), pointed at prediction latency instead of CPU utilization. Latency climbs, workers spin up. Load drops, they drain.

## Inference Auto-Scaler

<CodeBlock language="typescript" title="auto-scaler.ts">
{`// Auto-scaling inference workers — same pattern as web server scaling

interface InferenceWorker {
  id: string;
  status: 'active' | 'draining' | 'idle';
  currentLatencyMs: number;
  predictionsPerSecond: number;
}

interface ScalingPolicy {
  minWorkers: number;
  maxWorkers: number;
  targetLatencyMs: number;
  scaleUpThresholdMs: number;
  scaleDownThresholdMs: number;
  cooldownSeconds: number;
}

class InferenceAutoScaler {
  private workers: InferenceWorker[] = [];
  private lastScaleAction: number = 0;
  private policy: ScalingPolicy;

  constructor(policy: ScalingPolicy) {
    this.policy = policy;
    // Start with minimum workers
    for (let i = 0; i < policy.minWorkers; i++) {
      this.addWorker();
    }
  }

  private addWorker(): InferenceWorker {
    const worker: InferenceWorker = {
      id: \`worker-\${this.workers.length + 1}\`,
      status: 'active',
      currentLatencyMs: 0,
      predictionsPerSecond: 0,
    };
    this.workers.push(worker);
    return worker;
  }

  private removeWorker(): void {
    const idle = this.workers.find(w => w.status === 'idle');
    if (idle) {
      this.workers = this.workers.filter(w => w.id !== idle.id);
    }
  }

  evaluate(): { action: 'scale-up' | 'scale-down' | 'none'; workerCount: number } {
    const now = Date.now();
    const cooldownElapsed = (now - this.lastScaleAction) / 1000 > this.policy.cooldownSeconds;

    if (!cooldownElapsed) return { action: 'none', workerCount: this.workers.length };

    const avgLatency = this.workers.reduce((s, w) => s + w.currentLatencyMs, 0)
      / this.workers.length;

    if (avgLatency > this.policy.scaleUpThresholdMs
        && this.workers.length < this.policy.maxWorkers) {
      this.addWorker();
      this.lastScaleAction = now;
      return { action: 'scale-up', workerCount: this.workers.length };
    }

    if (avgLatency < this.policy.scaleDownThresholdMs
        && this.workers.length > this.policy.minWorkers) {
      this.removeWorker();
      this.lastScaleAction = now;
      return { action: 'scale-down', workerCount: this.workers.length };
    }

    return { action: 'none', workerCount: this.workers.length };
  }
}

// Terra Grid scaling policy
const scaler = new InferenceAutoScaler({
  minWorkers: 2,
  maxWorkers: 8,
  targetLatencyMs: 15,
  scaleUpThresholdMs: 20,
  scaleDownThresholdMs: 8,
  cooldownSeconds: 60,
});

// During stress test
const result = scaler.evaluate();
console.log(\`Action: \${result.action}, Workers: \${result.workerCount}\`);`}
</CodeBlock>
