You write JavaScript once, but V8, SpiderMonkey, and JavaScriptCore each compile and execute it differently. Each engine has different JIT strategies, different garbage collection, different performance characteristics. ML runtimes work the same way: one model, multiple execution engines, each optimized for different hardware.

type RuntimeTarget = 'tflite' | 'onnx' | 'wasm' | 'tfjs';

interface RuntimeConfig {
  target: RuntimeTarget;
  format: string;
  optimizations: string[];
  avgInferenceMs: number;

// Like selecting the right polyfill bundle per browser
const runtimeMatrix: Record<string, RuntimeConfig> = {
  substation: {
    target: 'onnx',
    format: 'model.onnx',
    optimizations: ['graph-optimization', 'operator-fusion'],
    avgInferenceMs: 8,
  },
  sensor: {
    target: 'tflite',
    format: 'model.tflite',
    optimizations: ['neon-simd', 'xnnpack-delegate'],
    avgInferenceMs: 15,
  },
  remote: {
    target: 'wasm',
    format: 'model_wasm.bin',
    optimizations: ['simd128', 'threads'],
    avgInferenceMs: 45,
  },
  dashboard: {
    target: 'tfjs',
    format: 'model.json',
    optimizations: ['webgl-backend', 'shader-compilation'],
    avgInferenceMs: 22,
  },
};

function selectRuntime(deviceType: string): RuntimeConfig {
  const config = runtimeMatrix[deviceType];
  if (!config) throw new Error(\`No runtime configured for \${deviceType}\`);
  console.log(\`\${deviceType} → \${config.target} (\${config.avgInferenceMs}ms)\`);
  return config;
}`}

## Runtime Abstraction

Just as a framework-agnostic component library works across React and Vue, you can build a runtime abstraction that works across ML engines:

<CodeBlock language="typescript" title="runtime-abstraction.ts">
{`interface InferenceEngine {
  load(modelPath: string): Promise<void>;
  predict(input: Float32Array): Promise<Float32Array>;
  dispose(): void;
}

class TFJSEngine implements InferenceEngine {
  private model: tf.LayersModel | null = null;

  async load(modelPath: string) {
    this.model = await tf.loadLayersModel(modelPath);
  }

  async predict(input: Float32Array): Promise<Float32Array> {
    return tf.tidy(() => {
      const tensor = tf.tensor2d(input, [1, input.length]);
      const result = this.model!.predict(tensor) as tf.Tensor;
      return new Float32Array(result.dataSync());
    });
  }

  dispose() { this.model?.dispose(); }
}

// Same interface, different engine, like a polyfill
class WASMEngine implements InferenceEngine {
  private session: any = null;

  async load(modelPath: string) {
    const wasmModule = await WebAssembly.instantiateStreaming(fetch(modelPath));
    this.session = wasmModule.instance.exports;
  }

  async predict(input: Float32Array): Promise<Float32Array> {
    const ptr = this.session.allocate(input.length * 4);
    new Float32Array(this.session.memory.buffer, ptr, input.length).set(input);
    this.session.inference(ptr, input.length);
    return new Float32Array(this.session.memory.buffer, ptr, 3); // 3 output classes
  }

  dispose() { this.session = null; }
}`}
</CodeBlock>
