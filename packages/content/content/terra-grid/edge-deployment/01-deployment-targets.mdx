## The Hardware Landscape

<BridgeCallout bridge={frontmatter.concept.bridge} />

In frontend development, you use feature detection to serve the right experience. Edge deployment uses the same pattern — but instead of checking for `IntersectionObserver`, you check for available RAM and compute.

<CodeBlock language="typescript" title="device-profiling.ts">
{`import * as tf from '@tensorflow/tfjs';

interface DeviceProfile {
  tier: 'high' | 'medium' | 'low';
  ram: number;        // MB
  cpuCores: number;
  hasGPU: boolean;
  connectivity: 'wired' | 'cellular' | 'satellite';
}

function profileDevice(specs: DeviceProfile): string {
  // Like choosing a bundle size based on connection speed
  if (specs.ram >= 2048 && specs.hasGPU) {
    return 'models/terra-grid-full.bin';      // 12MB, full precision
  } else if (specs.ram >= 512) {
    return 'models/terra-grid-medium.bin';    // 4MB, int8 quantized
  } else {
    return 'models/terra-grid-lite.bin';      // 1.2MB, pruned + quantized
  }
}

// Sector 7 deployment matrix
const fleet: DeviceProfile[] = [
  { tier: 'high', ram: 4096, cpuCores: 4, hasGPU: true, connectivity: 'wired' },
  { tier: 'medium', ram: 512, cpuCores: 2, hasGPU: false, connectivity: 'cellular' },
  { tier: 'low', ram: 256, cpuCores: 1, hasGPU: false, connectivity: 'satellite' },
];

fleet.forEach(device => {
  const modelPath = profileDevice(device);
  console.log(\`\${device.tier} tier → \${modelPath}\`);
});`}
</CodeBlock>

## Constraint Budgets

Every edge device has a budget, the same way every browser tab has a performance budget:

<CodeBlock language="typescript" title="constraint-budget.ts">
{`interface DeploymentBudget {
  maxModelSizeMB: number;     // Like max bundle size
  maxInferenceMs: number;     // Like time-to-interactive
  maxMemoryMB: number;        // Like JS heap limit
  updateBandwidthMB: number;  // Like data-saver mode
}

const budgets: Record<string, DeploymentBudget> = {
  substation: { maxModelSizeMB: 12, maxInferenceMs: 50, maxMemoryMB: 2048, updateBandwidthMB: 50 },
  sensor:     { maxModelSizeMB: 4,  maxInferenceMs: 100, maxMemoryMB: 256, updateBandwidthMB: 10 },
  remote:     { maxModelSizeMB: 1.5, maxInferenceMs: 200, maxMemoryMB: 128, updateBandwidthMB: 2 },
};

function validateDeployment(modelSizeMB: number, target: string): boolean {
  const budget = budgets[target];
  if (modelSizeMB > budget.maxModelSizeMB) {
    console.error(\`Model \${modelSizeMB}MB exceeds \${target} budget of \${budget.maxModelSizeMB}MB\`);
    return false;
  }
  return true;
}`}
</CodeBlock>
