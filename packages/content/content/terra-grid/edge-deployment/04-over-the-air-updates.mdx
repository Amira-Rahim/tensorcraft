You have seen this pattern before. A Service Worker installs a new version in the background while the current version keeps serving requests. When the new version is verified and ready, it activates on the next navigation. That is the right transport model for OTA: download, verify, stage, activate, zero downtime. It is the wrong risk model. New code behaves identically on every page load; new weights can shift the prediction distribution the moment they activate. So this pipeline adds what Service Workers never needed: staged rollout, shadow inference running old and new side by side, and an accuracy gate against a holdout set before the rest of the fleet sees the update.

## The Update Lifecycle

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="ota-lifecycle.ts">
{`import * as tf from '@tensorflow/tfjs';

type UpdateState = 'idle' | 'downloading' | 'verifying' | 'staging' | 'activating' | 'active' | 'failed';

interface OTAManager {
  currentVersion: string;
  stagedVersion: string | null;
  state: UpdateState;
}

async function performOTAUpdate(
  manager: OTAManager,
  updateUrl: string,
  expectedChecksum: string,
): Promise<boolean> {
  // Phase 1: Download (like SW install event)
  manager.state = 'downloading';
  const packageData = await fetch(updateUrl).then(r => r.arrayBuffer());

  // Phase 2: Verify integrity (like SRI check)
  manager.state = 'verifying';
  const checksum = await computeChecksum(packageData);
  if (checksum !== expectedChecksum) {
    manager.state = 'failed';
    console.error('Checksum mismatch — update rejected');
    return false;
  }

  // Phase 3: Stage (like SW waiting state)
  manager.state = 'staging';
  await writeToStorage('staged-model', packageData);

  // Phase 4: Validate with test inference
  const stagedModel = await tf.loadLayersModel('indexeddb://staged-model');
  const testInput = tf.randomNormal([1, 24]);
  const testOutput = stagedModel.predict(testInput) as tf.Tensor;
  const outputValid = testOutput.shape[1] === 3;
  testInput.dispose();
  testOutput.dispose();
  stagedModel.dispose();

  if (!outputValid) {
    manager.state = 'failed';
    await deleteFromStorage('staged-model');
    return false;
  }

  // Phase 5: Activate (like SW activate event)
  manager.state = 'activating';
  await copyStorage('current-model', 'rollback-model');  // backup
  await copyStorage('staged-model', 'current-model');    // atomic swap
  manager.state = 'active';

  console.log('OTA update complete — new model active');
  return true;
}`}
</CodeBlock>

## Delta Updates

Sending a full model over a $0.50/MB satellite link is expensive. Delta updates send only what changed, like git sending diffs instead of full snapshots:

<CodeBlock language="typescript" title="delta-update.ts">
{`interface DeltaUpdate {
  fromVersion: string;
  toVersion: string;
  changedLayers: string[];
  delta: ArrayBuffer;    // only the changed weights
  fullSizeMB: number;
  deltaSizeMB: number;
}

function applyDelta(currentWeights: ArrayBuffer, delta: DeltaUpdate): ArrayBuffer {
  // Like applying a git patch — merge changes into existing weights
  const current = new Float32Array(currentWeights);
  const patch = new Float32Array(delta.delta);

  // Apply weight updates at specified offsets
  const result = new Float32Array(current);
  let offset = 0;
  for (const layerName of delta.changedLayers) {
    const layerSize = getLayerSize(layerName);
    const layerOffset = getLayerOffset(layerName);
    result.set(patch.slice(offset, offset + layerSize), layerOffset);
    offset += layerSize;
  }

  console.log(\`Delta update: \${delta.deltaSizeMB}MB vs \${delta.fullSizeMB}MB full\`);
  // Typical savings: 60-80% bandwidth reduction
  return result.buffer;
}`}
</CodeBlock>
