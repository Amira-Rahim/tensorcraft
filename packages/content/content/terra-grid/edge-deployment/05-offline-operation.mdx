## Offline Model Cache

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="offline-cache.ts">
{`import * as tf from '@tensorflow/tfjs';

class OfflineModelCache {
  private modelKey = 'indexeddb://terra-grid-model';
  private model: tf.LayersModel | null = null;

  async initialize(): Promise<void> {
    try {
      // Try loading from local cache first (like SW cache-first strategy)
      this.model = await tf.loadLayersModel(this.modelKey);
      console.log('Model loaded from local cache');
    } catch {
      // Cache miss — download and store
      this.model = await tf.loadLayersModel('/api/models/terra-grid/model.json');
      await this.model.save(this.modelKey);
      console.log('Model downloaded and cached locally');
    }
  }

  async predict(sensorData: number[]): Promise<Float32Array> {
    if (!this.model) throw new Error('Model not initialized');
    return tf.tidy(() => {
      const input = tf.tensor2d([sensorData]);
      const output = this.model!.predict(input) as tf.Tensor;
      return new Float32Array(output.dataSync());
    });
  }
}`}
</CodeBlock>

## Prediction Buffer

When offline, predictions queue in a ring buffer — a fixed-size circular array that overwrites the oldest entries when full, preventing memory from growing unbounded on a 256MB device:

<CodeBlock language="typescript" title="prediction-buffer.ts">
{`interface BufferedPrediction {
  timestamp: number;
  sensorId: string;
  input: number[];
  output: number[];
  synced: boolean;
}

class PredictionRingBuffer {
  private buffer: BufferedPrediction[];
  private head = 0;
  private count = 0;

  constructor(private capacity: number = 10000) {
    this.buffer = new Array(capacity);
  }

  push(prediction: BufferedPrediction): void {
    this.buffer[this.head] = prediction;
    this.head = (this.head + 1) % this.capacity;
    this.count = Math.min(this.count + 1, this.capacity);
  }

  getUnsynced(): BufferedPrediction[] {
    return this.buffer
      .filter(p => p && !p.synced)
      .sort((a, b) => a.timestamp - b.timestamp);
  }

  async syncToServer(endpoint: string): Promise<number> {
    const unsynced = this.getUnsynced();
    if (unsynced.length === 0) return 0;

    // Batch upload — like Background Sync API
    const response = await fetch(endpoint, {
      method: 'POST',
      body: JSON.stringify({ predictions: unsynced }),
    });

    if (response.ok) {
      unsynced.forEach(p => p.synced = true);
      console.log(\`Synced \${unsynced.length} buffered predictions\`);
    }
    return unsynced.length;
  }
}

// Usage during the 14-hour outage:
const buffer = new PredictionRingBuffer(50000);
const cache = new OfflineModelCache();
await cache.initialize();

// Continuous inference loop — runs regardless of connectivity
setInterval(async () => {
  const sensorData = await readSensors();
  const prediction = await cache.predict(sensorData);
  buffer.push({
    timestamp: Date.now(),
    sensorId: 'station-7-delta',
    input: sensorData,
    output: Array.from(prediction),
    synced: false,
  });
}, 5000); // every 5 seconds`}
</CodeBlock>
