GitHub Actions triggers on `push`, `pull_request`, `schedule`, or `workflow_dispatch`. You define the event, the conditions, and the workflow runs automatically. Automated retraining uses the same pattern — define trigger events (data change, drift detected, schedule), and the training pipeline runs when conditions are met.

<CodeBlock language="typescript" title="retraining-triggers.ts">
{`type TriggerType = 'data-drift' | 'dataset-growth' | 'schedule' | 'performance-drop' | 'manual';

interface RetrainingTrigger {
  type: TriggerType;
  condition: () => Promise<boolean>;
  cooldownMinutes: number;    // prevent trigger storms
  lastTriggered: number | null;
}

const triggers: RetrainingTrigger[] = [
  {
    // Like: on: push: branches: [main]
    type: 'data-drift',
    condition: async () => {
      const driftScore = await measureDrift('sector-all');
      return driftScore > 0.3;
    },
    cooldownMinutes: 360,  // max once per 6 hours
    lastTriggered: null,
  },
  {
    // Like: on: schedule: cron: '0 2 * * 1'
    type: 'schedule',
    condition: async () => {
      const dayOfWeek = new Date().getDay();
      const hour = new Date().getHours();
      return dayOfWeek === 1 && hour === 2;  // Monday 2 AM
    },
    cooldownMinutes: 1440,  // once per day max
    lastTriggered: null,
  },
  {
    // Like: on: check_suite: completed: failure
    type: 'performance-drop',
    condition: async () => {
      const currentAccuracy = await getProductionAccuracy();
      return currentAccuracy < 0.93;
    },
    cooldownMinutes: 720,  // max once per 12 hours
    lastTriggered: null,
  },
];

async function evaluateTriggers(triggers: RetrainingTrigger[]): Promise<TriggerType | null> {
  for (const trigger of triggers) {
    // Check cooldown — prevent runaway retraining
    if (trigger.lastTriggered) {
      const elapsed = (Date.now() - trigger.lastTriggered) / 60000;
      if (elapsed < trigger.cooldownMinutes) continue;
    }

    if (await trigger.condition()) {
      trigger.lastTriggered = Date.now();
      console.log(\`Trigger fired: \${trigger.type}\`);
      return trigger.type;
    }
  }
  return null;
}`}
</CodeBlock>

## The Retraining Loop

<CodeBlock language="typescript" title="retraining-loop.ts">
{`async function retrainingLoop(
  triggers: RetrainingTrigger[],
  pipeline: MLPipeline,
  registry: ModelRegistry,
): Promise<void> {
  // Check triggers every 15 minutes
  const checkInterval = 15 * 60 * 1000;

  setInterval(async () => {
    const firedTrigger = await evaluateTriggers(triggers);
    if (!firedTrigger) return;

    console.log(\`Retraining triggered by: \${firedTrigger}\`);

    // Run the full pipeline
    const success = await runPipeline(pipeline);

    if (success) {
      // Register the new model — but do NOT auto-promote to production
      // Hassan's rule: automated does not mean unsupervised
      await registry.register({
        name: 'terra-grid-intelligence',
        version: incrementVersion(),
        stage: 'staging',     // human must promote to production
        // ... metrics, lineage
      });

      await notifyTeam({
        message: \`New model ready for review (trigger: \${firedTrigger})\`,
        channel: '#ml-pipeline',
      });
    } else {
      await notifyTeam({
        message: \`Retraining failed (trigger: \${firedTrigger}). Check pipeline logs.\`,
        channel: '#ml-alerts',
        severity: 'warning',
      });
    }
  }, checkInterval);
}`}
</CodeBlock>
