<Dialogue speaker="Priya Sharma">
I have 31 trained models. Which one is in production? Which one should Tariq deploy next? Without a registry, nobody knows.
</Dialogue>

<Dialogue speaker="Tariq Hussain">
I need one command: give me the latest approved model. Not the latest trained model. The latest approved model. Those are different things.
</Dialogue>

You do not deploy npm packages by copying files around. You publish to a registry with a version, metadata, and dist-tags like `latest`, `next`, and `beta`. Tariq's one command exists because that design is worth stealing, minus one assumption: npm's `latest` is chronological, and for models chronology means nothing. The newest of Priya's 31 models is frequently not the best one. A model registry's `production` tag is earned through evaluation metrics and explicit approval, which is why it also tracks training metadata and per-version eval results that npm never needed.

## The Model Registry

<CodeBlock language="typescript" title="model-registry.ts">
{`type ModelStage = 'development' | 'staging' | 'production' | 'archived';

interface RegisteredModel {
  name: string;
  version: string;
  stage: ModelStage;          // like npm dist-tags
  artifactPath: string;
  metrics: {
    accuracy: number;
    f1Score: number;
    inferenceMs: number;
    modelSizeMB: number;
  };
  dataVersion: string;        // which dataset trained it
  experimentId: string;       // which experiment produced it
  registeredAt: string;
  promotedBy: string | null;  // who approved the promotion
}

class ModelRegistry {
  private models: RegisteredModel[] = [];

  register(model: RegisteredModel): void {
    this.models.push(model);
    console.log(\`Registered: \${model.name}@\${model.version} [\${model.stage}]\`);
  }

  promote(name: string, version: string, toStage: ModelStage, approver: string): void {
    const model = this.models.find(m => m.name === name && m.version === version);
    if (!model) throw new Error(\`Model \${name}@\${version} not found\`);

    // Enforce promotion order: dev → staging → production
    const order: ModelStage[] = ['development', 'staging', 'production'];
    const currentIdx = order.indexOf(model.stage);
    const targetIdx = order.indexOf(toStage);

    if (targetIdx !== currentIdx + 1 && toStage !== 'archived') {
      throw new Error(\`Cannot promote from \${model.stage} to \${toStage}\`);
    }

    // Demote current production model when promoting a new one
    if (toStage === 'production') {
      const current = this.getByStage(name, 'production');
      if (current) current.stage = 'archived';
    }

    model.stage = toStage;
    model.promotedBy = approver;
    console.log(\`Promoted \${name}@\${version} → \${toStage} by \${approver}\`);
  }

  getByStage(name: string, stage: ModelStage): RegisteredModel | undefined {
    return this.models.find(m => m.name === name && m.stage === stage);
  }

  // Tariq's one command: "give me the production model"
  getProductionModel(name: string): RegisteredModel {
    const model = this.getByStage(name, 'production');
    if (!model) throw new Error(\`No production model for \${name}\`);
    return model;
  }
}`}
</CodeBlock>

## Registry in Action

<CodeBlock language="typescript" title="registry-usage.ts">
{`const registry = new ModelRegistry();

// Priya registers her best experiment
registry.register({
  name: 'terra-grid-intelligence',
  version: '2.3.0',
  stage: 'development',
  artifactPath: 's3://models/tgi-2.3.0.bin',
  metrics: { accuracy: 0.961, f1Score: 0.948, inferenceMs: 12, modelSizeMB: 4.2 },
  dataVersion: 'a3f8e2',
  experimentId: 'run-31',
  registeredAt: new Date().toISOString(),
  promotedBy: null,
});

// Hassan reviews and promotes to staging
registry.promote('terra-grid-intelligence', '2.3.0', 'staging', 'hassan');

// After staging tests pass, promote to production
registry.promote('terra-grid-intelligence', '2.3.0', 'production', 'hassan');

// Tariq deploys the production model
const model = registry.getProductionModel('terra-grid-intelligence');
console.log(\`Deploying: \${model.name}@\${model.version}\`);
// Deploying: terra-grid-intelligence@2.3.0`}
</CodeBlock>
