Somewhere in this week's 31 training runs is a model that hit 96.1% accuracy. Nobody can say which one.

<Dialogue speaker="Priya Sharma">
I watched that number print. But I can't tell you the learning rate, the batch size, or which data version produced it. I have to rerun them all.
</Dialogue>

<Dialogue speaker="Hassan Al-Rashid">
That is 31 GPU-hours spent buying answers we then threw away. Logging a run costs one line of code; rerunning one costs a GPU-hour and a day of queue time. We do not repeat experiments we cannot remember.
</Dialogue>

A CI dashboard would never permit the equivalent: every test run records what ran, how long it took, and what coverage it hit, and Tuesday's run sits one click from Friday's. Experiment tracking gives training runs the same memory.

{`interface ExperimentRun {
  id: string;
  timestamp: string;
  params: {
    learningRate: number;
    batchSize: number;
    epochs: number;
    optimizer: string;
    architecture: string;
  };
  metrics: {
    accuracy: number;
    loss: number;
    f1Score: number;
    inferenceMs: number;
  };
  dataVersion: string;     // hash of training dataset
  codeVersion: string;     // git commit SHA
  durationSeconds: number;
  status: 'completed' | 'failed' | 'stopped';
}

class ExperimentTracker {
  private runs: ExperimentRun[] = [];

  logRun(run: ExperimentRun): void {
    this.runs.push(run);
    console.log(
      \`Run \${run.id}: accuracy=\${run.metrics.accuracy.toFixed(3)}, \` +
      \`loss=\${run.metrics.loss.toFixed(4)}, lr=\${run.params.learningRate}\`
    );
  }

  // Two of these four metrics are better when they are smaller
  private static readonly higherIsBetter: Record<keyof ExperimentRun['metrics'], boolean> = {
    accuracy: true,
    loss: false,
    f1Score: true,
    inferenceMs: false,
  };

  getBestRun(metric: keyof ExperimentRun['metrics']): ExperimentRun | null {
    const completed = this.runs.filter(r => r.status === 'completed');
    if (completed.length === 0) return null;
    const wantsHigher = ExperimentTracker.higherIsBetter[metric];
    return completed.reduce((best, run) => {
      const beatsBest = wantsHigher
        ? run.metrics[metric] > best.metrics[metric]
        : run.metrics[metric] < best.metrics[metric];
      return beatsBest ? run : best;
    });
  }

  compareRuns(idA: string, idB: string): Record<string, any> {
    const a = this.runs.find(r => r.id === idA);
    const b = this.runs.find(r => r.id === idB);
    if (!a || !b) throw new Error('Run not found');
    return {
      accuracyDiff: b.metrics.accuracy - a.metrics.accuracy,
      lossDiff: b.metrics.loss - a.metrics.loss,
      paramChanges: Object.entries(a.params)
        .filter(([k, v]) => b.params[k as keyof typeof b.params] !== v)
        .map(([k, v]) => \`\${k}: \${v} → \${b.params[k as keyof typeof b.params]}\`),
    };
  }
}`}
</CodeBlock>

## Tracking a Training Run

<CodeBlock language="typescript" title="tracked-training.ts">
{`import * as tf from '@tensorflow/tfjs';

async function trackedTrainingRun(
  tracker: ExperimentTracker,
  data: { train: tf.Tensor; labels: tf.Tensor; test: tf.Tensor; testLabels: tf.Tensor },
  params: ExperimentRun['params'],
  dataVersion: string,
): Promise<ExperimentRun> {
  const runId = \`run-\${Date.now()}\`;
  const startTime = Date.now();

  const model = tf.sequential();
  model.add(tf.layers.dense({ units: 64, activation: 'relu', inputShape: [24] }));
  model.add(tf.layers.dense({ units: 32, activation: 'relu' }));
  model.add(tf.layers.dense({ units: 3, activation: 'softmax' }));

  model.compile({
    optimizer: tf.train.adam(params.learningRate),
    loss: 'categoricalCrossentropy',
    metrics: ['accuracy'],
  });

  const history = await model.fit(data.train, data.labels, {
    epochs: params.epochs,
    batchSize: params.batchSize,
    validationSplit: 0.2,
  });

  // The per-epoch curve is the most useful artifact fit() hands back:
  // log it now or rerun the whole thing to get it later
  const lossCurve = (history.history.loss as number[]).map(l => l.toFixed(3));
  console.log(\`\${runId} loss curve: \${lossCurve.join(' → ')}\`);

  const evalResult = model.evaluate(data.test, data.testLabels) as tf.Scalar[];

  const run: ExperimentRun = {
    id: runId,
    timestamp: new Date().toISOString(),
    params,
    metrics: {
      accuracy: evalResult[1].dataSync()[0],
      loss: evalResult[0].dataSync()[0],
      f1Score: 0,  // computed separately
      inferenceMs: await benchmarkInference(model),
    },
    dataVersion,
    codeVersion: 'a3f8e2c',
    durationSeconds: (Date.now() - startTime) / 1000,
    status: 'completed',
  };

  tracker.logRun(run);
  return run;
}`}
