```text
$ git log --oneline training-data/
fatal: not a git repository
```

That command is the whole problem. The code that trained last Tuesday's regressed model has full history; the data it trained on has none.

<Dialogue speaker="Priya Sharma">
The model regressed last Tuesday. But I can't tell you which training data produced it. Was it the batch before or after the Sector 3 sensor recalibration? Nobody versioned the data.
</Dialogue>

<Dialogue speaker="Hassan Al-Rashid">
If you cannot reproduce the training run, you cannot debug the regression. Version everything. Code, data, hyperparameters.
</Dialogue>

No history means no blame, no bisect, no diff. The model is provably wrong and the why is untraceable, because nobody tracked what data trained it. Data versioning closes that gap with the same primitive git uses: a content hash for every state worth returning to.

{`import { createHash } from 'crypto';

interface DatasetVersion {
  hash: string;              // content hash — like a git commit SHA
  version: string;           // semantic version
  parentHash: string | null; // previous version — like git parent
  timestamp: string;
  rowCount: number;
  schema: string;            // schema version
  source: string;            // where the data came from
  changelog: string;         // what changed — like a commit message
}

function hashDataset(data: Float32Array): string {
  return createHash('sha256')
    .update(Buffer.from(data.buffer))
    .digest('hex')
    .slice(0, 12);  // short hash, like git's abbreviated SHA
}

function createDatasetVersion(
  data: Float32Array,
  parentHash: string | null,
  changelog: string,
): DatasetVersion {
  const hash = hashDataset(data);
  return {
    hash,
    version: parentHash ? incrementVersion(parentHash) : '1.0.0',
    parentHash,
    timestamp: new Date().toISOString(),
    rowCount: data.length / 24,  // 24 features per row
    schema: '2.1',
    source: 'sector-7-sensors',
    changelog,
  };
}

// Versioning in action:
const v1 = createDatasetVersion(batch1, null, 'Initial Sector 7 training data');
const v2 = createDatasetVersion(batch2, v1.hash, 'Added Sector 3 sensor data');
const v3 = createDatasetVersion(batch3, v2.hash, 'Removed corrupted Sector 3 batch');

// Now we can trace: model v2.1 was trained on dataset a3f8e2 (v2)
// which included the corrupted Sector 3 batch — that is the regression source`}
</CodeBlock>

## Dataset Diff

{`interface DatasetDiff {
  addedRows: number;
  removedRows: number;
  modifiedRows: number;
  schemaChanges: string[];
  distributionShift: Record<string, number>;  // feature → KL divergence
}

function diffDatasets(
  oldData: Float32Array,
  newData: Float32Array,
  featureCount: number,
): DatasetDiff {
  const oldRows = oldData.length / featureCount;
  const newRows = newData.length / featureCount;

  // Compute per-feature distribution shift
  const shift: Record<string, number> = {};
  for (let f = 0; f < featureCount; f++) {
    const oldFeature = extractColumn(oldData, f, featureCount);
    const newFeature = extractColumn(newData, f, featureCount);
    shift[\`feature_\${f}\`] = klDivergence(oldFeature, newFeature);
  }

  return {
    addedRows: Math.max(0, newRows - oldRows),
    removedRows: Math.max(0, oldRows - newRows),
    modifiedRows: 0, // simplified
    schemaChanges: [],
    distributionShift: shift,
  };
}

// Like git diff but for data:
// +1,200 rows from Sector 3 (added)
// feature_7 distribution shift: 0.42 (significant!)
// feature_12 distribution shift: 0.01 (negligible)`}
</CodeBlock>
