Manual retraining doesn't scale. We have 12 sectors generating new sensor data every hour. If I have to manually kick off training every time the data changes, we're already behind.
</Dialogue>

<Dialogue speaker="Hassan Al-Rashid">
Automate it. But with human oversight at the gates. Automated does not mean unsupervised.
</Dialogue>

`git push`, and the machinery wakes up: lint, test, build, deploy, each stage gating the next. A red test means the build never ships, and nobody has to remember to check. Now swap the trigger. New sensor data lands instead of new code, training replaces the build step, evaluation replaces the test suite. The gates work the same way: a model that fails evaluation never reaches a device.

{`type StageStatus = 'pending' | 'running' | 'passed' | 'failed' | 'skipped';

interface PipelineStage {
  name: string;
  status: StageStatus;
  gate: (prior: Record<string, any>) => Promise<boolean>;    // must pass to proceed
  execute: (prior: Record<string, any>) => Promise<any>;
}

interface MLPipeline {
  stages: PipelineStage[];
  triggerSource: 'schedule' | 'data-change' | 'manual';
}

// The pipeline below, stage by stage, with its CI/CD counterpart:
// ingest (checkout) → validate (lint) → train (build) → evaluate (test) → deploy
const terraGridPipeline: MLPipeline = {
  triggerSource: 'data-change',
  stages: [
    {
      name: 'data-ingestion',
      status: 'pending',
      gate: async () => true,  // always allowed to start
      execute: async () => {
        const newData = await fetchSectorData('all');
        return { rows: newData, rowCount: newData.length, sectors: 12 };
      },
    },
    {
      name: 'data-validation',       // ← like lint
      status: 'pending',
      gate: async () => true,
      execute: async (prior) => {
        const stats = await validateSchema(prior['data-ingestion'].rows);
        const driftScore = await checkDistribution(prior['data-ingestion'].rows, baseline);
        return { valid: stats.errors === 0, driftScore };
      },
    },
    {
      name: 'training',              // ← like build
      status: 'pending',
      gate: async (prior) => prior['data-validation'].valid && prior['data-validation'].driftScore < 0.3,
      execute: async (prior) => {
        const model = await trainModel(prior['data-ingestion'].rows, hyperparams);
        return { model, epochs: 50, finalLoss: 0.023 };
      },
    },
    {
      name: 'evaluation',            // ← like test
      status: 'pending',
      gate: async (prior) => prior['training'].finalLoss < 0.05,
      execute: async (prior) => {
        const metrics = await evaluateModel(prior['training'].model, testSet);
        return { accuracy: metrics.accuracy, f1: metrics.f1 };
      },
    },
    {
      name: 'deployment',            // ← like deploy
      status: 'pending',
      gate: async (prior) => prior['evaluation'].accuracy > 0.95 && prior['evaluation'].f1 > 0.92,
      execute: async (prior) => {
        const pkg = await packageModel(prior['training'].model, version);
        await deployToFleet(pkg, 'sector-7');
        return { devicesUpdated: 847 };
      },
    },
  ],
};`}

## Pipeline Orchestration

<CodeBlock language="typescript" title="orchestrator.ts">
{`async function runPipeline(pipeline: MLPipeline): Promise<boolean> {
  console.log(\`Pipeline triggered by: \${pipeline.triggerSource}\`);

  // Each stage reads what the stages before it produced, the way a CI job
  // reads the previous job's artifacts
  const prior: Record<string, any> = {};

  for (const stage of pipeline.stages) {
    // Check gate, like a CI check that must pass
    const gateOpen = await stage.gate(prior);
    if (!gateOpen) {
      stage.status = 'failed';
      console.error(\`Gate blocked at stage: \${stage.name}\`);
      await notifyTeam(\`Pipeline blocked: \${stage.name} gate failed\`);
      return false;
    }

    stage.status = 'running';
    console.log(\`Running: \${stage.name}\`);

    try {
      prior[stage.name] = await stage.execute(prior);
      stage.status = 'passed';
      console.log(\`Passed: \${stage.name}\`, prior[stage.name]);
    } catch (error) {
      stage.status = 'failed';
      console.error(\`Failed: \${stage.name}\`, error);
      return false;
    }
  }

  console.log('Pipeline complete: model deployed');
  return true;
}`}
</CodeBlock>
