The discipline is the OWASP audit you know from web applications: check for XSS, verify CSRF tokens, enforce CSP headers, sanitize inputs. ML security hardening follows the same checklist approach with a different threat list: model tampering, adversarial inputs, data poisoning, unauthorized updates.

## Security Audit

<CodeBlock language="typescript" title="security-audit.ts">
{`// ML security audit, OWASP-style checklist for edge ML

interface SecurityCheck {
  id: string;
  category: 'model-integrity' | 'input-validation' | 'update-auth' | 'data-protection';
  name: string;
  verify: () => Promise<boolean>;
}

class SecurityAuditor {
  private checks: SecurityCheck[] = [];

  addCheck(check: SecurityCheck): void {
    this.checks.push(check);
  }

  async runAudit(): Promise<{
    passed: number; failed: number; results: { id: string; name: string; passed: boolean }[];
  }> {
    const results = [];
    for (const check of this.checks) {
      const passed = await check.verify();
      results.push({ id: check.id, name: check.name, passed });
    }
    return {
      passed: results.filter(r => r.passed).length,
      failed: results.filter(r => !r.passed).length,
      results,
    };
  }
}

// Model integrity: verify model files have not been tampered with
async function verifyModelIntegrity(
  modelBuffer: ArrayBuffer,
  expectedHash: string
): Promise<boolean> {
  const hashBuffer = await crypto.subtle.digest('SHA-256', modelBuffer);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  const hash = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
  return hash === expectedHash;
}

// Input validation: reject readings outside physical bounds or with an impossible jump between samples.
// This is the floor, not adversarial detection: a slow in-bounds ramp passes both tests.
function validateSensorInput(readings: Float32Array, bounds: {
  min: number; max: number; maxRateOfChange: number;
}): { valid: boolean; anomalies: number[] } {
  const anomalies: number[] = [];
  for (let i = 0; i < readings.length; i++) {
    if (readings[i] < bounds.min || readings[i] > bounds.max) {
      anomalies.push(i);
    }
    if (i > 0 && Math.abs(readings[i] - readings[i - 1]) > bounds.maxRateOfChange) {
      anomalies.push(i);
    }
  }
  return { valid: anomalies.length === 0, anomalies };
}

// Web Crypto (crypto.subtle) is global in browsers and in Node 19+.
// These two come from the deploy pipeline: the model file and the hash recorded at build time.
const modelBuffer = await fetch('/models/anomaly-detector-v1.0.bin').then(r => r.arrayBuffer());
const knownHash = '9f2c4b1e...'; // model registry entry for v1.0
const testData = new Float32Array([21.4, 21.6, 21.9, 22.3, 61.0]);

// Run audit
const auditor = new SecurityAuditor();
auditor.addCheck({
  id: 'SEC-001', category: 'model-integrity',
  name: 'Model file hash verification',
  verify: async () => verifyModelIntegrity(modelBuffer, knownHash),
});
auditor.addCheck({
  id: 'SEC-002', category: 'input-validation',
  name: 'Sensor bounds validation',
  verify: async () => {
    const result = validateSensorInput(testData, { min: -50, max: 200, maxRateOfChange: 10 });
    return result.valid;
  },
});

const report = await auditor.runAudit();
console.log(\`Audit: \${report.passed}/\${report.passed + report.failed} checks passed\`);`}
</CodeBlock>
