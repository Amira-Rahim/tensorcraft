You have synced state across browser tabs using `BroadcastChannel` or `localStorage` events. Each tab has its own local state. When something changes, the tab broadcasts its update. A coordinator merges the updates into a consistent global state. Federated averaging is the same pattern applied to model weights.

## The FedAvg Algorithm

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="federated-averaging.ts">
{`import * as tf from '@tensorflow/tfjs';

interface ParticipantUpdate {
  organizationId: string;
  weights: Float32Array[];     // one array per layer
  dataSize: number;            // number of training samples
  localLoss: number;
  localAccuracy: number;
}

function federatedAverage(updates: ParticipantUpdate[]): Float32Array[] {
  // Total data across all participants
  const totalData = updates.reduce((sum, u) => sum + u.dataSize, 0);

  // Weighted average of each layer's weights
  const numLayers = updates[0].weights.length;
  const averaged: Float32Array[] = [];

  for (let layer = 0; layer < numLayers; layer++) {
    const layerSize = updates[0].weights[layer].length;
    const result = new Float32Array(layerSize);

    for (const update of updates) {
      const weight = update.dataSize / totalData;  // proportional to data contribution
      for (let i = 0; i < layerSize; i++) {
        result[i] += update.weights[layer][i] * weight;
      }
    }

    averaged.push(result);
  }

  return averaged;
}

// Example:
// Alpha: 2.4M rows (33%) → weights contribute 33% to average
// Beta:  1.8M rows (25%) → weights contribute 25% to average
// Volta: 3.1M rows (42%) → weights contribute 42% to average`}
</CodeBlock>

## Training Rounds

<CodeBlock language="typescript" title="training-rounds.ts">
{`async function runFederatedTraining(
  initialModel: tf.LayersModel,
  participants: string[],
  rounds: number,
): Promise<{ model: tf.LayersModel; history: { round: number; accuracy: number }[] }> {
  let globalWeights = extractWeights(initialModel);
  const history: { round: number; accuracy: number }[] = [];

  for (let round = 0; round < rounds; round++) {
    console.log(\`=== Federated Round \${round + 1}/\${rounds} ===\`);

    // Send global model to all participants
    const updates: ParticipantUpdate[] = [];
    for (const org of participants) {
      // Each org trains locally on their own data
      const update = await trainLocally(org, globalWeights);
      updates.push(update);
      console.log(\`  \${org}: accuracy=\${update.localAccuracy.toFixed(3)}, loss=\${update.localLoss.toFixed(4)}\`);
    }

    // Aggregate updates using weighted average
    globalWeights = federatedAverage(updates);

    // Evaluate global model
    const globalAccuracy = await evaluateGlobalModel(globalWeights);
    history.push({ round: round + 1, accuracy: globalAccuracy });
    console.log(\`  Global accuracy: \${globalAccuracy.toFixed(3)}\`);
  }

  const finalModel = applyWeights(initialModel, globalWeights);
  return { model: finalModel, history };
}

// Round 1: 87.3% → Round 5: 92.1% → Round 10: 94.7% → Round 15: 95.4%
// Each round improves as the model sees patterns from all three data sources`}
</CodeBlock>
