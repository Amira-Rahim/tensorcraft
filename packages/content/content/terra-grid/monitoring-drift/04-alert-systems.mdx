You have configured Sentry alert rules. You know the pain of alert fatigue — when every minor JavaScript error pages the on-call engineer at 3 AM. Good alerting requires thresholds that separate signal from noise, severity tiers that route appropriately, and cooldowns that prevent duplicate alerts from burying the real ones.

{`type AlertSeverity = 'info' | 'warning' | 'critical';
type AlertChannel = 'dashboard' | 'slack' | 'pager';

interface AlertRule {
  name: string;
  metric: string;
  severity: AlertSeverity;
  threshold: number;
  windowMinutes: number;        // aggregate over this window
  cooldownMinutes: number;      // suppress duplicates for this long
  channel: AlertChannel;
  lastFired: number | null;
}

const alertRules: AlertRule[] = [
  {
    name: 'high-latency',
    metric: 'p95_latency_ms',
    severity: 'warning',
    threshold: 100,
    windowMinutes: 15,
    cooldownMinutes: 60,
    channel: 'slack',
    lastFired: null,
  },
  {
    name: 'data-drift-detected',
    metric: 'max_kl_divergence',
    severity: 'warning',
    threshold: 0.3,
    windowMinutes: 60,
    cooldownMinutes: 360,
    channel: 'slack',
    lastFired: null,
  },
  {
    name: 'accuracy-critical',
    metric: 'rolling_accuracy',
    severity: 'critical',
    threshold: 0.90,       // alert if below this
    windowMinutes: 30,
    cooldownMinutes: 120,
    channel: 'pager',
    lastFired: null,
  },
  {
    name: 'device-offline',
    metric: 'offline_device_count',
    severity: 'info',
    threshold: 10,
    windowMinutes: 5,
    cooldownMinutes: 30,
    channel: 'dashboard',
    lastFired: null,
  },
];

function evaluateAlerts(
  rules: AlertRule[],
  metrics: Record<string, number>,
): AlertRule[] {
  const fired: AlertRule[] = [];

  for (const rule of rules) {
    const value = metrics[rule.metric];
    if (value === undefined) continue;

    // Check if threshold is breached
    const breached = rule.metric === 'rolling_accuracy'
      ? value < rule.threshold   // accuracy: alert when below
      : value > rule.threshold;  // everything else: alert when above

    if (!breached) continue;

    // Check cooldown
    if (rule.lastFired) {
      const elapsed = (Date.now() - rule.lastFired) / 60000;
      if (elapsed < rule.cooldownMinutes) continue;
    }

    rule.lastFired = Date.now();
    fired.push(rule);
  }

  return fired;
}`}

## Alert Routing

<CodeBlock language="typescript" title="alert-routing.ts">
{`async function routeAlerts(alerts: AlertRule[], metrics: Record<string, number>): Promise<void> {
  for (const alert of alerts) {
    const message = formatAlert(alert, metrics[alert.metric]);

    switch (alert.channel) {
      case 'dashboard':
        await logToDashboard(message);
        break;
      case 'slack':
        await sendSlack('#ml-alerts', message);
        break;
      case 'pager':
        await pageOnCall(message);
        await sendSlack('#ml-critical', message);
        break;
    }
  }
}

function formatAlert(rule: AlertRule, value: number): string {
  const severity = rule.severity.toUpperCase();
  return \`[\${severity}] \${rule.name}: \${rule.metric}=\${value.toFixed(3)} (threshold: \${rule.threshold})\`;
}

// Output:
// [WARNING] data-drift-detected: max_kl_divergence=0.482 (threshold: 0.3)
// [CRITICAL] accuracy-critical: rolling_accuracy=0.891 (threshold: 0.9)`}
</CodeBlock>
