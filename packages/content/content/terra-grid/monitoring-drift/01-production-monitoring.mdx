You would never deploy a web application without Sentry, DataDog, or at least basic error tracking. You monitor response times, error rates, and throughput. Production ML models need the same observability, but instead of HTTP status codes you track prediction confidence, inference latency, and output distributions.

## Model Health Metrics

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="model-monitor.ts">
{`interface PredictionLog {
  timestamp: number;
  deviceId: string;
  sectorId: string;
  latencyMs: number;
  confidence: number;        // max softmax output
  prediction: number;        // predicted class
  inputStats: {
    mean: number;
    std: number;
    min: number;
    max: number;
  };
}

class ModelMonitor {
  private logs: PredictionLog[] = [];
  private windowSize = 1000;  // rolling window

  logPrediction(log: PredictionLog): void {
    this.logs.push(log);
    if (this.logs.length > this.windowSize * 2) {
      this.logs = this.logs.slice(-this.windowSize);
    }
  }

  getHealthMetrics(): Record<string, number> {
    const recent = this.logs.slice(-this.windowSize);
    if (recent.length === 0) return {};

    const latencies = recent.map(l => l.latencyMs);
    const confidences = recent.map(l => l.confidence);

    return {
      p50Latency: percentile(latencies, 50),
      p95Latency: percentile(latencies, 95),
      p99Latency: percentile(latencies, 99),
      avgConfidence: mean(confidences),
      lowConfidenceRate: recent.filter(l => l.confidence < 0.5).length / recent.length,
      throughput: recent.length,
      errorRate: recent.filter(l => l.latencyMs > 200).length / recent.length,
    };
  }

  getPerSectorHealth(): Record<string, Record<string, number>> {
    const sectors = new Map<string, PredictionLog[]>();
    for (const log of this.logs.slice(-this.windowSize)) {
      const existing = sectors.get(log.sectorId) || [];
      existing.push(log);
      sectors.set(log.sectorId, existing);
    }

    const health: Record<string, Record<string, number>> = {};
    for (const [sector, logs] of sectors) {
      health[sector] = {
        avgLatency: mean(logs.map(l => l.latencyMs)),
        avgConfidence: mean(logs.map(l => l.confidence)),
        count: logs.length,
      };
    }
    return health;
  }
}`}
</CodeBlock>

## Dashboard Aggregation

<CodeBlock language="typescript" title="health-dashboard.ts">
{`function printHealthDashboard(monitor: ModelMonitor): void {
  const metrics = monitor.getHealthMetrics();
  const sectors = monitor.getPerSectorHealth();

  console.log('=== TERRA GRID MODEL HEALTH ===');
  console.log(\`Throughput: \${metrics.throughput} predictions/window\`);
  console.log(\`Latency:    p50=\${metrics.p50Latency}ms  p95=\${metrics.p95Latency}ms  p99=\${metrics.p99Latency}ms\`);
  console.log(\`Confidence: avg=\${metrics.avgConfidence?.toFixed(3)}  low-rate=\${(metrics.lowConfidenceRate * 100).toFixed(1)}%\`);
  console.log('');
  console.log('Per-Sector:');
  for (const [sector, health] of Object.entries(sectors)) {
    const status = health.avgConfidence > 0.85 ? 'OK' : 'WARN';
    console.log(\`  \${sector}: confidence=\${health.avgConfidence.toFixed(3)} latency=\${health.avgLatency.toFixed(0)}ms [\${status}]\`);
  }
}`}
</CodeBlock>
