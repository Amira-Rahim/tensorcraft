Region 4 accuracy degraded from 96.2% to 89.1% over 14 days. The model did not change. The data changed. A sensor firmware update altered the input distribution.
</Dialogue>

<Dialogue speaker="Park Soo-Yeon">
This is why we monitor inputs, not just outputs. The model is doing exactly what we trained it to do — the problem is that the world changed underneath it.
</Dialogue>

Lighthouse CI catches regressions by comparing performance metrics against a baseline. When LCP jumps from 2.1s to 3.8s, you know something changed — maybe a new font, maybe an unoptimized image. Data drift detection works the same way: compare current input distributions against the training baseline, and flag when they diverge.

## Detecting Distribution Shift

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="data-drift-detector.ts">
{`interface FeatureBaseline {
  name: string;
  mean: number;
  std: number;
  histogram: number[];     // binned distribution
  binEdges: number[];
}

interface DriftReport {
  feature: string;
  klDivergence: number;
  meanShift: number;
  stdRatio: number;
  isDrifting: boolean;
}

function computeKLDivergence(p: number[], q: number[]): number {
  // KL(P || Q) — measures how different Q is from reference P
  const epsilon = 1e-10;
  let kl = 0;
  for (let i = 0; i < p.length; i++) {
    const pi = Math.max(p[i], epsilon);
    const qi = Math.max(q[i], epsilon);
    kl += pi * Math.log(pi / qi);
  }
  return kl;
}

function detectDataDrift(
  baseline: FeatureBaseline[],
  currentData: number[][],
  threshold: number = 0.3,
): DriftReport[] {
  return baseline.map((feat, featureIdx) => {
    const currentValues = currentData.map(row => row[featureIdx]);
    const currentMean = mean(currentValues);
    const currentStd = std(currentValues);

    // Build histogram of current data using baseline bins
    const currentHist = buildHistogram(currentValues, feat.binEdges);

    // Normalize to probability distributions
    const baselineDist = normalize(feat.histogram);
    const currentDist = normalize(currentHist);

    const kl = computeKLDivergence(baselineDist, currentDist);

    return {
      feature: feat.name,
      klDivergence: kl,
      meanShift: Math.abs(currentMean - feat.mean) / feat.std,  // in std devs
      stdRatio: currentStd / feat.std,
      isDrifting: kl > threshold,
    };
  });
}`}
</CodeBlock>

## Pinpointing the Drift Source

<CodeBlock language="typescript" title="drift-analysis.ts">
{`function analyzeDrift(reports: DriftReport[]): void {
  const drifting = reports.filter(r => r.isDrifting);

  console.log(\`=== DATA DRIFT REPORT ===\`);
  console.log(\`\${drifting.length} of \${reports.length} features drifting\`);

  for (const report of drifting.sort((a, b) => b.klDivergence - a.klDivergence)) {
    console.log(
      \`  \${report.feature}: KL=\${report.klDivergence.toFixed(3)}, \` +
      \`mean shifted \${report.meanShift.toFixed(1)} std devs, \` +
      \`std ratio=\${report.stdRatio.toFixed(2)}\`
    );
  }

  // Region 4 output:
  // === DATA DRIFT REPORT ===
  // 3 of 24 features drifting
  //   temperature: KL=1.847, mean shifted 8.2 std devs, std ratio=1.01
  //   humidity:    KL=0.412, mean shifted 1.8 std devs, std ratio=0.94
  //   pressure:   KL=0.334, mean shifted 0.9 std devs, std ratio=1.12
  //
  // Root cause: firmware v3.2 changed temperature from Celsius to Kelvin
  // Fix: update preprocessing to normalize the new scale
}`}
</CodeBlock>
