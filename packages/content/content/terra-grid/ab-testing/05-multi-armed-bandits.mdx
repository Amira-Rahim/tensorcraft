You have seen dynamic ad rotation on websites: the system shows the highest-performing ad most of the time while occasionally testing alternatives. Multi-armed bandits apply the same explore-exploit tradeoff to model selection. Instead of fixed traffic splits, the algorithm adapts — sending more traffic to the model that is performing better while still exploring alternatives.

## Epsilon-Greedy Bandit

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="epsilon-greedy.ts">
{`// Epsilon-greedy bandit — same algorithm as dynamic ad rotation

interface ModelArm {
  version: string;
  predict: (input: Float32Array) => Promise<number[]>;
  totalReward: number;
  pulls: number;
}

class EpsilonGreedyBandit {
  private arms: ModelArm[] = [];
  private epsilon: number;
  private decayRate: number;
  private totalPulls: number = 0;

  constructor(epsilon: number = 0.1, decayRate: number = 0.999) {
    this.epsilon = epsilon;
    this.decayRate = decayRate;
  }

  addArm(version: string, predict: ModelArm['predict']): void {
    this.arms.push({ version, predict, totalReward: 0, pulls: 0 });
  }

  selectArm(): ModelArm {
    // Explore: random arm
    const currentEpsilon = this.epsilon * Math.pow(this.decayRate, this.totalPulls);
    if (Math.random() < currentEpsilon) {
      return this.arms[Math.floor(Math.random() * this.arms.length)];
    }

    // Exploit: best average reward
    return this.arms.reduce((best, arm) => {
      const avgReward = arm.pulls > 0 ? arm.totalReward / arm.pulls : Infinity;
      const bestAvg = best.pulls > 0 ? best.totalReward / best.pulls : Infinity;
      return avgReward > bestAvg ? arm : best;
    });
  }

  recordReward(version: string, reward: number): void {
    const arm = this.arms.find(a => a.version === version);
    if (arm) {
      arm.totalReward += reward;
      arm.pulls++;
      this.totalPulls++;
    }
  }

  getStats(): { version: string; pulls: number; avgReward: number; trafficShare: number }[] {
    return this.arms.map(arm => ({
      version: arm.version,
      pulls: arm.pulls,
      avgReward: arm.pulls > 0 ? arm.totalReward / arm.pulls : 0,
      trafficShare: this.totalPulls > 0 ? arm.pulls / this.totalPulls : 0,
    }));
  }
}

// Terra Grid: 3 model versions competing
const bandit = new EpsilonGreedyBandit(0.15, 0.999);
bandit.addArm('v3.0', modelV3.predict);
bandit.addArm('v3.1', modelV3_1.predict);
bandit.addArm('v3.2-beta', modelV3_2.predict);

// Each prediction cycle
const selected = bandit.selectArm();
const prediction = await selected.predict(sensorData);
const correct = prediction[0] === actualLabel ? 1 : 0;
bandit.recordReward(selected.version, correct);

console.log('Traffic distribution:', bandit.getStats());`}

## UCB1 Strategy

<CodeBlock language="typescript" title="ucb1.ts">
{`// Upper Confidence Bound — balances exploitation with exploration bonus

class UCB1Bandit {
  private arms: ModelArm[] = [];
  private totalPulls: number = 0;

  addArm(version: string, predict: ModelArm['predict']): void {
    this.arms.push({ version, predict, totalReward: 0, pulls: 0 });
  }

  selectArm(): ModelArm {
    // Pull each arm at least once
    const unpulled = this.arms.find(a => a.pulls === 0);
    if (unpulled) return unpulled;

    // UCB1 formula: avgReward + sqrt(2 * ln(totalPulls) / armPulls)
    return this.arms.reduce((best, arm) => {
      const ucb = (arm.totalReward / arm.pulls) +
        Math.sqrt(2 * Math.log(this.totalPulls) / arm.pulls);
      const bestUcb = (best.totalReward / best.pulls) +
        Math.sqrt(2 * Math.log(this.totalPulls) / best.pulls);
      return ucb > bestUcb ? arm : best;
    });
  }

  recordReward(version: string, reward: number): void {
    const arm = this.arms.find(a => a.version === version);
    if (arm) {
      arm.totalReward += reward;
      arm.pulls++;
      this.totalPulls++;
    }
  }
}

// UCB1 naturally balances explore/exploit without epsilon tuning
const ucb = new UCB1Bandit();
ucb.addArm('v3.0', modelV3.predict);
ucb.addArm('v3.1', modelV3_1.predict);`}
