</Dialogue>

You have deployed frontend canaries: push to one server, watch error rates, promote or roll back. Model canaries follow the exact same protocol. Deploy the new model to one region, monitor production metrics, and only promote when the data confirms improvement. The key difference: ML canaries must track model-specific health — accuracy, detection speed, false positive rates — not just HTTP errors.

{`// Canary deployment manager — staged rollout with health gates

type DeploymentStage = 'canary' | 'early-access' | 'wide-rollout' | 'full' | 'rolled-back';

interface HealthCheck {
  accuracy: number;
  avgLatencyMs: number;
  falsePositiveRate: number;
  missedAnomalies: number;
}

interface StageConfig {
  name: DeploymentStage;
  regions: string[];
  requiredHealthDuration: number; // hours
  healthThresholds: HealthCheck;
  nextStage: DeploymentStage | null;
}

class CanaryManager {
  private currentStage: DeploymentStage = 'canary';
  private stageStartTime: number = Date.now();
  private stages: Map<DeploymentStage, StageConfig>;

  constructor(stages: StageConfig[]) {
    this.stages = new Map(stages.map(s => [s.name, s]));
  }

  getCurrentStage(): StageConfig | undefined {
    return this.stages.get(this.currentStage);
  }

  checkHealth(metrics: HealthCheck): { healthy: boolean; failures: string[] } {
    const stage = this.getCurrentStage();
    if (!stage) return { healthy: false, failures: ['Unknown stage'] };

    const failures: string[] = [];
    const t = stage.healthThresholds;

    if (metrics.accuracy < t.accuracy)
      failures.push(\`Accuracy \${metrics.accuracy} < \${t.accuracy}\`);
    if (metrics.avgLatencyMs > t.avgLatencyMs)
      failures.push(\`Latency \${metrics.avgLatencyMs}ms > \${t.avgLatencyMs}ms\`);
    if (metrics.falsePositiveRate > t.falsePositiveRate)
      failures.push(\`FP rate \${metrics.falsePositiveRate} > \${t.falsePositiveRate}\`);
    if (metrics.missedAnomalies > t.missedAnomalies)
      failures.push(\`Missed \${metrics.missedAnomalies} > \${t.missedAnomalies}\`);

    return { healthy: failures.length === 0, failures };
  }

  async evaluate(metrics: HealthCheck): Promise<DeploymentStage> {
    const health = this.checkHealth(metrics);

    if (!health.healthy) {
      console.error('Health check failed:', health.failures);
      this.currentStage = 'rolled-back';
      return this.currentStage;
    }

    const stage = this.getCurrentStage()!;
    const elapsed = (Date.now() - this.stageStartTime) / 3600000;

    if (elapsed >= stage.requiredHealthDuration && stage.nextStage) {
      this.currentStage = stage.nextStage;
      this.stageStartTime = Date.now();
      console.log(\`Promoted to stage: \${this.currentStage}\`);
    }

    return this.currentStage;
  }
}

// Terra Grid canary rollout plan
const manager = new CanaryManager([
  {
    name: 'canary', regions: ['region-3'],
    requiredHealthDuration: 48, nextStage: 'early-access',
    healthThresholds: { accuracy: 0.90, avgLatencyMs: 20, falsePositiveRate: 0.05, missedAnomalies: 0 },
  },
  {
    name: 'early-access', regions: ['region-3', 'region-1', 'region-7'],
    requiredHealthDuration: 72, nextStage: 'wide-rollout',
    healthThresholds: { accuracy: 0.91, avgLatencyMs: 18, falsePositiveRate: 0.04, missedAnomalies: 0 },
  },
  {
    name: 'wide-rollout', regions: ['region-1', 'region-2', 'region-3', 'region-5', 'region-7'],
    requiredHealthDuration: 168, nextStage: 'full',
    healthThresholds: { accuracy: 0.92, avgLatencyMs: 15, falsePositiveRate: 0.03, missedAnomalies: 0 },
  },
  {
    name: 'full', regions: ['all'],
    requiredHealthDuration: 0, nextStage: null,
    healthThresholds: { accuracy: 0.92, avgLatencyMs: 15, falsePositiveRate: 0.03, missedAnomalies: 0 },
  },
]);`}
