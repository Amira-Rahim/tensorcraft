      interface LayerProfile {
        name: string;
        params: number;
      }

      interface ModelProfile {
        parameterCount: number;
        memorySizeMB: number;
        avgInferenceMs: number;
        layerBreakdown: LayerProfile[];
        leakedTensors: number;
      }

      async function profileModel(
        model: {
          countParams: () => number;
          layers: { name: string; countParams: () => number }[];
          predict: (x: any) => { data: () => Promise<unknown>; dispose: () => void };
        },
        createInput: () => { dispose: () => void },
        runs: number,
        now: () => number = () => performance.now(),
      ): Promise<ModelProfile> {
        if (!Number.isInteger(runs) || runs < 1) throw new Error('Invalid run count');
        const parameterCount = model.countParams();
        const memorySizeMB = (parameterCount * 4) / 1e6;

        // Include layers with no parameters, such as reshape layers.
        const layerBreakdown = model.layers.map((layer) => ({
          name: layer.name,
          params: layer.countParams(),
        }));

        const before = tf.memory().numTensors;

        // Run one untimed warmup to allow initial compilation.
        // Release its input and output too.
        const warmInput = createInput();
        let warmOutput;
        try { warmOutput = model.predict(warmInput); await warmOutput.data(); }
        finally { warmOutput?.dispose(); warmInput.dispose(); }

        const times: number[] = [];
        for (let i = 0; i < runs; i++) {
          const input = createInput();
          let output;
          try {
            const start = now();
            output = model.predict(input);
            await output.data();
            times.push(Math.max(0, now() - start));
          } finally { output?.dispose(); input.dispose(); }
        }

        const avgInferenceMs = times.reduce((a, b) => a + b, 0) / times.length;
        const leakedTensors = tf.memory().numTensors - before;

        return { parameterCount, memorySizeMB, avgInferenceMs, layerBreakdown, leakedTensors };
      }

Start by measuring weight size, prediction time, and parameters per layer. The running app also needs memory for activations, temporary tensors, and backend buffers. Measure the highest memory use during loading before deciding whether the model fits.

You will measure completed predictions and distinguish stored weight size from total runtime memory.

{`import * as tf from '@tensorflow/tfjs';

interface ModelProfile {
  memorySizeMB: number;
  parameterCount: number;
  inferenceMeanMs: number;
  inferenceP95Ms: number;
  layerBreakdown: { name: string; params: number; outputShape: (number | null)[] | (number | null)[][] }[];
}

// Let the caller create inputs that match the model.
// tf.randomNormal would not suit a model that expects integer token IDs.
async function profileModel(model: tf.LayersModel, createInput: () => tf.Tensor): Promise<ModelProfile> {
  // Logical bytes tracked by TFJS, with backend-specific reliability limits. Not the same
  // number as the weight bytes below: tf.memory() counts every tensor
  // currently allocated, including the ones a sloppy loop forgets to
  // dispose.
  const residentBytesBefore = tf.memory().numBytes;

  // Parameter count
  const parameterCount = model.countParams();
  // Each float32 weight uses 4 bytes. Divide by 1e6 for MB.
  // Dividing by 1024^2 gives MiB: these weights are 340.0 MB, or 324.2 MiB.
  const memorySizeMB = (parameterCount * 4) / 1e6;

  // Layer breakdown
  const layerBreakdown = model.layers.map(layer => ({
    name: layer.name,
    params: layer.countParams(),
    outputShape: layer.outputShape,
  }));

  // Warm up before timing 100 calls. Read the warmup output, then dispose it.
  // Some backends need more warmup. Check that timings have settled.
  const warmupInput = createInput();
  let warmupOutput: tf.Tensor | undefined;
  try {
    warmupOutput = model.predict(warmupInput) as tf.Tensor;
    await warmupOutput.data();
  } finally {
    warmupOutput?.dispose(); warmupInput.dispose();
  }

  const times: number[] = [];
  for (let i = 0; i < 100; i++) {
    const input = createInput();
    let output: tf.Tensor | undefined;
    try {
      const start = performance.now();
      output = model.predict(input) as tf.Tensor;
      // Keep readback inside the timed section, as in module 4.
    // Otherwise GPU timing may only measure how fast work is queued.
      await output.data();
      times.push(performance.now() - start);
    } finally {
      output?.dispose(); input.dispose();
    }
  }

  times.sort((a, b) => a - b);

  // Net growth needs investigation. Warmup caches or other page work can
  // also affect the count; this is not a process-memory measurement.
  const trackedByteChange = tf.memory().numBytes - residentBytesBefore;
  if (trackedByteChange !== 0) console.warn('Tracked tensor-byte change:', trackedByteChange);

  return {
    memorySizeMB,
    parameterCount,
    inferenceMeanMs: times.reduce((a, b) => a + b) / times.length,
    inferenceP95Ms: times[Math.ceil(times.length * 0.95) - 1], // nearest-rank P95
    layerBreakdown,
  };
}

// Integration input: supply model and an input factory matching its contract.
// The table below uses a size-accounting proxy for the fictional 340 MB
// component, not a generative language-model architecture. Its layers are:
// embedding [50000, 256], flatten [80 * 256], Dense 3320, Dense 1264.
// The embedding needs int32 token IDs in [0, 49999], not random floats.
// Its parameter arithmetic is 12,800,000 + 67,996,920 + 4,197,744.
// The runtime lab profiles a separate, small classifier.
const profile = await profileModel(model, () => tf.randomUniform([1, 80], 0, 50000, 'int32'));
console.log('Model size:', profile.memorySizeMB.toFixed(1), 'MB');
console.log('Parameters:', profile.parameterCount.toLocaleString());
console.log('Inference mean:', profile.inferenceMeanMs.toFixed(1), 'ms');
console.log('Inference P95:', profile.inferenceP95Ms.toFixed(1), 'ms');

// A total tells you the model is too big. A share tells you where to cut.
console.log('Layer breakdown:');
for (const layer of profile.layerBreakdown) {
  const share = (layer.params / profile.parameterCount) * 100;
  console.log(\`  \${layer.name.padEnd(22)}\${layer.params.toLocaleString().padStart(12)}  \${share.toFixed(1)}%\`);
}

// Fictional profiling table, with parameter counts from the proxy above:
//
//   Model size: 340.0 MB
//   Parameters: 84,994,664
//   Inference mean: 45.0 ms
//   Inference P95: 112.0 ms
//   Layer breakdown:
//     embedding_Embedding1    12,800,000  15.1%
//     flatten_Flatten1                 0  0.0%
//     dense_Dense1            67,996,920  80.0%
//     dense_Dense2             4,197,744  4.9%
//
// Both supplied timings describe the fictional WASM run: the dashboard had no working
// WebGL context that night, so the profile ran on the fallback.
//
// The flatten row is in there because the loop walks model.layers, not
// only the layers that hold weights. It reshapes the embedding's 80 by 256
// output into 20,480 values and owns nothing, so its share is 0.0%.
// Rounded to millions that is 85 million weights. The profiler prints the
// exact count, and 84,994,664 float32 weights at 4 bytes each are the
// 340.0 MB on Nazari's board.
//
// Read the shares, not the total. 80% of this model is one Dense layer, so
// that layer holds most of the 340 MB. The 45 ms mean and the 112 ms P95 are
// the latency baseline, not an explanation of the crash: the dashboard died
// with the free-memory line at zero, 340 MB of weights loading into a 512 MB
// budget. In that baseline the tail is the number to gate on. It is 2.5x the
// mean, but neither statistic alone gives the fraction exceeding an arbitrary latency budget. Count those violations directly.
// These scenario values carry into the capstone; they are not reproduced device measurements.`}
