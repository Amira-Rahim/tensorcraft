Profiling a model means three measurements: memory footprint, inference latency (mean and tail), and which layers hold the parameters. Get all three before touching a single weight; the numbers decide where the knife goes.

<CodeBlock language="typescript" title="profiling.ts">
{`import * as tf from '@tensorflow/tfjs';

interface ModelProfile {
  memorySizeMB: number;
  parameterCount: number;
  inferenceMeanMs: number;
  inferenceP95Ms: number;
  layerBreakdown: { name: string; params: number; outputShape: number[] }[];
}

async function profileModel(model: tf.LayersModel, inputShape: number[]): Promise<ModelProfile> {
  // Memory
  const memBefore = tf.memory();

  // Parameter count
  const parameterCount = model.countParams();
  const memorySizeMB = (parameterCount * 4) / (1024 * 1024);  // float32 = 4 bytes

  // Layer breakdown
  const layerBreakdown = model.layers.map(layer => ({
    name: layer.name,
    params: layer.countParams(),
    outputShape: layer.outputShape as number[],
  }));

  // Inference timing (warm up + 100 iterations)
  const warmupInput = tf.randomNormal([1, ...inputShape]);
  (model.predict(warmupInput) as tf.Tensor).dispose();
  warmupInput.dispose();

  const times: number[] = [];
  for (let i = 0; i < 100; i++) {
    const input = tf.randomNormal([1, ...inputShape]);
    const start = performance.now();
    const output = model.predict(input) as tf.Tensor;
    times.push(performance.now() - start);
    output.dispose();
    input.dispose();
  }

  times.sort((a, b) => a - b);

  return {
    memorySizeMB,
    parameterCount,
    inferenceMeanMs: times.reduce((a, b) => a + b) / times.length,
    inferenceP95Ms: times[Math.floor(times.length * 0.95)],
    layerBreakdown,
  };
}

const profile = await profileModel(model, [80, 11]);
console.log('Model size:', profile.memorySizeMB.toFixed(1), 'MB');
console.log('Parameters:', profile.parameterCount.toLocaleString());
console.log('Inference mean:', profile.inferenceMeanMs.toFixed(1), 'ms');
console.log('Inference P95:', profile.inferenceP95Ms.toFixed(1), 'ms');`}
