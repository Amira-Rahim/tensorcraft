      function makeInferenceCache(model, decimals = 2, maxEntries = 100) {
        if (
          !Number.isInteger(decimals) ||
          decimals < 0 ||
          decimals > 10 ||
          !Number.isInteger(maxEntries) ||
          maxEntries < 1
        )
          throw new Error('Invalid cache settings');
        const cache = new Map();
        const stats = { hits: 0, misses: 0 };
        const factor = 10 ** decimals;
        return {
          stats,
          predict(input) {
            const values = Array.from(input.dataSync()).map((v) => Math.round(v * factor) / factor);
            const key = JSON.stringify([input.dtype, input.shape, values]);
            if (cache.has(key)) {
              stats.hits++;
              const scores = cache.get(key);
              cache.delete(key);
              cache.set(key, scores);
              return scores.slice();
            }
            stats.misses++;
            const out = model.predict(input);
            let scores;
            try {
              scores = Array.from(out.dataSync());
            } finally {
              out.dispose();
            }
            cache.set(key, scores);
            if (cache.size > maxEntries) cache.delete(cache.keys().next().value);
            return scores.slice();
          },
        };
      }

<CodeBlock language="typescript" title="inference-cache.ts">
{`class InferenceCache<T> {
  private cache = new Map<string, { result: T; timestamp: number }>();
  private readonly maxSize: number;
  private readonly ttlMs: number;
  private readonly hashInput: (input: number[]) => string;
  private hits = 0;
  private misses = 0;

  // The key function comes in from outside on purpose. Rounding the values
  // before joining them is what lands near-identical signals on one entry,
  // and how far to round is not a decision this class can make for you:
  // round too little and every repeat misses, too much and two signals that
  // should have stayed apart are served one verdict between them. Writing
  // that function, and keeping the values in their own places while you do,
  // is the Challenge below.
  constructor(
    hashInput: (input: number[]) => string,
    maxSize: number = 1000,
    ttlMs: number = 60000,
  ) {
    if (!Number.isInteger(maxSize) || maxSize < 1 || !Number.isFinite(ttlMs) || ttlMs < 0) {
      throw new Error('Invalid cache limits');
    }
    this.hashInput = hashInput;
    this.maxSize = maxSize;
    this.ttlMs = ttlMs;
  }

  get(input: number[]): T | null {
    const key = this.hashInput(input);
    const entry = this.cache.get(key);
    if (!entry) {
      this.misses++;
      return null;
    }
    if (Date.now() - entry.timestamp > this.ttlMs) {
      this.cache.delete(key);
      this.misses++;
      return null;
    }
    // True LRU: a Map iterates in insertion order, so delete + re-set
    // moves this key to the back of the eviction line. Without it you have
    // built FIFO, which evicts your hottest entry because it arrived first.
    this.cache.delete(key);
    this.cache.set(key, entry);
    this.hits++;
    return entry.result;
  }

  set(input: number[], result: T): void {
    const key = this.hashInput(input);
    // Re-setting an existing key must refresh its recency (the same
    // delete-then-set trick get() uses) and must NOT evict a stranger:
    // the size isn't growing.
    if (this.cache.has(key)) {
      this.cache.delete(key);
    } else if (this.cache.size >= this.maxSize) {
      // Map's first key = least recently used (get() refreshes position)
      const lru = this.cache.keys().next().value;
      if (lru !== undefined) this.cache.delete(lru);
    }
    this.cache.set(key, { result, timestamp: Date.now() });
  }

  get hitRate(): number {
    return this.hits / (this.hits + this.misses) || 0;
  }
}

// Usage. signalKey rounds the way the Challenge's decimals = 2 does. The
// TTL is six hours, past the end of the watch, because the repeats this
// cache is built for arrive about a hundred seconds apart: on a 30-second
// clock nearly every entry expires before its repeat comes back, and most
// of the 40 percent below goes with it.
const signalKey = (input: number[]) => input.map(v => Math.round(v * 100) / 100).join(',');
const cache = new InferenceCache<string>(signalKey, 500, 6 * 60 * 60 * 1000);

function cachedClassify(signal: number[]): string {
  const cached = cache.get(signal);
  if (cached !== null) return cached;

  const result = classify(model, signal);
  cache.set(signal, result);
  return result;
}

// After the overnight run: 4,110 lookups, 1,644 hits. cache.hitRate reads 0.40.

// Illustrative mean: hitRate * hit + (1 - hitRate) * miss.
// A real miss includes lookup, key construction, inference and insertion.
// The following numbers are supplied scenario inputs, not lab measurements. A
// classification-only time is 0.035 ms, and assumed hit time is 0.0013 ms.
// Ignoring miss-path overhead gives an optimistic estimate:
const optimisticMs = 0.4 * 0.0013 + 0.6 * 0.035;
console.log(optimisticMs.toFixed(4));  // 0.0215

// On these assumptions, 1,644 hits save about 55 milliseconds overnight,
// before subtracting the miss-path overhead. This is an optimistic estimate.
//
// The 45 ms in this module's first lesson is not this call. It belongs to
// the 85-million-parameter language component behind the report pipeline.
// A cache for that call might save more, if requests repeat and its
// keys cover every input and generation setting. Measure the complete path.`}
</CodeBlock>

This example assumes immutable string verdicts from one fixed model and preprocessing version. For object or array results, copy them on storage and retrieval or enforce immutability. The full exercise tests returned-array mutation.

This cache stores plain verdicts, so evicting an entry does not require tensor disposal. Caching tensors is possible but needs explicit ownership: dispose them on eviction and prevent callers from disposing shared handles. Otherwise later hits can fail with `Tensor is disposed.`

Keep the ledger honest on both sides while you read that. On the credit side this cache removes 40 percent of the classifier's forward passes, worth 55 milliseconds a night. On the debit side it retains a bounded set of earlier verdicts, with its own expiry rules and its own ways of being wrong, and the next page you get will be about that copy. A cache earns its place when the call behind it costs enough to pay for the invalidation you just signed up for. Time the call first.
