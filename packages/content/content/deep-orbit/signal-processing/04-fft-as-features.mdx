      import * as tf from '@tensorflow/tfjs';

      function extractFeatures(signal: number[], sampleRate: number, loHz: number, hiHz: number): number[] {
        return tf.tidy(() => {
          // Raw features, unchanged by the FFT work.
          const amplitude = Math.max(...signal.map(Math.abs));
          const zeroCrossings = signal.reduce((sum, v, i) =>
            i > 0 && v * signal[i - 1] < 0 ? sum + 1 : sum, 0
          );

          // Bin k has its center at k * binWidth hertz. Keep every bin whose

          // every signal of this length at this sample rate.
          const binWidth = sampleRate / signal.length;

You will turn each capture into a consistent set of measurements that a model can use.

## Mapping Captures into Features

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="feature-engineering.ts">
{`import * as tf from '@tensorflow/tfjs';

// BEFORE: raw signal features only (3 values)
// The model sees: [amplitude, frequency, phase]
// Story result, not a benchmark: 78%

// AFTER: raw + FFT features (3 + N frequency bins)
// The model sees: [amplitude, frequency, phase, fft_bin_0, fft_bin_1, ..., fft_bin_N]
// Story result, not a benchmark: 87%

function extractFeatures(signal: number[], windowSize: number = 64): number[] {
  // The fixed bin band below requires exactly 64 samples at 256 Hz.
  if (windowSize !== 64 || signal.length < 64) throw new Error('Expected a 64-sample window at 256 Hz');
  return tf.tidy(() => {
    // All eleven features read the SAME window, so a row means the same
    // thing no matter how long the capture is. windowSize counts SAMPLES
    // from one channel: 64 of them at 256 Hz is a quarter second. Module
    // 1's archive row carried a 64 too, and that one counted channels,
    // one per hull sensor. Two different 64s.
    const windowSlice = signal.slice(0, windowSize);

    // Raw features (the original 3). The "frequency" feature is a raw
    // zero-crossing count over that window. For a clean zero-centered
    // sinusoid it relates to frequency; mixtures, offsets and noise change
    // that relationship. Exact zeros also need a convention. 64 samples
    // can hold at most 63 crossings, which is the top of the range the
    // classifier's normalizer assumes.
    const amplitude = Math.max(...windowSlice.map(Math.abs));
    const zeroCrossings = windowSlice.reduce((sum, v, i) =>
      i > 0 && v * windowSlice[i - 1] < 0 ? sum + 1 : sum, 0
    );
    // "phase" is the angle of the first two samples, atan2(s1, s0): a crude
    // feature derived from two samples, not a general estimate of
    // sinusoidal phase. The trained model expects it, so it stays, and it keeps the
    // name the contract gave it.
    const phase = Math.atan2(windowSlice[1], windowSlice[0]);

    // FFT features, computed from those same samples.
    const fft = tf.spectral.rfft(tf.tensor1d(windowSlice));
    const magnitudes = tf.abs(fft).arraySync() as number[];

    // Eight FIXED bins, the same bins for every signal. Sorting by
    // magnitude would throw away frequency identity: "fft_3" would mean
    // a different frequency in every row, and no downstream model could
    // ever ask "what lives at 72 Hz?". At this window's 4 Hz resolution
    // (256 Hz / 64 samples), bins 13-20 have centers 52-80 Hz and cover
    // 50-82 Hz, above the catalogued background lines. Bin 18 is centered
    // at 72 Hz, nearest to 73.4 Hz. An off-bin tone can contribute to
    // several of these features; bins are not isolated frequency buckets.
    const fixedBins = magnitudes.slice(13, 21);

    // Combine: [raw features..., fixed-bin FFT magnitudes...]
    return [amplitude, zeroCrossings, phase, ...fixedBins];
  });
}

// Before: model input shape [null, 3]
// After:  model input shape [null, 11]  (3 raw + 8 FFT)
console.log('Feature vector:', extractFeatures(signalSamples));`}
</CodeBlock>

## Normalization Matters

Features can have very different scales. That can affect initial activations and optimization. A large input does not establish feature importance, and normalization does not guarantee that all features become equally useful. Fit scaling on training rows only and reuse it on validation, test and inference rows.

<CodeBlock language="typescript" title="normalization.ts">
{`import * as tf from '@tensorflow/tfjs';

// Integration sketch: trainingFeatures contains only the training split.
// Each row has the same finite feature columns. No empty or ragged rows.
// These are population statistics for this training sample. Constant
// columns use scale 1 and become zero; their standard deviation stays zero.

function normalizeFeatures(features: number[][]): { normalized: number[][]; stats: { mean: number; std: number }[] } {
  if (!features.length || !features[0].length ||
      features.some(row => row.length !== features[0].length || row.some(v => !Number.isFinite(v)))) {
    throw new Error('Expected finite, nonempty rectangular feature rows');
  }
  // Z-score normalization: (value - mean) / std
  // Mean ≈ 0; nonconstant columns have population std ≈ 1.
  const stats = features[0].map((_, colIdx) => {
    const column = features.map(row => row[colIdx]);
    const mean = column.reduce((a, b) => a + b, 0) / column.length;
    const std = Math.sqrt(
      column.reduce((sum, v) => sum + (v - mean) ** 2, 0) / column.length
    );
    return { mean, std };  // preserve the measured standard deviation
  });

  const normalized = features.map(row =>
    row.map((val, i) => (val - stats[i].mean) / (stats[i].std || 1))
  );

  return { normalized, stats };
}

// Save stats from training data; apply the SAME stats during inference
// This is critical: normalize test data with training statistics
const { normalized: trainNorm, stats } = normalizeFeatures(trainingFeatures);

// During inference:
function normalizeWithStats(features: number[], stats: { mean: number; std: number }[]): number[] {
  return features.map((val, i) => (val - stats[i].mean) / (stats[i].std || 1));
}`}
</CodeBlock>

## Building the Enhanced Model

<CodeBlock language="typescript" title="enhanced-model.ts">
{`import * as tf from '@tensorflow/tfjs';

// Original model: 3 inputs → 32 → 16 → 1 (78% accuracy)
// Enhanced model: 11 inputs → 32 → 16 → 1 (87% accuracy)
// Integration sketch: split raw captures BEFORE extracting overlapping
// windows. Supply trainingFeatures, validationFeatures, testFeatures and
// corresponding label tensors. Fit stats on trainingFeatures only.
// Same hidden stack, same optimizer, same epoch count. The parameter
// counts are in the paragraph under this block.

const enhancedModel = tf.sequential();
enhancedModel.add(tf.layers.dense({
  units: 32,
  activation: 'relu',
  inputShape: [11],  // 3 raw + 8 FFT features
}));
enhancedModel.add(tf.layers.dense({ units: 16, activation: 'relu' }));
enhancedModel.add(tf.layers.dense({ units: 1, activation: 'sigmoid' }));

enhancedModel.compile({
  optimizer: 'adam',
  loss: 'binaryCrossentropy',
  metrics: ['accuracy'],
});

// Train on the normalized rows from above, not the raw feature vectors.
const trainingX = tf.tensor2d(trainNorm);
const validationX = tf.tensor2d(validationFeatures.map(row => normalizeWithStats(row, stats)));
const history = await enhancedModel.fit(trainingX, trainY, {
  epochs: 50,
  validationData: [validationX, validationY],
  callbacks: {
    onEpochEnd: (epoch, logs) => {
      if (epoch % 10 === 0) {
        console.log(\`Epoch \${epoch}: acc=\${logs?.acc.toFixed(3)}, val_acc=\${logs?.val_acc.toFixed(3)}\`);
      }
    },
  },
});
// Record your own training and validation metrics. Evaluate the untouched
// test split only after choosing features, scaling and model settings.
const testX = tf.tensor2d(testFeatures.map(row => normalizeWithStats(row, stats)));
const [testLoss, testAcc] = enhancedModel.evaluate(testX, testY) as tf.Tensor[];
console.log('Held-out accuracy:', testAcc.dataSync()[0].toFixed(3));
tf.dispose([trainingX, validationX, testX, testLoss, testAcc]);
enhancedModel.dispose();
// The caller still owns the supplied label tensors.`}

Nazari was right that this is a bigger network. `countParams()` reports 673 parameters for the three-input stack and 929 for the eleven-input one, 38% more. All 256 of the added weights sit in the first layer, one for each new input on each of its 32 units. Changing the features also changed the parameter count. This comparison cannot isolate those effects. Use the same held-out rows and training budget, repeat appropriate runs and include a simpler baseline before attributing a gain to the new features. A validation score can exceed or trail a test score because of sampling or distribution differences; neither ordering proves contamination.

One thing DEEP SCAN 2.0 is not: the build running on the bridge dashboard. That deployment keeps reading module 4's three min-max normalized features, and it keeps reading them for another five modules. The eleven-feature vector does its work off the dashboard first. Module 7's autoencoder trains on exactly this vector, and module 10 is where the shipped classifier is retrained on it, with a five-way head in place of the binary one.
