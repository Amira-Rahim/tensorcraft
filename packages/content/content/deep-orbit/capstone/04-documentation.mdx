      interface ModelMetadata {
        name: string;
        version: string;
        accuracy: number;
        benchmarkedOn: string;
        weightBytes: number;
        inferenceMs: number;
        recallByClass: Record<string, number>;
        limitations: string[];
        inputFormat: string;
        outputFormat: string;
      }

      function generateModelCard(meta: ModelMetadata): string {
        const pct = (v: number) => `${(v * 100).toFixed(1)}%`;
        const [worstClass, worstRecall] = Object.entries(meta.recallByClass).sort(
          (a, b) => a[1] - b[1],
        )[0];

        const lines = [
          `# ${meta.name} v${meta.version}`,
          '',
          '## Performance',
          `- Accuracy: ${pct(meta.accuracy)} (measured at ${meta.benchmarkedOn})`,
          `- Size: ${meta.weightBytes} bytes`,
          `- Inference: ${meta.inferenceMs}ms`,
          '',
          '## Limitations',
          `- Lowest recall: ${worstClass} at ${pct(worstRecall)}`,
          ...meta.limitations.map(l => `- ${l}`),
          '',
          '## Usage',
          `- Input: ${meta.inputFormat}`,
          `- Output: ${meta.outputFormat}`,
        ];
        return lines.join('\n');
      }

{`// Fictional mission card. These scores, timings and bundle sizes are
// supplied story values, not measurements of the downloadable model.
// A Model Card is structured documentation for an ML model
// Think of it as package.json + README + CHANGELOG for your model

interface ModelCard {
  name: string;
  version: string;
  description: string;
  author: string;

  // Training details
  training: {
    dataset: string;
    datasetSize: number;
    features: string[];
    trainTestSplit: string;
    epochs: number;
    optimizer: string;
  };

  // Performance metrics
  performance: {
    accuracy: number;
    benchmarkedOn: string;   // which evaluation produced the number above
    perClass: { label: string; examples: number; correct: number }[];
    inferenceMs: number;   // the classifier alone
    pipelineMs: number;    // the whole classify() call, end to end
    weightBytes: number;    // this model's weight file
    bundleSizeMB: number;   // everything the dashboard downloads to run it
  };

  // What the model cannot do
  limitations: string[];

  // How to use it
  usage: {
    inputFormat: string;
    outputFormat: string;
    exampleCode: string;
  };
}

const deepScanCard: ModelCard = {
  name: 'DEEP SCAN Signal Classifier',
  version: '1.0.0',
  description: 'Classifies deep space radio signals into 5 categories: pulsar, quasar, noise, interference, anomaly.',
  author: 'Ensign Demir, Archimedes',

  training: {
    dataset: 'Sector 7-G Signal Archive (11 days, one reading per 33-second window)',
    datasetSize: 28_800,
    features: ['amplitude', 'zeroCrossings', 'phase', 'FFT bin magnitudes (8 fixed bins)'],
    trainTestSplit: '80/10/10 (train/val/test)',
    epochs: 50,
    optimizer: 'Adam (lr=0.001)',
  },

  performance: {
    accuracy: 1859 / 2025, // 91.8% rounded; distinct from the earlier six-anomaly pruning slice
    benchmarkedOn: 'handover evaluation after pruning, 2025 labeled captures',
    perClass: [
      { label: 'pulsar', examples: 500, correct: 470 },
      { label: 'quasar', examples: 500, correct: 468 },
      { label: 'noise', examples: 500, correct: 455 },
      { label: 'interference', examples: 500, correct: 445 },
      { label: 'anomaly', examples: 25, correct: 21 },
    ],
    inferenceMs: 0.035,  // the eleven-feature classifier alone at the dashboard station, one
                         // backend, readback inside the clock (module 10)
    pipelineMs: 5,       // the whole classify() call on WebGPU, 15 on WebGL: feature extraction,
                         // the classifier and the readback, which is what the monitor times
    weightBytes: 3_988,  // final selective export retains this pruned classifier at float32
    bundleSizeMB: 85,    // rounded weight payload, mainly the quantized language component;
                         // transfer compression, JS and metadata are separate
  },

  limitations: [
    'Trained on Sector 7-G captures only; accuracy may drop in other sectors',
    'Anomaly class has lowest recall: 21 of the 25 labelled anomalies in the holdout, 84%; rare signals may be misclassified as noise',
    'Requires exactly 64 samples per capture (0.25 seconds of one channel at 256 Hz); the 8 FFT bins and the zero-crossing scale are defined at that length, so shorter bursts cannot be classified',
    'WebGPU backend required for the 5 ms end-to-end path; the pipeline falls back to 15 ms on WebGL. The classifier itself was timed in microseconds on one backend, which is not a figure for all of them',
    'Not validated on signals below -20 dB SNR (signal-to-noise ratio; signal RMS amplitude is one tenth of noise RMS amplitude, with comparable impedance)',
  ],

  usage: {
    inputFormat: 'Float32Array of exactly 64 raw signal samples; the pipeline extracts the 11 features and applies the saved training statistics itself',
    outputFormat: '{ label: string, confidence: number, isAnomaly: boolean }',
    exampleCode: 'const result = await deepScan.classify(signalData);',
  },
};`}
</CodeBlock>

Keep weight-file bytes, total download bytes, classifier latency and whole-pipeline latency separate. They describe different scopes. The values on this mission card are fictional. The 11→32→16→5 network has 997 parameters: 384 + 528 + 85. The final selective export retains this pruned classifier at float32, giving 3,988 weight bytes, but the course does not ship that five-class artifact or the fictional 85 MB language component. The downloadable runtime lab records measurements for its own named models and device.

A production card also needs the artifact and preprocessing versions, label map, evaluation data identity and split method, sample counts, per-class metrics and averaging convention. Report uncertainty where the evaluation supports it, plus the device, backend, timing protocol and date. A single accuracy and a date cannot establish fitness for another use.

Caching may reduce repeat downloads; it does not remove the first-load cost or guarantee that storage persists. Accepting a larger bundle needs an explicit cold-start and memory budget, with a usable failure path.

Two different counts of 64 appear in this course. An archive row records 33 seconds across 64 channels at 256 Hz. That is 540,672 samples, a little over two megabytes in float32.

The classifier uses a much smaller input: 64 samples from one channel, or a quarter of a second at 256 Hz. This gives 8 selected FFT bins spaced 4 Hz apart and at most 63 zero crossings. Label channel count and window length separately in the input schema.

The runbook now names the checks: inspect calibration and preprocessing, review recent labeled cases, and evaluate any replacement before release. ARIA accepts the steps and the person responsible for escalation. Demir signs the handover as the pulse interval reaches 31.6 seconds. Tomorrow they deploy.
