<CodeBlock language="typescript" title="model-card.ts">
{`// A Model Card is structured documentation for an ML model
// Think of it as package.json + README + CHANGELOG for your model

interface ModelCard {
  name: string;
  version: string;
  description: string;
  author: string;

  // Training details
  training: {
    dataset: string;
    datasetSize: number;
    features: string[];
    trainTestSplit: string;
    epochs: number;
    optimizer: string;
  };

  // Performance metrics
  performance: {
    accuracy: number;
    precision: number;
    recall: number;
    f1Score: number;
    inferenceMs: number;
    modelSizeMB: number;
  };

  // What the model cannot do
  limitations: string[];

  // How to use it
  usage: {
    inputFormat: string;
    outputFormat: string;
    exampleCode: string;
  };
}

const deepScanCard: ModelCard = {
  name: 'DEEP SCAN Signal Classifier',
  version: '1.0.0',
  description: 'Classifies deep space radio signals into 5 categories: pulsar, quasar, noise, interference, anomaly.',
  author: 'Ensign Demir, Archimedes',

  training: {
    dataset: 'Sector 7-G Signal Archive (11 days, one reading per 33-second window)',
    datasetSize: 28_800,
    features: ['amplitude', 'zeroCrossings', 'phase', 'FFT bin magnitudes (8 fixed bins)'],
    trainTestSplit: '80/10/10 (train/val/test)',
    epochs: 50,
    optimizer: 'Adam (lr=0.001)',
  },

  performance: {
    accuracy: 0.918,
    precision: 0.92,
    recall: 0.91,
    f1Score: 0.915,
    inferenceMs: 5,
    modelSizeMB: 42,
  },

  limitations: [
    'Trained on Sector 7-G captures only; accuracy may drop in other sectors',
    'Anomaly class has lowest recall (84%); rare signals may be misclassified as noise',
    'Requires minimum 80-sample window; cannot classify very short bursts',
    'WebGPU backend required for <10ms inference; falls back to 15ms on WebGL',
    'Not validated on signals below -20 dB SNR',
  ],

  usage: {
    inputFormat: 'Float32Array of 80 signal samples, normalized to [-1, 1]',
    outputFormat: '{ label: string, confidence: number, isAnomaly: boolean }',
    exampleCode: 'const result = await deepScan.classify(signalData);',
  },
};`}

<RecallPrompt question="Why does documentation discipline transfer from software to ML, but verifiability does not?" answer="JSDoc describes a deterministic function whose behavior you can confirm by reading the source. A model card describes statistical performance measured on a held-out benchmark that may not match production data. You can verify a documented function by running it; you cannot verify a model card's accuracy claim without running a fresh evaluation, and that benchmark may already be stale by the time anyone reads it." />
