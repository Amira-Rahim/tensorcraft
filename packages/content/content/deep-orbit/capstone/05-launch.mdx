Every web app you've built ends the same way: `npm run build`, push to production, watch the deploy. Deploying DEEP SCAN is no different. The model file is a static asset — like an image or a font. It goes in the build output, gets served from a CDN, and loads in the browser. The entire ML system runs client-side.

// Model files are static assets — treat them like images or fonts

interface BuildConfig {
  entry: string;
  output: string;
  assets: {
    models: string[];    // model.json + weight shards
    testData: string[];  // holdout data for validation
  };
  optimization: {
    minify: boolean;
    treeshake: boolean;
    splitChunks: boolean;
  };
}

const config: BuildConfig = {
  entry: 'src/index.ts',
  output: 'dist',
  assets: {
    models: [
      'models/deep-scan-v1/model.json',      // 2 KB — model topology
      'models/deep-scan-v1/weights.bin',      // 42 MB — quantized + pruned
    ],
    testData: [
      'data/validation-sample.json',          // 100 samples for smoke test
    ],
  },
  optimization: {
    minify: true,
    treeshake: true,
    splitChunks: true,   // Separate model loading from app code
  },
};

// Build output:
// dist/
// ├── index.html              (~5 KB)
// ├── app.js                  (~150 KB gzipped)
// ├── models/
// │   ├── model.json          (~2 KB)
// │   └── weights.bin         (~42 MB)
// └── data/
//     └── validation.json     (~50 KB)`}

## Deploy and Verify

<CodeBlock language="typescript" title="deploy.ts">
{`// Deployment verification — smoke test the live system

async function verifyDeployment(baseUrl: string): Promise<boolean> {
  console.log('=== DEEP SCAN Deployment Verification ===');
  console.log('');

  // 1. Check model loads
  console.log('1. Loading model...');
  const modelUrl = \`\${baseUrl}/models/deep-scan-v1/model.json\`;
  const response = await fetch(modelUrl);
  if (!response.ok) {
    console.error('✗ Model failed to load:', response.status);
    return false;
  }
  console.log('✓ Model loaded successfully');

  // 2. Check inference works
  console.log('2. Running test inference...');
  const testSignal = new Float32Array(80).fill(0).map(() => Math.random() * 2 - 1);
  const result = await pipeline.classify(testSignal);
  if (!result.label || result.confidence < 0) {
    console.error('✗ Inference returned invalid result');
    return false;
  }
  console.log(\`✓ Inference works: \${result.label} (\${(result.confidence * 100).toFixed(1)}%)\`);

  // 3. Check performance
  console.log('3. Benchmarking...');
  const times: number[] = [];
  for (let i = 0; i < 20; i++) {
    const input = new Float32Array(80).fill(0).map(() => Math.random() * 2 - 1);
    const t0 = performance.now();
    await pipeline.classify(input);
    times.push(performance.now() - t0);
  }
  const avgMs = times.reduce((a, b) => a + b) / times.length;
  console.log(\`✓ Avg inference: \${avgMs.toFixed(1)}ms\`);

  console.log('');
  console.log('=== Deployment verified ===');
  return true;
}`}
</CodeBlock>
