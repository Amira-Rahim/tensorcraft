{`interface PredictionLog {
  timestamp: number;
  confidence: number;
  latencyMs: number;
  inputFeatures: number[];
  label: string;
}

class ModelMonitor {
  private logs: PredictionLog[] = [];
  private baselineFeatures: number[][] = [];

  constructor(baselineData: number[][]) {
    this.baselineFeatures = baselineData;
  }

  log(prediction: PredictionLog): void {
    this.logs.push(prediction);
    this.checkAlerts(prediction);
  }

  // Confidence tracking — like error rate monitoring
  get avgConfidence(): number {
    const recent = this.logs.slice(-100);
    return recent.reduce((sum, p) => sum + p.confidence, 0) / recent.length;
  }

  // Latency percentiles — like response time monitoring
  getLatencyPercentile(p: number): number {
    const sorted = this.logs.slice(-1000)
      .map(l => l.latencyMs)
      .sort((a, b) => a - b);
    return sorted[Math.floor(sorted.length * p / 100)];
  }

  // Data drift detection — unique to ML monitoring
  computeDrift(recentInputs: number[][]): number {
    const baselineMean = this.computeMean(this.baselineFeatures);
    const recentMean = this.computeMean(recentInputs);

    // Simple drift: Euclidean distance between feature means
    let sumSqDiff = 0;
    for (let i = 0; i < baselineMean.length; i++) {
      sumSqDiff += (baselineMean[i] - recentMean[i]) ** 2;
    }
    return Math.sqrt(sumSqDiff);
  }

  private computeMean(data: number[][]): number[] {
    const featureCount = data[0].length;
    const means = new Array(featureCount).fill(0);
    for (const row of data) {
      for (let i = 0; i < featureCount; i++) {
        means[i] += row[i] / data.length;
      }
    }
    return means;
  }

  private checkAlerts(prediction: PredictionLog): void {
    if (prediction.confidence < 0.5) {
      console.warn('⚠ Low confidence prediction:', prediction.label);
    }
    if (prediction.latencyMs > 50) {
      console.warn('⚠ Slow inference:', prediction.latencyMs, 'ms');
    }
  }
}

// Usage
const monitor = new ModelMonitor(trainingFeatures);

// After each prediction
monitor.log({
  timestamp: Date.now(),
  confidence: 0.94,
  latencyMs: 5.2,
  inputFeatures: signalFeatures,
  label: 'pulsar',
});

console.log('Avg confidence:', monitor.avgConfidence.toFixed(3));
console.log('P95 latency:', monitor.getLatencyPercentile(95).toFixed(1), 'ms');`}
</CodeBlock>

## Drift Detection

<CodeBlock language="typescript" title="drift.ts">
{`// Data drift: when production data diverges from training data
// Like monitoring API response shapes — if they change, your app breaks

interface DriftReport {
  driftScore: number;
  isSignificant: boolean;
  features: { name: string; baselineMean: number; currentMean: number; shift: number }[];
}

function detectDrift(
  baseline: number[][],
  current: number[][],
  threshold: number = 0.5
): DriftReport {
  const featureNames = ['frequency', 'amplitude', 'phase', 'snr', 'bandwidth'];
  const features = featureNames.map((name, i) => {
    const baselineMean = baseline.reduce((s, r) => s + r[i], 0) / baseline.length;
    const currentMean = current.reduce((s, r) => s + r[i], 0) / current.length;
    return {
      name,
      baselineMean,
      currentMean,
      shift: Math.abs(currentMean - baselineMean),
    };
  });

  const driftScore = features.reduce((s, f) => s + f.shift, 0) / features.length;

  return {
    driftScore,
    isSignificant: driftScore > threshold,
    features,
  };
}

const drift = detectDrift(trainingData, recentSignals);
if (drift.isSignificant) {
  console.error('🚨 Data drift detected! Score:', drift.driftScore.toFixed(3));
  console.log('Drifted features:', drift.features
    .filter(f => f.shift > 0.1)
    .map(f => f.name)
  );
  // Trigger retraining alert
}`}
