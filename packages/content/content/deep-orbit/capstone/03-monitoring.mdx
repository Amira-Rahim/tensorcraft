      class ModelMonitor {
        private confidences: number[] = [];
        private latencies: number[] = [];
        private baselineMean: number[];

        constructor(baselineMean: number[]) {
          if (baselineMean.length === 0 || !baselineMean.every(Number.isFinite))
            throw new Error('Invalid baseline');
          this.baselineMean = baselineMean.slice();
        }

        logPrediction(confidence: number): void {
          if (!Number.isFinite(confidence) || confidence < 0 || confidence > 1)
            throw new Error('Invalid confidence');
          this.confidences.push(confidence);
        }

        get avgConfidence(): number | null {
          if (this.confidences.length === 0) return null;
          return this.confidences.reduce((a, b) => a + b, 0) / this.confidences.length;
        }

        logLatency(ms: number): void {
          if (!Number.isFinite(ms) || ms < 0) throw new Error('Invalid latency');
          this.latencies.push(ms);
        }

        getLatencyPercentile(p: number): number | null {
          if (!Number.isFinite(p) || p < 0 || p > 100) throw new Error('Invalid percentile');
          if (this.latencies.length === 0) return null;
          const sorted = [...this.latencies].sort((a, b) => a - b);
          const index = Math.max(0, Math.ceil((sorted.length * p) / 100) - 1);
          return sorted[index];
        }

        computeDrift(recentInputs: number[][]): number | null {
          if (recentInputs.length === 0) return null;
          const featureCount = this.baselineMean.length;
          if (!recentInputs.every((row) => row.length === featureCount && row.every(Number.isFinite)))
            throw new Error('Invalid feature row');
          const recentMean = new Array(featureCount).fill(0);

          for (const row of recentInputs) {
            for (let i = 0; i < featureCount; i++) {
              recentMean[i] += row[i] / recentInputs.length;
            }
          }

          return Math.hypot(...recentMean.map((value, i) => value - this.baselineMean[i]));
        }
      }

{`interface PredictionLog {
  timestamp: number;
  confidence: number; // model score; calibration must be measured separately
  latencyMs: number; // awaited end-to-end inference, with readback
  inputFeatures: number[];
  label: string;
}

class ModelMonitor {
  private entries: PredictionLog[] = [];
  private baselineMean: number[];
  constructor(baselineData: number[][]) {
    this.baselineMean = this.mean(baselineData);
  }
  get logs(): PredictionLog[] {
    return this.entries.map(row => ({ ...row, inputFeatures: [...row.inputFeatures] }));
  }
  log(prediction: PredictionLog): void {
    if (!Number.isFinite(prediction.timestamp) || !Number.isFinite(prediction.confidence) ||
        prediction.confidence < 0 || prediction.confidence > 1 ||
        !Number.isFinite(prediction.latencyMs) || prediction.latencyMs < 0 ||
        typeof prediction.label !== 'string' || !prediction.label.trim()) throw new Error('Invalid log');
    this.mean([prediction.inputFeatures], this.baselineMean.length);
    this.entries.push({ ...prediction, inputFeatures: [...prediction.inputFeatures] });
    if (this.entries.length > 1000) this.entries.shift(); // bounded demonstration retention
    // Illustrative alert budgets. Set them from evaluation and operating cost.
    if (prediction.confidence < 0.6) console.warn('Low score:', prediction.label);
    if (prediction.latencyMs > 50) console.warn('Slow inference:', prediction.latencyMs);
  }
  get avgConfidence(): number | null {
    if (!this.entries.length) return null;
    return this.entries.reduce((sum, row) => sum + row.confidence, 0) / this.entries.length;
  }
  getLatencyPercentile(p: number): number | null {
    if (!Number.isFinite(p) || p < 0 || p > 100) throw new Error('Invalid percentile');
    const sorted = this.entries.map(row => row.latencyMs).sort((a,b) => a-b);
    if (!sorted.length) return null;
    return sorted[Math.max(0, Math.ceil(sorted.length * p / 100) - 1)]; // nearest rank; P0=min
  }
  computeDrift(recentInputs: number[][]): number | null {
    if (!recentInputs.length) return null;
    const current = this.mean(recentInputs, this.baselineMean.length);
    return Math.hypot(...current.map((v,i) => v - this.baselineMean[i]));
  }
  private mean(data: number[][], width = data[0]?.length): number[] {
    if (!data.length || !width || !data.every(row => row.length === width && row.every(Number.isFinite))) {
      throw new Error('Invalid feature rows');
    }
    return Array.from({length:width}, (_,i) => data.reduce((sum,row) => sum + row[i]/data.length, 0));
  }
}

## Drift Detection

<CodeBlock language="typescript" title="drift.ts">
{`interface DriftReport {
  driftScore: number;
  thresholdExceeded: boolean; // operational cutoff, not a significance test
  features: { name: string; baselineMean: number; currentMean: number; shift: number }[];
}

function detectDrift(baseline: number[][], current: number[][], threshold: number): DriftReport {
  const names = ['amplitude', 'zeroCrossings', 'phase', 'fft_0', 'fft_1', 'fft_2', 'fft_3', 'fft_4', 'fft_5', 'fft_6', 'fft_7'];
  if (!Number.isFinite(threshold) || threshold < 0 || !baseline.length || !current.length ||
      ![...baseline, ...current].every(row => row.length === names.length && row.every(Number.isFinite))) {
    throw new Error('Invalid drift inputs');
  }
  // Both periods must use the SAME frozen scaler. Per-window normalization
  // would erase the shifts this check is meant to observe.
  const features = names.map((name,i) => {
    const baselineMean = baseline.reduce((s,r) => s + r[i]/baseline.length, 0);
    const currentMean = current.reduce((s,r) => s + r[i]/current.length, 0);
    return {name, baselineMean, currentMean, shift:Math.abs(currentMean-baselineMean)};
  });
  const driftScore = Math.hypot(...features.map(f => f.shift));
  return {driftScore, thresholdExceeded:driftScore > threshold, features};
}

// Application supplies the arrays. 0.5 is a fictional mission cutoff for
// 100-row windows, not a calibrated threshold for another dataset.
const drift = detectDrift(trainingData, recentSignals, 0.5);
if (drift.thresholdExceeded) {
  console.warn('Investigate mean shift:', drift.driftScore);
  console.table(drift.features);
}`}
</CodeBlock>

Accuracy requires trusted outcome labels. In this mission they arrive after manual review, so live scores and drift cannot establish whether a prediction was correct. Track delayed label coverage and compute evaluation metrics when those labels arrive. Some applications receive outcomes quickly; others never obtain reliable labels for every case.

A drift alert calls for investigation. Check sensor calibration, preprocessing versions, missing values and traffic composition before considering retraining. Supervised retraining needs suitable labels and a new evaluation; an unsupervised model has a different training contract. A shifted input mean does not prove that the model has become inaccurate or that its training set must be replaced.

Watch the output distribution too: class proportions and score histograms can change while input means remain stable. Conversely, identical means can hide changes in spread, multimodality or relationships between features. Per-feature distribution comparisons, such as a Kolmogorov-Smirnov statistic, can reveal some of these changes, but their interpretation depends on sample size, dependence and repeated testing. They do not test every kind of joint shift.

Build the monitor: confidence average, latency percentiles off the recorded inference times, and a drift score against the baseline.
