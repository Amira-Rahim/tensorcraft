## ML-Aware CI/CD

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="yaml" title="deploy.yml">
{`name: DEEP SCAN CI/CD

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20

      # Standard web app steps
      - run: npm ci
      - run: npm run lint
      - run: npm run typecheck
      - run: npm test

  validate-model:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - run: npm ci

      # Model artifact: 42 MB of weights, too big for git. Restored from
      # the CI cache; the training rig seeds it after every retrain.
      - name: Restore model artifact
        uses: actions/cache@v4
        with:
          path: models/
          key: deep-scan-model

      # ML-specific: validate model accuracy
      - name: Validate model accuracy
        run: |
          node scripts/validate-model.js \\
            --model models/deep-scan-v1.json \\
            --test-data data/holdout.json \\
            --min-accuracy 0.90

      # ML-specific: benchmark inference speed
      - name: Benchmark inference
        run: |
          node scripts/benchmark.js \\
            --model models/deep-scan-v1.json \\
            --max-inference-ms 20 \\
            --max-memory-mb 50

  deploy:
    runs-on: ubuntu-latest
    needs: [test, validate-model]
    if: github.ref == 'refs/heads/main'
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - run: npm ci
      - run: npm run build
      - uses: peaceiris/actions-gh-pages@v4
        with:
          github_token: \${{ secrets.GITHUB_TOKEN }}
          publish_dir: ./dist`}
</CodeBlock>

## Validation Scripts

<CodeBlock language="typescript" title="validate-model.ts">
{`// Model accuracy gate: blocks deployment if accuracy drops

interface ValidationResult {
  accuracy: number;
  passed: boolean;
  details: { total: number; correct: number; failures: string[] };
}

async function validateModel(
  modelPath: string,
  testDataPath: string,
  minAccuracy: number
): Promise<ValidationResult> {
  const model = await loadModel(modelPath);
  const testData = await loadTestData(testDataPath);

  let correct = 0;
  const failures: string[] = [];

  for (const sample of testData) {
    const prediction = await model.predict(sample.input);
    if (prediction.label === sample.expectedLabel) {
      correct++;
    } else {
      failures.push(
        \`Expected \${sample.expectedLabel}, got \${prediction.label}\`
      );
    }
  }

  const accuracy = correct / testData.length;
  return {
    accuracy,
    passed: accuracy >= minAccuracy,
    details: { total: testData.length, correct, failures: failures.slice(0, 10) },
  };
}

// Usage in CI
const result = await validateModel(
  'models/deep-scan-v1.json',
  'data/holdout.json',
  0.90
);

console.log(\`Accuracy: \${(result.accuracy * 100).toFixed(1)}%\`);
console.log(\`Gate: \${result.passed ? 'PASSED ✓' : 'FAILED ✗'}\`);

if (!result.passed) {
  console.error('Model accuracy below threshold; blocking deployment');
  process.exit(1);
}`}
</CodeBlock>

<RecallPrompt question="Why does a passing CI workflow not guarantee you tested the model you are about to ship?" answer="CI workflows cache artifacts by path or time, not by content hash. If the cache key does not include the model's file hash, a stale weight file from a previous training run can satisfy the cache hit and pass all accuracy gates, measuring a model that no longer exists. You must version weights separately and key the cache on the artifact hash." />

<RecallPrompt
  question="What do ML model gates add to a standard web CI pipeline, and what do they share with failing unit tests?"
  answer="Model gates check accuracy thresholds, inference latency, and memory budget against the candidate artifact. They share the blocking contract with unit tests: any gate failure stops the merge. The difference is what they test: learned behavior that can regress without any code change."
/>
