## Modeling Normality

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="baseline-stats.ts">
{`// Step 1: Define "normal" from known-good data
// Like establishing performance baselines before deploying a change

interface SignalStats {
  mean: number;
  std: number;
  p5: number;   // 5th percentile
  p95: number;  // 95th percentile
}

function computeBaseline(normalSignals: number[]): SignalStats {
  const sorted = [...normalSignals].sort((a, b) => a - b);
  const n = sorted.length;
  const mean = normalSignals.reduce((s, v) => s + v, 0) / n;
  const std = Math.sqrt(normalSignals.reduce((s, v) => s + (v - mean) ** 2, 0) / n);

  return {
    mean,
    std,
    p5: sorted[Math.floor(n * 0.05)],
    p95: sorted[Math.floor(n * 0.95)],
  };
}

// Step 2: Flag deviations
function isAnomaly(value: number, baseline: SignalStats, sigmas: number = 3): boolean {
  const zScore = Math.abs(value - baseline.mean) / baseline.std;
  return zScore > sigmas;  // beyond 3 standard deviations = anomaly
}

const baseline = computeBaseline(normalReadings);
console.log('Baseline:', baseline);
// { mean: 0.52, std: 0.15, p5: 0.28, p95: 0.76 }

console.log(isAnomaly(0.55, baseline));  // false: normal
console.log(isAnomaly(1.82, baseline));  // true: 8.7 sigmas away!`}
</CodeBlock>

## Precision vs Recall

<CodeBlock language="typescript" title="precision-recall.ts">
{`// The threshold is a dial between two types of errors:
// - False positives: normal signals flagged as anomalies (crying wolf)
// - False negatives: real anomalies missed (sleeping on duty)

// Low threshold (2 sigma): catches more anomalies, but more false alarms
// High threshold (4 sigma): fewer false alarms, but might miss subtle anomalies

// Nazari wants zero false positives; she hates wasting time investigating nothing
// Farah wants zero false negatives; she can't afford to miss the signal

// The right threshold depends on the cost of each error type
// For DEEP SCAN: missing an alien signal is worse than a few false alarms
// So we bias toward lower thresholds (higher recall)

interface DetectionResult {
  truePositives: number;   // correctly flagged anomalies
  falsePositives: number;  // normal signals wrongly flagged
  falseNegatives: number;  // anomalies we missed
}

function precision(r: DetectionResult): number {
  return r.truePositives / (r.truePositives + r.falsePositives);
}

function recall(r: DetectionResult): number {
  return r.truePositives / (r.truePositives + r.falseNegatives);
}`}
</CodeBlock>
