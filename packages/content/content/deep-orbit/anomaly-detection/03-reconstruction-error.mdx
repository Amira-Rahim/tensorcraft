## Measuring the Diff

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="reconstruction-error.ts">
{`import * as tf from '@tensorflow/tfjs';

// Compute reconstruction error for each sample
function computeAnomalyScores(
  autoencoder: tf.LayersModel,
  signals: tf.Tensor2D,
): number[] {
  return tf.tidy(() => {
    const reconstructed = autoencoder.predict(signals) as tf.Tensor2D;
    // Per-sample MSE: mean of squared differences across features.
    // Here is where Demir's git-diff analogy stops paying out: a diff
    // names the exact lines that changed and you can patch them. This
    // collapses eleven dimensions into ONE scalar. It says how wrong,
    // never which feature, which position, which channel.
    const errors = tf.mean(tf.square(tf.sub(signals, reconstructed)), 1);
    return Array.from(errors.dataSync());
  });
}

// Tiny helper for the printouts below (Math has no built-in mean)
const mean = (xs: number[]) => xs.reduce((a, b) => a + b, 0) / xs.length;

// Normal signals: low reconstruction error
const normalScores = computeAnomalyScores(autoencoder, normalTestData);
console.log('Normal, mean error:', mean(normalScores).toFixed(6));  // ~0.001
console.log('Normal, max error:', Math.max(...normalScores).toFixed(6));  // ~0.005

// Anomalous signal: high reconstruction error
const anomalyScores = computeAnomalyScores(autoencoder, anomalousData);
console.log('Anomaly, error:', anomalyScores[0].toFixed(6));  // ~0.062
// 12x the normal maximum: this is clearly anomalous`}
</CodeBlock>

## Feature-Level Error Analysis

<CodeBlock language="typescript" title="feature-errors.ts">
{`import * as tf from '@tensorflow/tfjs';

// Which features contribute most to the anomaly?
// This map is Farah's actual answer, and she wants it computed
// before she signs off on any alert.
function featureErrors(
  autoencoder: tf.LayersModel,
  signal: tf.Tensor2D,
  featureNames: string[],
): { feature: string; error: number }[] {
  return tf.tidy(() => {
    const reconstructed = autoencoder.predict(signal) as tf.Tensor2D;
    const errors = tf.square(tf.sub(signal, reconstructed));
    const perFeature = errors.arraySync()[0] as number[];

    return featureNames.map((name, i) => ({
      feature: name,
      error: perFeature[i],
    })).sort((a, b) => b.error - a.error);
  });
}

const analysis = featureErrors(autoencoder, anomalyTensor, [
  'amplitude', 'frequency', 'phase',
  'fft_0', 'fft_1', 'fft_2', 'fft_3', 'fft_4', 'fft_5', 'fft_6', 'fft_7',
]);

console.log('Top contributors:');
// fft_5: 0.6412  ← the 73.4 Hz bin: the anomalous frequency!
// fft_6: 0.0289
// amplitude: 0.0071
// Everything else: < 0.001
// (mean over all 11 = the 0.062 anomaly score from the last block)`}
</CodeBlock>
