      interface Section { heading: string; content: string; validated: boolean; }

      function assembleReport(title: string, sections: Section[]): string {
        const shipped = sections.filter(s => s.validated);
        const withheld = sections.filter(s => !s.validated);
        const parts = shipped.map(s => `## ${s.heading}\n\n${s.content}`);
        if (withheld.length > 0) {
          parts.push(`## Withheld\n\n${withheld.map(s => `- ${s.heading}`).join('\n')}`);
        }
        return `# ${title}\n\n${parts.join('\n\n---\n\n')}`;
      }

## Section-by-Section Generation

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="report-pipeline.ts">
{`// reviewSection is supplied by the application: it shows the draft and
// evidence to a reviewer and waits for an explicit decision.

interface ReportSection {
  heading: string;
  prompt: string;
  data: string;
  generated?: string;
  validated?: boolean;
}

async function generateReport(
  sections: ReportSection[],
  llm: LLMClient,
  reviewSection: (input: { heading: string; draft: string; evidence: string }) => Promise<boolean>,
): Promise<ReportSection[]> {
  // A new draft never inherits approval from an older one.
  const drafts = sections.map(section => ({ ...section, generated: undefined as string | undefined, validated: false }));
  for (const section of drafts) {
    const prompt = \`## Task
Write the "\${section.heading}" section of a scientific signal analysis report.

## Data
\${section.data}

## Requirements
\${section.prompt}

## Format
- Use markdown
- Every number you write has to appear in the Data block above, copied from it exactly
- Follow the section length requested above; do not pad missing evidence
- Be precise and factual\`;

    const content = await llm.complete(prompt, { temperature: 0.2 });
    if (typeof content !== 'string' || !content.trim()) throw new Error('Empty section draft');
    section.generated = content;
    // Lesson 3's structured checks can assist review. They cannot approve
    // free prose, whose claims may have no corresponding numeric field.
    section.validated = (await reviewSection({
      heading: section.heading,
      draft: content,
      evidence: section.data,
    })) === true;
  }

  return drafts;
}

// Integration inputs: anomalyDataSummary, llm and reviewSection come
// from the application. Review the complete report again after assembly.
// Define the report template
const reportSections: ReportSection[] = [
  {
    heading: 'Executive Summary',
    prompt: 'Summarize key findings in 3-4 sentences. Lead with the anomaly discovery.',
    data: 'Total: 12,001 signals. Normal: 11,847. Anomalous: 153. Unclassified: 1.',
  },
  {
    heading: 'Anomaly Analysis',
    prompt: 'Detail the unclassified signal: frequency, timing pattern, Fibonacci structure.',
    data: anomalyDataSummary,
  },
  // ... additional sections
];

const marked = await generateReport(reportSections, llm, reviewSection);
// Assembling the marked sections into a signable document is the challenge below.`}
</CodeBlock>

The pipeline passes `temperature: 0.2` on every section call. For a positive temperature, divide the vocabulary logits by that value before softmax. A low value concentrates the resulting probabilities; the decoding rule then selects a token. This does not establish truth: greedy token selection can still invent an instrument, and serving details can prevent exact repeatability. Farah's Band-3 cluster back in lesson 1 arrived at 0.3. The review step checks claims against the evidence. A number scan alone would miss an invented instrument or a false causal explanation.

## Challenge

The model has written the sections and a reviewer has marked each one. The boolean records that decision; it is not evidence that a numeric matcher proved the prose true. Assemble the document a human can sign: the validated sections in order, and an honest list of what was withheld. The challenge hands each section over as `{ heading, content, validated }`, so what this pipeline stored in `generated` arrives there under the name `content`.
