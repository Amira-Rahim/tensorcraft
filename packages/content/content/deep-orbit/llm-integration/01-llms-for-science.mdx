        const response = await apiCall(prompt);
        const jsonMatch = response.match(/\{[\s\S]*\}/);
        if (!jsonMatch) throw new Error('No JSON in response');
        const parsed = JSON.parse(jsonMatch[0]);
        if (!['noise', 'anomaly', 'unknown'].includes(parsed.classification) || !Number.isFinite(parsed.confidence) || parsed.confidence < 0 || parsed.confidence > 1 || typeof parsed.reasoning !== 'string' || parsed.reasoning.trim().length === 0) {

{`interface LLMClient {
  complete(prompt: string, options?: { system?: string; temperature?: number; maxTokens?: number }): Promise<string>;
}

// Server-side adapter for an endpoint that supports this chat-completions
// contract. Other APIs need their own adapter. Configure a real model ID.
// Keep provider credentials on the server. Browser code calls your
// authenticated server route without receiving the provider key.
function createLLMClient(apiKey: string, baseUrl: string, modelId: string): LLMClient {
  return {
    async complete(prompt, options = {}) {
      const response = await fetch(\`\${baseUrl.replace(/\\/$/, '')}/v1/chat/completions\`, {
        method: 'POST',
        signal: AbortSignal.timeout(30_000),
        headers: { 'Content-Type': 'application/json', Authorization: \`Bearer \${apiKey}\` },
        body: JSON.stringify({
          model: modelId,
          messages: [
            { role: 'system', content: options.system ?? 'Report only what the supplied evidence supports. State missing evidence.' },
            { role: 'user', content: prompt },
          ],
          temperature: options.temperature ?? 0.3,
          max_tokens: options.maxTokens ?? 1024,
        }),
      });
      if (!response.ok) throw new Error(\`LLM request failed: HTTP \${response.status}\`);
      const data = await response.json();
      const content = data?.choices?.[0]?.message?.content;
      if (typeof content !== 'string' || !content.trim()) throw new Error('Missing text response');
      return content; // text validated; claims still unchecked
    },
  };
}

## What You Are Actually Calling

A decoder language model predicts the next token from preceding tokens. Tokens are text pieces; their size depends on the tokenizer and language. The model produces vocabulary logits, and softmax converts them into a distribution. A decoding rule selects a token, appends it and continues until a stop condition.

Inference usually has two phases. **Prefill** processes the supplied prompt. **Decoding** generates new tokens, commonly one at a time while reusing cached attention keys and values. It does not normally recompute the entire prompt for every token. Some serving systems generate candidates in larger groups, then verify them. Token count alone therefore does not tell you the number or duration of runtime calls.

The connection to next-value prediction is the conditional prediction task. Here the output is a token category instead of three continuous sensor values. The model and decoding procedure differ from the LSTM you built.

## What a Transformer Does With the Prompt

In dense self-attention, each token has a query, key and value produced by learned projections. Every query is compared with the allowed keys. A decoder's causal mask blocks access to future tokens. An explicit T-by-T score matrix has 6,400 entries at T = 80 and 25,600 at T = 160. That quadratic scaling matters for dense attention over a full prompt; optimized kernels can avoid storing the entire matrix, and cached decoding has a different cost.

Each head uses its own learned projections. Head outputs are combined, then passed through blocks that also contain feed-forward layers, residual connections and normalization. Attention is one component of a transformer, not the whole architecture. A model's supported context length is a limit on the token sequence; keep room for its answer as well as its prompt.

For positive sampling **temperature** T, divide the vocabulary logits by T before softmax. Smaller values concentrate probability on higher-scoring tokens. Larger values flatten it. A zero-temperature API setting commonly requests greedy selection rather than division by zero; ties and serving details can still affect repeatability. This setting does not change the attention softmax inside the model or verify a claim.

Next-token training rewards predictions of text in the training distribution. Later training can improve instruction following and factual performance, but neither objective checks a new report against your ship's measurements. A fluent answer still needs evidence.

The [original transformer paper](https://arxiv.org/html/1706.03762v7#S3) describes the attention and block structure. Hugging Face’s [cache explanation](https://huggingface.co/docs/transformers/v4.57.1/cache_explanation) shows how cached keys and values are reused during decoding.

The onboard language component and its bundle sizes belong to the fictional mission. This adapter demonstrates the HTTP boundary; it does not download or implement that component. The browser runtime lab measures a separate pretrained sentiment model.

## The Human-AI Workflow

<CodeBlock language="typescript" title="workflow.ts">
{`// The workflow mirrors Copilot usage:
// 1. CONTEXT: provide data and instructions
// 2. GENERATE: LLM produces a draft
// 3. REVIEW: expert checks for accuracy
// 4. CORRECT: fix any errors or hallucinations
// 5. APPROVE: human signs off on final output

interface AnalysisWorkflow {
  context: string;      // raw data and instructions
  draft: string;        // LLM-generated output
  corrections: string[];// expert corrections
  approved: boolean;    // human sign-off
  approvedBy: string;   // who verified
}

// CRITICAL: LLMs can hallucinate, generating plausible but false statements
// Always verify claims against actual data
// Like reviewing Copilot suggestions: accept what's correct, fix what's not

function screenNumbers(
  draft: string,
  actualData: Record<string, number>,
): { matched: number[]; unmatched: number[] } {
  // A limited numeric screen, not factual verification. It finds numbers
  // absent from the data but ignores source fields, units and meaning.
  // Even a matched value can support a false sentence.
  const claims = draft.replace(/(\\d),(?=\\d)/g, '$1').match(/-?\\d+(?:\\.\\d+)?/g) ?? [];
  const values = Object.values(actualData);
  const matched: number[] = [];
  const unmatched: number[] = [];

  for (const claim of claims) {
    const n = parseFloat(claim);   // keep the number, not the matched text
    if (values.some((v) => Math.abs(v - n) <= 0.05)) matched.push(n);
    else unmatched.push(n);
  }
  return { matched, unmatched };
}

// Run it on a draft with one invented figure:
const check = screenNumbers(
  'Across 12,001 catalogued events, 153 were anomalous and 1 resists ' +
  'classification. A cluster of 9 anomalies repeats hourly.',
  { totalEvents: 12001, anomalous: 153, unclassified: 1 },
);
console.log(check.matched);    // [12001, 153, 1]
console.log(check.unmatched);  // [9]  ← no such cluster in the data
// Lesson 3 checks structured claims against named source fields.
// Neither this numeric screen nor those field checks can approve free text.`}
</CodeBlock>
