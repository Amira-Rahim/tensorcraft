{`// Setting up an LLM client — like configuring any API service
interface LLMClient {
  complete(prompt: string, options?: { temperature?: number; maxTokens?: number }): Promise<string>;
}

// In production, this calls an API (OpenAI, Anthropic, etc.)
// For DEEP SCAN, we use the ship's onboard model
async function createLLMClient(apiKey: string, baseUrl: string): Promise<LLMClient> {
  return {
    async complete(prompt, options = {}) {
      const response = await fetch(\`\${baseUrl}/v1/completions\`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': \`Bearer \${apiKey}\`,
        },
        body: JSON.stringify({
          prompt,
          temperature: options.temperature ?? 0.3,  // low temp for factual analysis
          max_tokens: options.maxTokens ?? 1024,
        }),
      });
      const data = await response.json();
      return data.choices[0].text;
    },
  };
}

// Usage: draft a signal analysis report
const llm = await createLLMClient(API_KEY, BASE_URL);
const report = await llm.complete(\`
Summarize the following signal classification results:
- Total signals analyzed: 12,000
- Normal signals: 11,847 (98.7%)
- Anomalous signals: 153 (1.3%)
- Unclassified: 1 (0.008%)

Focus on the single unclassified signal and its significance.
\`);`}
