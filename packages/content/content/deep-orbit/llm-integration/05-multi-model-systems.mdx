      interface Verdict {
        class: string;
        confidence: number;
      }

      interface PipelineResult {
        classification: string;
        confidence: number;
        isAnomaly: boolean;
        alert: string | null;
      }

      const RANGES = {
        amplitude: { min: 0, max: 2048 },
        frequency: { min: 0, max: 63 },
        phase: { min: -Math.PI, max: Math.PI },
      };

      function classifierStage(model: tf.LayersModel): (s: number[]) => Promise<Verdict> {
        const order = [RANGES.amplitude, RANGES.frequency, RANGES.phase];
        return async (reading: number[]) => {
          if (reading.length !== order.length) {
            throw new Error('the classifier reads ' + order.length + ' features and was handed ' + reading.length);
          }
          if (!reading.every(Number.isFinite)) throw new Error('Non-finite reading');
          const features = reading.map((v, i) => (v - order[i].min) / (order[i].max - order[i].min));
          return tf.tidy(() => {
            const row = tf.tensor2d([features]);
            const output = (model.predict(row) as tf.Tensor).dataSync();
            if (output.length !== 1 || !Number.isFinite(output[0]) || output[0] < 0 || output[0] > 1) throw new Error('Invalid sigmoid output');
            const p = output[0];
            return { class: p > 0.5 ? 'anomaly' : 'noise', confidence: p > 0.5 ? p : 1 - p };
          });
        };
      }

      async function pipeline(
        signal: number[],
        classify: (s: number[]) => Promise<Verdict>,
        detectAnomaly: (s: number[], cls: string) => Promise<boolean>,
        generateAlert: (s: number[], cls: string) => Promise<string>,
      ): Promise<PipelineResult> {
        const verdict = await classify(signal);
        const confidence = verdict.confidence;
        if (!Number.isFinite(confidence) || confidence < 0 || confidence > 1) {
          throw new Error('classifier broke its contract');
        }
        const classification = ['noise', 'anomaly'].includes(verdict.class)
          ? verdict.class
          : 'unknown';
        const isAnomaly = await detectAnomaly(signal, classification);
        if (typeof isAnomaly !== 'boolean') throw new Error('detector must return a boolean');
        const alert = isAnomaly ? await generateAlert(signal, classification) : null;
        if (isAnomaly && (typeof alert !== 'string' || alert.trim().length === 0)) throw new Error('alert must be a non-empty string');
        return { classification, confidence, isAnomaly, alert };
      }

<Dialogue speaker="Ensign Demir">
The classifier returns a class, the autoencoder rebuilds a reading, and the LSTM predicts what comes next. The LLM turns their results into a draft. I need to connect them, like services behind an API gateway.
</Dialogue>

Demir needs a checked handoff between each stage. The classifier stage supplies a class and score; the autoencoder stage measures reconstruction error; the LSTM supplies a forecast. The reporter needs to know when any of those results is unavailable. Connecting these stages and handling their failures is **model orchestration**.

Suppose four stages each have a 95% chance of being correct, independently. The chance that all four are correct is 0.95^4, about 81%.

That is not automatically the final system's accuracy. Some stages may recover from earlier errors, and some outputs may not affect the result. Shared inputs can also link errors across stages. Check each stage's input and output, then test the full pipeline against real outcomes. A plausible value can pass every type and range check and still be wrong.

Demir needs one result the watch can use, with failures caught before they reach the alert writer. You will connect the classifier, detector, and alert generator and inspect what each handoff allows through.

<CodeBlock language="typescript" title="pipeline.ts">
{`import * as tf from '@tensorflow/tfjs';

interface PipelineResult {
  classification: { class: string; confidence: number };
  anomaly: { isAnomaly: boolean; score: number; method: 'autoencoder-mse' | 'amplitude-z-score'; threshold: number };
  prediction: { nextValues: number[] }; // empty when unavailable
  report: string | null;
  draftForReview: string | null;
}
type Models = { classifier: tf.LayersModel; autoencoder: tf.LayersModel; lstm: tf.LayersModel; llm: LLMClient };

function requireVector(values: number[], width: number) {
  if (values.length !== width || !values.every(Number.isFinite)) throw new Error('Invalid feature vector');
}

function readClassifier(features: number[], model: tf.LayersModel) {
  requireVector(features, 3);
  return tf.tidy(() => {
    const output = (model.predict(tf.tensor2d([features])) as tf.Tensor).dataSync();
    if (output.length !== 1 || !Number.isFinite(output[0]) || output[0] < 0 || output[0] > 1) {
      throw new Error('Invalid classifier output');
    }
    const p = output[0];
    // Chosen-class score, not a demonstrated probability of correctness.
    return { class: p > 0.5 ? 'anomaly' : 'noise', confidence: p > 0.5 ? p : 1 - p };
  });
}

function readAutoencoder(signal: number[], model: tf.LayersModel, threshold: number): PipelineResult['anomaly'] {
  requireVector(signal, 11);
  if (!Number.isFinite(threshold) || threshold < 0) throw new Error('Invalid MSE threshold');
  return tf.tidy(() => {
    const input = tf.tensor2d([signal]);
    const reconstruction = model.predict(input) as tf.Tensor;
    if (reconstruction.shape.length !== 2 || reconstruction.shape[0] !== 1 || reconstruction.shape[1] !== 11) {
      throw new Error('Invalid reconstruction shape');
    }
    const score = tf.losses.meanSquaredError(input, reconstruction).dataSync()[0];
    if (!Number.isFinite(score)) throw new Error('Invalid reconstruction error');
    return { isAnomaly: score > threshold, score, method: 'autoencoder-mse' as const, threshold };
  });
}

function optionalForecast(window: number[][], model: tf.LayersModel): PipelineResult['prediction'] {
  try {
    if (window.length !== 80) throw new Error('Expected 80 steps');
    window.forEach(row => requireVector(row, 3));
    return tf.tidy(() => {
      const output = (model.predict(tf.tensor3d([window])) as tf.Tensor).dataSync();
      const nextValues = Array.from(output);
      requireVector(nextValues, 3);
      return { nextValues };
    });
  } catch {
    console.warn('Forecast unavailable; alert will identify the omission.');
    return { nextValues: [] };
  }
}

async function finishAlert(
  classification: PipelineResult['classification'],
  anomaly: PipelineResult['anomaly'],
  prediction: PipelineResult['prediction'],
  llm: LLMClient,
): Promise<PipelineResult> {
  let report: string | null = null;
  let draftForReview: string | null = null;
  if (anomaly.isAnomaly) {
    report = \`ALERT: \${anomaly.method} score \${anomaly.score.toFixed(4)} exceeds \${anomaly.threshold}.\`;
    if (!prediction.nextValues.length) report += ' Forecast unavailable.';
    if (anomaly.method === 'amplitude-z-score') report += ' Reduced coverage: amplitude only.';
    try {
      const draft = await llm.complete(JSON.stringify({
        task: 'Draft an explanation for review. Do not infer a cause.', classification, anomaly, prediction,
      }));
      if (typeof draft !== 'string' || !draft.trim()) throw new Error('Empty draft');
      draftForReview = draft;
    } catch {
      console.warn('Draft unavailable; retain the measured alert.');
    }
  }
  return { classification, anomaly, prediction, report, draftForReview };
}

async function processSignal(
  features: number[], // classifier: 3 min-max features, module 4 scaler
  signal: number[], // autoencoder: 11 z-scores, its training-period scaler
  window: number[][], // LSTM: 80 x 3 z-scores, its own training-period scaler
  models: Models,
  thresholds: { anomaly: number },
): Promise<PipelineResult> {
  const classification = readClassifier(features, models.classifier);
  const anomaly = readAutoencoder(signal, models.autoencoder, thresholds.anomaly);
  const prediction = optionalForecast(window, models.lstm);
  return finishAlert(classification, anomaly, prediction, models.llm);
}`}
</CodeBlock>

Each model needs a different version of the same capture. The classifier takes module 4's three min-max-scaled features. The autoencoder takes module 5's eleven z-scored features. The LSTM takes a window of module 6's per-channel z-scores.

A wrong length is easy to catch. Give the classifier eleven features and TensorFlow.js stops before prediction:

`expected dense_Dense1_input to have shape [null,3] but got array with shape [1,11]`

The right length on the wrong scale is harder to catch. Three z-scores fit the classifier's input shape. It can return a class score inside the valid 0..1 range even though the input is wrong for that model. A range check will not expose the scaling mistake.

Load each model with metadata identifying its preprocessing version. These guards cannot prove which scaler produced an array. Each stage takes its own input because the format and scale are part of that stage's requirements.

## Error Handling and Fallbacks

The classifier and anomaly-score checks stop the run when their values are invalid. A failed forecast can be omitted. The alert itself uses a fixed template; generated prose stays in draftForReview until a reviewer checks it. If drafting fails, the alert still ships. The resilience sketch below also shows a weaker detector fallback when the autoencoder is unavailable.

The alert identifies the detector, its score type and its threshold. An MSE and a z-score are different quantities; changing detectors silently would mislead the operator. Even a normal fallback result carries its method so the UI can show reduced coverage. These examples share helpers and require the LLMClient interface from lesson 1.

The fallback checks amplitude alone against a fixed normal-period baseline. It cannot detect every relationship the eleven-feature autoencoder might detect. In the fictional baseline, mean 0.52 and standard deviation 0.15 give a 3-sigma boundary at 0.07 and 0.97.

For valid sensor readings scaled into [0, 1], the largest absolute z-score is about 3.47 at zero. That bound depends on the physical range and this fixed baseline, not on min-max scaling alone. The helpers check shape and finite values; an application must also validate sensor ranges.

Refitting the baseline changes the score even if the reading stays fixed. Holding the mean at 0.52, a reading of 1.00 scores 342.86 with standard deviation 0.0014, but 2.29 with standard deviation 0.21. Pass the approved baseline explicitly so an unusually quiet or noisy window cannot silently replace it.

<CodeBlock language="typescript" title="resilience.ts">
{`// Continue in the same module: reuse the checked stage helpers above.
async function resilientPipeline(
  features: number[], signal: number[], window: number[][],
  models: Models, thresholds: { anomaly: number },
  baseline: { mean: number; std: number }, // fixed normal-period amplitude stats, min-max units
): Promise<PipelineResult> {
  const classification = readClassifier(features, models.classifier);
  // A bad threshold is a configuration error, not a reason to switch models.
  if (!Number.isFinite(thresholds.anomaly) || thresholds.anomaly < 0) throw new Error('Invalid threshold');
  let anomaly: PipelineResult['anomaly'];
  try {
    anomaly = readAutoencoder(signal, models.autoencoder, thresholds.anomaly);
  } catch {
    if (!Number.isFinite(baseline.mean) || !Number.isFinite(baseline.std) || baseline.std <= 0) {
      throw new Error('No usable amplitude fallback baseline');
    }
    const score = Math.abs(features[0] - baseline.mean) / baseline.std;
    if (!Number.isFinite(score)) throw new Error('Invalid fallback score');
    anomaly = { isAnomaly: score > 3, score, method: 'amplitude-z-score', threshold: 3 };
  }
  const prediction = optionalForecast(window, models.lstm);
  return finishAlert(classification, anomaly, prediction, models.llm);
}`}
