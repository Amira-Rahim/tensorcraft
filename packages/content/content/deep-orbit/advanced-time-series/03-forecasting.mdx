      async function forecastTwoWays(trainX, trainY, past, arrived, epochs, providedModel) {
        const windowSize = trainX.shape[1];
        const model = providedModel || tf.sequential();
        if (!providedModel) {
          model.add(tf.layers.lstm({ units: 8, inputShape: [windowSize, 1] }));
          model.add(tf.layers.dense({ units: 1 }));
          model.compile({ optimizer: 'adam', loss: 'meanSquaredError' });
        }
        try {
          const history = await model.fit(trainX, trainY, { epochs, batchSize: 8 });

          const predictFrom = (readings) =>
            tf.tidy(() => {
              const window = readings.slice(-windowSize).map((value) => [value]);
              const output = model.predict(tf.tensor3d([window])).dataSync();
              if (output.length !== 1 || !Number.isFinite(output[0])) throw new Error('Invalid prediction');
              return output[0];
            });

          const fedBack = [...past];
          const rollout = [];
          for (let step = 0; step < arrived.length; step++) {
            const next = predictFrom(fedBack);
            rollout.push(next);
            fedBack.push(next);
          }

          const oneStep = [];
          for (let step = 0; step < arrived.length; step++) {
            oneStep.push(predictFrom([...past, ...arrived.slice(0, step)]));
          }

          return { model, history, rollout, oneStep };
        } catch (error) {
          if (!providedModel) model.dispose();
          throw error;
        }
      }

<CodeBlock language="typescript" title="forecasting.ts">
{`import * as tf from '@tensorflow/tfjs';

// The rolling loop from module 6, lesson 5: predict one step, clamp it,
// append it, slide the window, repeat. What matters this time is what its
// error does as the horizon grows.

// Same sensor spec module 6 clamped against: amplitude in counts,
// "frequency" as a zero-crossing count over a 64-sample window, phase in
// radians.
const LIMITS = [
  { min: 0, max: 2048 },
  { min: 0, max: 63 },
  { min: -Math.PI, max: Math.PI },
];

// The model predicts per-channel z-scores. Convert them to sensor units for reports.
// Convert physical limits using the same saved training MEAN and STD
// before clamping. A count limit cannot be compared directly with a z-score.
// savedScaler must come from this model's training rows.
const { mean: MEAN, std: STD } = savedScaler;
if (MEAN.length !== 3 || STD.length !== 3 || !MEAN.every(Number.isFinite) ||
    !STD.every(v => Number.isFinite(v) && v > 0)) throw new Error('Invalid scaler');
const bounds = LIMITS.map((limit, j) => ({
  min: (limit.min - MEAN[j]) / STD[j],
  max: (limit.max - MEAN[j]) / STD[j],
}));

const clamp = (reading: number[]) =>
  reading.map((v, j) => Math.min(Math.max(v, bounds[j].min), bounds[j].max));

async function forecast(
  model: tf.LayersModel,
  seed: number[][],
  horizon: number,
  windowSize: number,
): Promise<number[][]> {
  if (!Number.isInteger(horizon) || horizon < 0 || !Number.isInteger(windowSize) || windowSize < 1 || seed.length < windowSize) throw new Error('Invalid forecast window');
  const window = [...seed];
  const predictions: number[][] = [];

  for (let step = 0; step < horizon; step++) {
    const nextValue = tf.tidy(() => {
      const input = tf.tensor3d([window.slice(-windowSize)]);
      const pred = model.predict(input) as tf.Tensor;
      return Array.from(pred.dataSync());
    });

    // Clamp before the row re-enters the window. Step + 1 reads it as
    // history, so one out-of-range value rides the rest of the run.
    if (nextValue.length !== 3 || !nextValue.every(Number.isFinite)) throw new Error('Invalid prediction');
    const bounded = clamp(nextValue);
    predictions.push(bounded);
    window.push(bounded);
  }

  return predictions;
}

// Forecast 100 steps ahead. Step 1 uses observed data. By step 100,
// all 80 inputs are earlier predictions. The function does not produce
// a confidence score. A prediction interval needs held-out error quantiles
// for each horizon and a coverage check. MAE alone is not an interval.
const predictions = await forecast(model, recentWindow, 100, 80);`}
</CodeBlock>

## When the pattern changes

Non-stationarity means a series' statistical properties change over time. Some forecasting methods explicitly model trend or seasonality, so forecasting does not require a constant mean. The practical question is whether the relationships learned by this model remain useful.

Rolling forecasts and changes in the data can both affect error. Compare errors at the same horizon across earlier and later periods. One-step replay is a useful starting point because each prediction uses observed history. Rising error is a reason to investigate data quality, preprocessing, operating conditions and the model. It does not establish a cause by itself.

<CodeBlock language="typescript" title="non-stationary.ts">
{`// The forecast says the anomaly repeats at step 50
// It arrives at step 47: three steps EARLY
// The model was wrong. Why?

// NON-STATIONARY DATA: the statistical properties change over time
// The signal's timing is DRIFTING. It's not perfectly periodic
// Like a website's traffic patterns shifting after a redesign

// One prediction per reading that actually arrived. The window is re-seeded
// from real data every step, so no prediction ever feeds back into the next
// one and every error below costs exactly one model call.
function predictOneStepAhead(
  model: tf.LayersModel,
  observed: number[][],
  windowSize: number,
): number[][] {
  const predictions: number[][] = [];

  for (let t = windowSize; t < observed.length; t++) {
    predictions.push(tf.tidy(() => {
      const input = tf.tensor3d([observed.slice(t - windowSize, t)]);
      const pred = model.predict(input) as tf.Tensor;
      return Array.from(pred.dataSync());
    }));
  }

  return predictions;
}

// Compare one-step errors from recent and earlier time periods.
// Keep the forecast horizon the same. Comparing short- and long-range
// errors from one rollout would test a different thing.
function detectDrift(
  oneStepPredicted: number[][],
  actual: number[][],
  recentPeriod: number,
): { driftDetected: boolean; driftMagnitude: number | null } {
  if (!Number.isInteger(recentPeriod) || recentPeriod < 1 || actual.length <= recentPeriod ||
      oneStepPredicted.length !== actual.length || actual.some((row, i) => !row.length ||
        row.length !== oneStepPredicted[i].length || !row.every(Number.isFinite) ||
        !oneStepPredicted[i].every(Number.isFinite))) throw new Error('Invalid error comparison');
  const errors: number[] = [];
  for (let i = 0; i < actual.length; i++) {
    const err = actual[i].reduce(
      (sum, v, j) => sum + (v - oneStepPredicted[i][j]) ** 2,
      0,
    );
    errors.push(err);
  }

  // This period's errors against the earlier ones
  const recent = errors.slice(-recentPeriod);
  const historical = errors.slice(0, -recentPeriod);
  // No baseline, no comparison. Without this the mean of an empty slice is
  // NaN, driftMagnitude comes back NaN, and the function answers "no drift"
  // on a stream where drift cannot be measured at all.
  if (historical.length === 0) {
    throw new Error('drift needs a baseline: give it more readings than recentPeriod');
  }
  const recentMean = recent.reduce((a, b) => a + b, 0) / recent.length;
  const histMean = historical.reduce((a, b) => a + b, 0) / historical.length;

  return {
    driftDetected: recentMean > histMean * 2, // heuristic flag that requests review
    driftMagnitude: histMean === 0 ? null : recentMean / histMean,
  };
}

// Observations only, never a forecast. The last 200 errors are this period.
// everything before them is the baseline they are judged against.
const oneStep = predictOneStepAhead(model, observedReadings, 80);
const drift = detectDrift(oneStep, observedReadings.slice(80), 200);`}
</CodeBlock>

## A Reproducible Feedback Example

Use a fixed mathematical rule to see the effect of feeding predictions back without any variation from training. The example process uses 3.9. The forecaster uses 3.899. The rollout reuses each prediction. The replay uses the previous observed value instead.

<CodeBlock language="javascript" title="error-by-horizon.ts">
{`const processStep = x => 3.9 * x * (1 - x);
const forecastStep = x => 3.899 * x * (1 - x);
let actualValue = 0.4, rollingValue = 0.4;
const errorsByHorizon = [];
for (let horizon = 1; horizon <= 16; horizon++) {
  const oneStepValue = forecastStep(actualValue);
  actualValue = processStep(actualValue);
  rollingValue = forecastStep(rollingValue);
  errorsByHorizon.push({
    horizon,
    rollout: Math.abs(actualValue - rollingValue),
    replay: Math.abs(actualValue - oneStepValue),
  });
}
console.table(errorsByHorizon);`}
</CodeBlock>

We chose this numerical example to make error accumulation visible. It does not measure an LSTM or a spacecraft signal. The replay error is at most 0.00025 here: the coefficient difference is 0.001 and x(1-x) is at most 0.25 on [0, 1]. The rollout can depart much farther as it feeds its errors back. Inspect the table: individual errors need not rise at every horizon.

To evaluate a learned model, repeat forecasts from many held-out starting points, compare a simple baseline, and report each horizon separately. Mean absolute error is not a prediction interval. An interval needs an explicit construction and a coverage check on later data.
