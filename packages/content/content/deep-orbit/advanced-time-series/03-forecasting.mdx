## Multi-Step Forecasting

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="forecasting.ts">
{`import * as tf from '@tensorflow/tfjs';

async function forecast(
  model: tf.LayersModel,
  seed: number[][],
  horizon: number,
): Promise<{ predictions: number[][]; trustHeuristic: number[] }> {
  const window = [...seed];
  const predictions: number[][] = [];
  const trustHeuristic: number[] = [];

  for (let step = 0; step < horizon; step++) {
    const nextValue = tf.tidy(() => {
      const input = tf.tensor3d([window.slice(-80)]);
      const pred = model.predict(input) as tf.Tensor;
      return Array.from(pred.dataSync());
    });

    predictions.push(nextValue);
    window.push(nextValue);

    // Placeholder heuristic, NOT a real confidence interval: a hand-tuned
    // linear decay that encodes "trust later steps less." A measured
    // interval would come from the model's error distribution.
    trustHeuristic.push(Math.max(0, 1 - step * 0.015));
  }

  return { predictions, trustHeuristic };
}

// Forecast 100 steps ahead
const result = await forecast(model, recentWindow, 100);
console.log('Step 1 trust:', result.trustHeuristic[0]);    // 1.0 (decay starts from full trust)
console.log('Step 50 trust:', result.trustHeuristic[49]);  // 0.265
console.log('Step 100 trust:', result.trustHeuristic[99]); // 0.0: too far ahead`}
</CodeBlock>

## Non-Stationarity: When Predictions Fail

<CodeBlock language="typescript" title="non-stationary.ts">
{`// The forecast says the anomaly repeats at step 50
// It arrives at step 47: three steps EARLY
// The model was wrong. Why?

// NON-STATIONARY DATA: the statistical properties change over time
// The signal's timing is DRIFTING; it's not perfectly periodic
// Like a website's traffic patterns shifting after a redesign

// Detection: compare predicted vs actual over a moving window
function detectDrift(
  predicted: number[][],
  actual: number[][],
  windowSize: number,
): { driftDetected: boolean; driftMagnitude: number } {
  const errors: number[] = [];
  for (let i = 0; i < actual.length; i++) {
    const err = actual[i].reduce((sum, v, j) => sum + (v - predicted[i][j]) ** 2, 0);
    errors.push(err);
  }

  // Compare recent errors to historical errors
  const recent = errors.slice(-windowSize);
  const historical = errors.slice(0, -windowSize);
  const recentMean = recent.reduce((a, b) => a + b, 0) / recent.length;
  const histMean = historical.reduce((a, b) => a + b, 0) / historical.length;

  return {
    driftDetected: recentMean > histMean * 2,  // errors doubled
    driftMagnitude: recentMean / histMean,
  };
}`}
</CodeBlock>
