<Dialogue speaker="Ensign Demir">
The LSTM is great at short-range patterns. The attention model catches long-range dependencies. The Kalman filter tracks drift. Why choose one when we can combine all three?
</Dialogue>

<Dialogue speaker="Dr. Farah">
Because all three run on every input. An ensemble isn't a router that picks the right specialist. Every model fires every time, and you pay the sum of their inference costs. If the compute budget covers that, then yes: each model has blind spots, and the ensemble covers them.
</Dialogue>

Micro-frontends compose independent apps into one view by partitioning the work: each app owns its route. The analogy bends exactly where Farah said it would. An ensemble doesn't partition the work; it multiplies it per request.

<CodeBlock language="typescript" title="ensemble.ts">
{`import * as tf from '@tensorflow/tfjs';

interface ModelPrediction {
  modelName: string;
  prediction: number[];
  confidence: number;
}

// Simple averaging: each model gets equal vote
function averageEnsemble(predictions: ModelPrediction[]): number[] {
  const n = predictions.length;
  const dim = predictions[0].prediction.length;
  const result = new Array(dim).fill(0);

  for (const p of predictions) {
    for (let i = 0; i < dim; i++) {
      result[i] += p.prediction[i] / n;
    }
  }
  return result;
}

// Weighted averaging: trust better models more
function weightedEnsemble(predictions: ModelPrediction[]): number[] {
  const totalConf = predictions.reduce((s, p) => s + p.confidence, 0);
  const dim = predictions[0].prediction.length;
  const result = new Array(dim).fill(0);

  for (const p of predictions) {
    const weight = p.confidence / totalConf;
    for (let i = 0; i < dim; i++) {
      result[i] += p.prediction[i] * weight;
    }
  }
  return result;
}

// DEEP SCAN ensemble
const predictions: ModelPrediction[] = [
  { modelName: 'LSTM', prediction: [0.82, 74.1, 0.45], confidence: 0.85 },
  { modelName: 'Attention', prediction: [0.79, 73.8, 0.51], confidence: 0.90 },
  { modelName: 'Kalman', prediction: [0.80, 74.3, 0.48], confidence: 0.75 },
];

const avg = averageEnsemble(predictions);
const weighted = weightedEnsemble(predictions);
console.log('Average:', avg);    // [0.803, 74.067, 0.480]
console.log('Weighted:', weighted); // attention contributes most`}

## Why Ensembles Work

<CodeBlock language="typescript" title="why-ensembles.ts">
{`// Each model makes different errors
// When you average, individual errors cancel out

// LSTM:      good at short patterns, misses long-range drift
// Attention: good at long-range, may overfit on small datasets
// Kalman:    good at tracking drift, assumes linear dynamics

// Error reduction: if models have uncorrelated errors,
// ensemble error ≈ individual error / sqrt(N)

// Example:
// LSTM MAE: 0.045
// Attention MAE: 0.038
// Kalman MAE: 0.052
// Ensemble MAE: 0.028  ← better than any individual model

function evaluateEnsemble(
  models: { name: string; predictions: number[][] }[],
  actual: number[][],
): { individual: Record<string, number>; ensemble: number } {
  const individual: Record<string, number> = {};

  for (const model of models) {
    let totalError = 0;
    for (let i = 0; i < actual.length; i++) {
      for (let j = 0; j < actual[i].length; j++) {
        totalError += Math.abs(model.predictions[i][j] - actual[i][j]);
      }
    }
    individual[model.name] = totalError / (actual.length * actual[0].length);
  }

  // Ensemble prediction
  let ensembleError = 0;
  for (let i = 0; i < actual.length; i++) {
    for (let j = 0; j < actual[i].length; j++) {
      const avg = models.reduce((s, m) => s + m.predictions[i][j], 0) / models.length;
      ensembleError += Math.abs(avg - actual[i][j]);
    }
  }

  return { individual, ensemble: ensembleError / (actual.length * actual[0].length) };
}`}
</CodeBlock>
