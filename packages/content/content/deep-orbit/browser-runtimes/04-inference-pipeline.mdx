Leak two tensors per reading and the dashboard is dead before the next shift change.

## The Real-Time Inference Loop

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="inference-pipeline.ts">
{`import * as tf from '@tensorflow/tfjs';

// The model expects normalized 3-feature input: [amplitude, frequency, phase]
// Training used min-max normalization to [0, 1]
const FEATURE_RANGES = {
  amplitude: { min: 0, max: 2048 },
  frequency: { min: 20, max: 20000 },
  phase: { min: -Math.PI, max: Math.PI },
};

function normalize(value: number, min: number, max: number): number {
  return (value - min) / (max - min);
}

// Preprocess raw signal → model input
function preprocess(raw: { amplitude: number; frequency: number; phase: number }): number[] {
  return [
    normalize(raw.amplitude, FEATURE_RANGES.amplitude.min, FEATURE_RANGES.amplitude.max),
    normalize(raw.frequency, FEATURE_RANGES.frequency.min, FEATURE_RANGES.frequency.max),
    normalize(raw.phase, FEATURE_RANGES.phase.min, FEATURE_RANGES.phase.max),
  ];
}

// Run inference: the core predict call
function classify(model: tf.LayersModel, rawSignal: { amplitude: number; frequency: number; phase: number }) {
  // tf.tidy() automatically disposes all intermediate tensors
  // Like a try/finally that cleans up resources; no manual dispose() needed
  return tf.tidy(() => {
    const normalized = preprocess(rawSignal);
    const input = tf.tensor2d([normalized]);        // shape: [1, 3]
    const output = model.predict(input) as tf.Tensor;
    const confidence = output.dataSync()[0];         // synchronous read

    return {
      class: confidence > 0.5 ? 'anomaly' : 'noise',
      confidence: Math.round(confidence * 1000) / 1000,
    };
  });
}`}
</CodeBlock>

## Memory Management with tf.tidy()

This is the one concept with no frontend parallel. GPU tensors are invisible to the garbage collector; you free them yourself or they stay allocated until the tab dies.

<CodeBlock language="typescript" title="memory-management.ts">
{`import * as tf from '@tensorflow/tfjs';

// BAD: leaks memory every call; tensors accumulate on the GPU
function leakyPredict(model: tf.LayersModel, data: number[]) {
  const input = tf.tensor2d([data]);           // allocated on GPU
  const output = model.predict(input);          // allocated on GPU
  return (output as tf.Tensor).dataSync()[0];   // input and output NEVER freed
}

// GOOD: tf.tidy() cleans up all tensors created inside it
function cleanPredict(model: tf.LayersModel, data: number[]) {
  return tf.tidy(() => {
    const input = tf.tensor2d([data]);
    const output = model.predict(input) as tf.Tensor;
    return output.dataSync()[0];  // scalar return value (not a tensor) passes through
  });
  // input and output tensors automatically disposed here
}

// Monitor memory usage, like checking DevTools memory tab
console.log(tf.memory());
// { numTensors: 47, numDataBuffers: 47, numBytes: 12800, unreliable: false }`}
</CodeBlock>

If that `numTensors` figure climbs while the dashboard idles, something escaped a `tidy()`.

One more dial before the wiring: `dataSync()` blocks the JavaScript thread while the GPU read completes. At this model's scale that stall is invisible, which is why the code above uses it freely. On bigger outputs, or inside a loop that also has UI to paint, prefer `await tensor.data()`: it returns a Promise and yields to the event loop while the transfer happens. The catch is that `tf.tidy()` cannot wrap async code, so the async pattern splits in two: do the tensor math inside `tidy()` and return the tensor, then `await tensor.data()` outside and `dispose()` it yourself.

## Wiring It into the Dashboard

<CodeBlock language="typescript" title="dashboard-integration.ts">
{`import * as tf from '@tensorflow/tfjs';

interface SignalReading {
  timestamp: number;
  amplitude: number;
  frequency: number;
  phase: number;
}

interface Classification {
  class: 'anomaly' | 'noise';
  confidence: number;
  latencyMs: number;
}

// Real-time pipeline: process each signal as it arrives
function createInferencePipeline(model: tf.LayersModel) {
  return function processSignal(reading: SignalReading): Classification {
    const start = performance.now();

    const result = tf.tidy(() => {
      const normalized = preprocess(reading);
      const input = tf.tensor2d([normalized]);
      const output = model.predict(input) as tf.Tensor;
      return output.dataSync()[0];
    });

    return {
      class: result > 0.5 ? 'anomaly' : 'noise',
      confidence: result,
      latencyMs: performance.now() - start,
    };
  };
}

// Usage: exactly like setting up an event listener
const pipeline = createInferencePipeline(model);

// Process each signal reading as it arrives (1Hz)
signalFeed.addEventListener('reading', (event: CustomEvent<SignalReading>) => {
  const classification = pipeline(event.detail);
  updateDashboard(classification);
  // Typical: { class: 'noise', confidence: 0.12, latencyMs: 3.4 }
});`}
</CodeBlock>
