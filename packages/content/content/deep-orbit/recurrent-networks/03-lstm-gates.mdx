      import * as tf from '@tensorflow/tfjs';

      const sigmoid = (z) => 1 / (1 + Math.exp(-z));

      function lstmStep(prevHidden, prevCell, x, weights) {
        const preact = (row, j) => row[0] * prevHidden[j] + row[1] * x[j] + row[2];
        const newCell = prevCell.map((c, j) => {
          const forget = sigmoid(preact(weights.forget[j], j));
          const store = sigmoid(preact(weights.input[j], j));
          const candidate = Math.tanh(preact(weights.candidate[j], j));
          return forget * c + store * candidate;
        });
        const newHidden = newCell.map((c, j) => sigmoid(preact(weights.output[j], j)) * Math.tanh(c));
        return { newCell, newHidden };
      }

      function buildLstmCell(timeSteps) {
        const model = tf.sequential();
        const cell = tf.layers.lstm({ units: 1, inputShape: [timeSteps, 1], recurrentActivation: 'sigmoid', returnSequences: true });
        model.add(cell);
        const [kernel, recurrentKernel, biasTensor] = model.getWeights();
        const inputColumns = Array.from(kernel.dataSync());
        const recurrentColumns = Array.from(recurrentKernel.dataSync());
        const biases = Array.from(biasTensor.dataSync());
        const gateRow = (column) => [recurrentColumns[column], inputColumns[column], biases[column]];
        return {
          model,
          weights: {
            input: [gateRow(0)],
            forget: [gateRow(1)],
            candidate: [gateRow(2)],
            output: [gateRow(3)],
          },
        };
      }

## Three Gates, One Cell

<BridgeCallout bridge={frontmatter.concept.bridge} />

<CodeBlock language="typescript" title="lstm-gates.ts">
{`// LSTM Cell: three gates controlling information flow

// 1. FORGET GATE: what to discard from memory
//    forgetGate = sigmoid(W_f · [prevHidden, input] + b_f)
//    Output: values between 0 (forget completely) and 1 (keep fully)

// 2. INPUT GATE: what new information to store
//    inputGate = sigmoid(W_i · [prevHidden, input] + b_i)
//    candidate = tanh(W_c · [prevHidden, input] + b_c)
//    newCell = forgetGate * prevCell + inputGate * candidate

// 3. OUTPUT GATE: what to expose as hidden state
//    outputGate = sigmoid(W_o · [prevHidden, input] + b_o)
//    newHidden = outputGate * tanh(newCell)

// The cell state is only scaled and added to at each step, never pushed
// through a weight matrix and a tanh the way the simple RNN's hidden
// state is. Walking backward along that path, the gradient picks up
// the forget gate factor per step on the direct cell-state path.
// The full gradient also has paths through the gates. A gate that learns to sit
// at 0.99 still leaves 0.37 of the signal 100 steps later; at 0.9 only
// 0.000027 survives, and a gate that learns to close cuts that
// dimension off on purpose. The decay is learned per dimension here,
// not forced by the wiring`}
</CodeBlock>

## Building an LSTM in TF.js

<CodeBlock language="typescript" title="lstm-model.ts">
{`import * as tf from '@tensorflow/tfjs';

// LSTM: a gated recurrent layer; compare its quality and cost on your task.
const model = tf.sequential();
model.add(tf.layers.lstm({
  units: 64,              // hidden state size
  inputShape: [80, 3],    // the SAME 80-step window the SimpleRNN failed on
  recurrentActivation: 'sigmoid', // match the equations above
  returnSequences: false,  // only need final output for classification
}));
model.add(tf.layers.dense({ units: 1, activation: 'sigmoid' }));

model.compile({ optimizer: 'adam', loss: 'binaryCrossentropy', metrics: ['accuracy'] });

// Small execution drill, not evidence of long-range learning. These
// independent random labels do not define a useful prediction task.
// The story's archive result below is fictional, not measured by this fit.
const x = tf.randomNormal([10, 80, 3]);  // 10 samples, 80 time steps
const y = tf.tidy(() => tf.randomUniform([10, 1]).round());

const history = await model.fit(x, y, {
  epochs: 1,
  validationSplit: 0.2,
  callbacks: {
    onEpochEnd: (epoch, logs) => {
      if (epoch % 5 === 0) {
        console.log(\`Epoch \${epoch}: loss=\${logs?.loss.toFixed(4)}, val_loss=\${logs?.val_loss.toFixed(4)}\`);
      }
    },
  },
});

x.dispose();
y.dispose();
model.dispose();`}
</CodeBlock>

`model.layers[0].getWeights()` returns three tensors: an input kernel [features, 4 * units], a recurrent kernel [units, 4 * units], and a bias [4 * units]. Each holds four groups of weights, in this order: input, forget, candidate, output.

Read matching columns from the two kernels and the bias to build each gate’s weights. With `unitForgetBias: true`, the forget group, the second quarter of the bias, starts at one. The other groups start at zero.

Check the gates’ activation too. TensorFlow.js 4.22 defaults `recurrentActivation` to `hardSigmoid`: `max(0, min(1, 0.2 * z + 0.5))`. The candidate and cell output still use `tanh`.

Our handwritten step uses sigmoid, so set `recurrentActivation: 'sigmoid'` on the layer too. Otherwise, the calculations differ. With matching activations, this small test agrees to about seven decimal places.

## GRU: The Lighter Alternative

<CodeBlock language="typescript" title="gru-comparison.ts">
{`import * as tf from '@tensorflow/tfjs';

// GRU (Gated Recurrent Unit): one gate fewer, and a quarter fewer weights
// Two gates instead of three: update gate + reset gate

// 1. UPDATE GATE: one value per dimension, doing the forget gate's job
//    and the input gate's job at once
//    update = sigmoid(W_z · [prevHidden, input] + b_z)
//    newHidden = update * prevHidden + (1 - update) * candidate
//    Keeping 70% of the old state means taking 30% of the new candidate

// 2. RESET GATE: how much of the previous hidden state the candidate
//    is allowed to see
//    reset = sigmoid(W_r · [prevHidden, input] + b_r)
//    candidate = tanh(W_c · [reset * prevHidden, input] + b_c)

// There is no separate cell state, so there is no output gate: the hidden
// state is the only thing the cell carries, and it leaves the cell whole

const gruModel = tf.sequential();
gruModel.add(tf.layers.gru({
  units: 64,
  recurrentActivation: 'sigmoid', // match the gate equations above
  inputShape: [80, 3],
}));
gruModel.add(tf.layers.dense({ units: 1, activation: 'sigmoid' }));

// Comparison:
// LSTM: 4 sets of weight matrices: forget, input, and output gates plus the
//       candidate state
// GRU:  3 sets: reset gate, update gate, candidate state, which is exactly
//       25% fewer weights: 13,056 against the LSTM's 17,408 at these settings
// Which of the two scores better on a given series is a question you measure,
// so compare both on the same data and evaluation procedure.
gruModel.dispose();`}
</CodeBlock>
